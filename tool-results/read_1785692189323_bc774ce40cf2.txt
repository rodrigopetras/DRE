   116→// ─── Formatters ───
   117→function brl(value: number) {
   118→  return new Intl.NumberFormat('pt-BR', { style: 'currency', currency: 'BRL' }).format(value || 0)
   119→}
   120→function pct(value: number) {
   121→  if (!isFinite(value) || isNaN(value)) return '0,00%'
   122→  return new Intl.NumberFormat('pt-BR', { style: 'percent', minimumFractionDigits: 2, maximumFractionDigits: 2 }).format(value || 0)
   123→}
   124→function safeDiv(a: number, b: number) {
   125→  return b === 0 ? 0 : a / b
   126→}

   128→// ─── DRE Calculations ───
   129→function calcDRE(d: DREData) {
   130→  const receitaBruta = d.prestacaoNf + d.honorariosSucumbenciais
   131→  const receitaLiquida = receitaBruta - d.impostosCpp
   132→  const despesasVariaveis =
   133→    d.combustivel + d.insumos + d.materialEscritorio + d.servicosContratados +
   134→    d.materiaisUsoConsumo + d.taxas + d.tarifas + d.energiaEletrica + d.agua
   135→  const despesasFixas =
   136→    d.aluguel + d.salarios + d.fgts + d.prolabore + d.bonus +
   137→    d.inss + d.iptu + d.telefoneInternet + d.valeTransporte + d.planoSaude
   138→  const resultadoLiquido = receitaLiquida - despesasVariaveis - despesasFixas
   139→  return { receitaBruta, receitaLiquida, despesasVariaveis, despesasFixas, resultadoLiquido }
   140→}

   142→// ─── Number Input Component ───
   143→function NumInput({ label, value, onChange, prefix }: {
   144→  label: string; value: number; onChange: (v: number) => void; prefix?: string
   145→}) {
   146→  const [editing, setEditing] = useState(false)
   147→  const [local, setLocal] = useState('')
   148→
   149→  const formattedValue = value === 0 ? '' : brl(value)
   150→
   151→  const commit = () => {
   152→    setEditing(false)
   153→    const parsed = parseFloat(local.replace(/\./g, '').replace(',', '.')) || 0
   154→    onChange(parsed)
   155→  }
   156→
   157→  return (
   158→    <div className="flex items-center justify-between py-1.5 group">
   159→      <label className="text-sm text-muted-foreground flex items-center gap-2">
   160→        {label}
   161→      </label>
   162→      <div className="flex items-center gap-1 w-44 justify-end">
   163→        {prefix && <span className="text-xs text-muted-foreground">{prefix}</span>}
   164→        <Input
   165→          type="text"
   166→          inputMode="decimal"
   167→          className={`h-8 text-right text-sm font-mono ${editing ? 'ring-2 ring-primary/30' : ''} transition-all`}
   168→          value={editing ? local : (value === 0 ? '' : brl(value))}
   169→          onFocus={() => { setEditing(true); setLocal(value === 0 ? '' : value.toFixed(2).replace('.', ',')) }}
   170→          onBlur={commit}
   171→          onChange={(e) => {
   172→            setLocal(e.target.value)
   173→            const parsed = parseFloat(e.target.value.replace(/\./g, '').replace(',', '.'))
   174→            if (!isNaN(parsed)) onChange(parsed)
   175→          }}
   176→          onKeyDown={(e) => e.key === 'Enter' && commit()}
   177→          placeholder="0,00"
   178→        />
   179→      </div>
   180→    </div>
   181→  )
   182→}
