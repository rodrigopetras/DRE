    22→    22→interface ExcelCellLike {
    23→    23→  value: unknown
    24→    24→  font?: { color?: { argb?: string } }
    25→    25→}
    26→    26→
    27→    27→/**
    28→    28→ * Detect if an Excel cell's font color is red.
    29→    29→ * Red = alpha byte 0xFF (high byte), R > 150, G < 100, B < 100.
    30→    30→ */
    31→    31→function isRedFont(cell?: ExcelCellLike): boolean {
    32→    32→  if (!cell?.font?.color?.argb) return false
    33→    33→  const argb = cell.font.color.argb
    34→    34→  if (argb.length < 8) return false
    35→    35→  const alpha = parseInt(argb.substring(0, 2), 16)
    36→    36→  if (alpha !== 0xFF) return false
    37→    37→  const r = parseInt(argb.substring(2, 4), 16)
    38→    38→  const g = parseInt(argb.substring(4, 6), 16)
    39→    39→  const b = parseInt(argb.substring(6, 8), 16)
    40→    40→  return r > 150 && g < 100 && b < 100
    41→    41→}
    42→    42→
    43→    43→/**
    44→    44→ * Parse any ExcelJS cell value into a number.
    45→    45→ * Handles: number, formula, RichText, Hyperlink, string, Date.
    46→    46→ * Parenthesized values like (780,00) are converted to negative.
    47→    47→ * Red font color also makes the value negative.
    48→    48→ */
    49→    49→function parseCellToNumber(cellValue: unknown, cell?: ExcelCellLike): number {
    50→    50→  let result = 0
    51→    51→
    52→    52→  if (typeof cellValue === 'number') {
    53→    53→    result = cellValue
    54→    54→  }
    55→    55→  else if (cellValue && typeof cellValue === 'object' && 'result' in cellValue) {
    56→    56→    const formulaResult = (cellValue as any).result
    57→    57→    if (typeof formulaResult === 'number') result = formulaResult
    58→    58→    else if (typeof formulaResult === 'string') result = parseTextToNumber(formulaResult)
    59→    59→  }
    60→    60→  else if (cellValue && typeof cellValue === 'object' && 'richText' in cellValue) {
    61→    61→    const parts = (cellValue as any).richText as Array<{ text: string }>
    62→    62→    const text = parts.map(p => p.text).join('')
    63→    63→    result = parseTextToNumber(text)
    64→    64→  }
    65→    65→  else if (cellValue && typeof cellValue === 'object' && 'text' in cellValue && 'hyperlink' in cellValue) {
    66→    66→    result = parseTextToNumber((cellValue as any).text)
    67→    67→  }
    68→    68→  else if (typeof cellValue === 'string') {
    69→    69→    result = parseTextToNumber(cellValue)
    70→    70→  }
    71→    71→
    72→    72→  if (result > 0 && isRedFont(cell)) {
    73→    73→    result = -result
    74→    74→  }
    75→    75→
    76→    76→  return result
    77→    77→}
    78→    78→
    79→    79→/**
    80→    80→ * Parse a text string like "1.234,56" or "(780,00)" into a number.
    81→    81→ * Parenthesized values become negative.
    82→    82→ */
    83→    83→function parseTextToNumber(text: string): number {
    84→    84→  const trimmed = text.trim()
    85→    85→  if (!trimmed) return 0
    86→    86→  const parenMatch = trimmed.match(/^\((.+)\)$/)
    87→    87→  let numStr: string
    88→    88→  let sign = 1
    89→    89→  if (parenMatch) {
    90→    90→    numStr = parenMatch[1]
    91→    91→    sign = -1
    92→    92→  } else {
    93→    93→    numStr = trimmed
    94→    94→    if (numStr.startsWith('-')) {
    95→    95→      sign = -1
    96→    96→      numStr = numStr.substring(1)
    97→    97→    }
    98→    98→  }
    99→    99→  const cleaned = numStr.replace(/\./g, '').replace(',', '.')
   100→   100→  const parsed = parseFloat(cleaned)
   101→   101→  if (isNaN(parsed)) return 0
   102→   102→  return sign * Math.abs(parsed)
   103→   103→}

   107→   107→function excelSerialToDate(serial: number): Date {
   108→   108→  if (serial < 1) return new Date(NaN)
   109→   109→  const epoch = new Date(1899, 11, 30)
   110→   110→  const msPerDay = 24 * 60 * 60 * 1000
   111→   111→  return new Date(epoch.getTime() + serial * msPerDay)
   112→   112→}

   114→   114→/** Remove accents/diacritics from a string for comparison */
   115→   115→function normalizeAccents(str: string): string {
   116→   116→  return str.normalize('NFD').replace(/[\u0300-\u036f]/g, '').toLowerCase().trim()

   129→   129→const MONTH_NAMES_PT: Record<string, number> = {

   136→   136→function parseRowDate(rawValue: unknown): ParsedDateInfo | null {
   137→   137→  if (!rawValue) return null
   138→   138→
   139→   139→  if (rawValue instanceof Date && !isNaN(rawValue.getTime())) {
   140→   140→    return {
   141→   141→      month: rawValue.getMonth() + 1,
   142→   142→      year: rawValue.getFullYear(),
   143→   143→      dateStr: rawValue.toISOString().split('T')[0],
   144→   144→    }
   145→   145→  }
   146→   146→
   147→   147→  if (typeof rawValue === 'number' && rawValue > 25569 && rawValue < 100000) {
   148→   148→    const d = excelSerialToDate(rawValue)
   149→   149→    if (!isNaN(d.getTime())) {
   150→   150→      return {
   151→   151→        month: d.getMonth() + 1,
   152→   152→        year: d.getFullYear(),
   153→   153→        dateStr: d.toISOString().split('T')[0],
   154→   154→      }
   155→   155→    }
   156→   156→  }
   157→   157→
   158→   158→  if (typeof rawValue === 'string') {
   159→   159→    const s = rawValue.trim()
   160→   160→    if (!s) return null
   161→   161→
   162→   162→    // YYYY-MM-DD
   163→   163→    let m = s.match(/^(\d{4})-(\d{1,2})-(\d{1,2})/)
   164→   164→    if (m) {
   165→   165→      const yr = parseInt(m[1]); const mo = parseInt(m[2]); const dy = parseInt(m[3])
   166→   166→      if (mo >= 1 && mo <= 12 && dy >= 1 && dy <= 31) {
   167→   167→        return { month: mo, year: yr, dateStr: `${yr}-${String(mo).padStart(2, '0')}-${String(dy).padStart(2, '0')}` }
   168→   168→      }
   169→   169→    }
   170→   170→
   171→   171→    // DD/MM/YYYY or MM/DD/YYYY
   172→   172→    m = s.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/)
   173→   173→    if (m) {
   174→   174→      const a = parseInt(m[1]); const b = parseInt(m[2]); const yr = parseInt(m[3])
   175→   175→      let mo: number, dy: number
   176→   176→      if (a > 12) { dy = a; mo = b }
   177→   177→      else if (b > 12) { mo = a; dy = b }
   178→   178→      else { dy = a; mo = b }
   179→   179→      if (mo >= 1 && mo <= 12 && dy >= 1 && dy <= 31) {
   180→   180→        return { month: mo, year: yr, dateStr: `${yr}-${String(mo).padStart(2, '0')}-${String(dy).padStart(2, '0')}` }
   181→   181→      }
   182→   182→    }
   183→   183→
   184→   184→    // DD-MM-YYYY
   185→   185→    m = s.match(/^(\d{1,2})-(\d{1,2})-(\d{4})$/)
   186→   186→    if (m) {
   187→   187→      const a = parseInt(m[1]); const b = parseInt(m[2]); const yr = parseInt(m[3])
   188→   188→      let mo: number, dy: number
   189→   189→      if (a > 12) { dy = a; mo = b }
   190→   190→      else if (b > 12) { mo = a; dy = b }
   191→   191→      else { dy = a; mo = b }
   192→   192→      if (mo >= 1 && mo <= 12 && dy >= 1 && dy <= 31) {
   193→   193→        return { month: mo, year: yr, dateStr: `${yr}-${String(mo).padStart(2, '0')}-${String(dy).padStart(2, '0')}` }
   194→   194→      }
   195→   195→    }
   196→   196→
   197→   197→    // DD/MM/YY
   198→   198→    m = s.match(/^(\d{1,2})\/(\d{1,2})\/(\d{2})$/)
   199→   199→    if (m) {
   200→   200→      const a = parseInt(m[1]); const b = parseInt(m[2])
   201→   201→      let yr = parseInt(m[3])
   202→   202→      yr = yr < 100 ? 2000 + yr : yr
   203→   203→      let mo: number, dy: number
   204→   204→      if (a > 12) { dy = a; mo = b }
   205→   205→      else if (b > 12) { mo = a; dy = b }
   206→   206→      else { dy = a; mo = b }
   207→   207→      if (mo >= 1 && mo <= 12 && dy >= 1 && dy <= 31) {
   208→   208→        return { month: mo, year: yr, dateStr: `${yr}-${String(mo).padStart(2, '0')}-${String(dy).padStart(2, '0')}` }
   209→   209→      }
   210→   210→    }
   211→   211→
   212→   212→    // Month name in Portuguese (e.g. "Janeiro 2025", "Jan/2025", "janeiro")
   213→   213→    const normS = normalizeAccents(s)
   214→   214→    for (const [monthName, monthNum] of Object.entries(MONTH_NAMES_PT)) {
   215→   215→      const normName = normalizeAccents(monthName)
   216→   216→      const regex = new RegExp(`\\b${normName}\\b`, 'i')
   217→   217→      if (regex.test(normS)) {
   218→   218→        // Try to extract year from the string
   219→   219→        const yearMatch = s.match(/(\d{4})/)
   220→   220→        const yr = yearMatch ? parseInt(yearMatch[1]) : new Date().getFullYear()
   221→   221→        return { month: monthNum, year: yr, dateStr: `${yr}-${String(monthNum).padStart(2, '0')}-01` }
   222→   222→      }
   223→   223→    }
   224→   224→
   225→   225→    // MM/YYYY (just month and year)
   226→   226→    m = s.match(/^(\d{1,2})\/(\d{4})$/)
   227→   227→    if (m) {
   228→   228→      const mo = parseInt(m[1]); const yr = parseInt(m[2])
   229→   229→      if (mo >= 1 && mo <= 12) {
   230→   230→        return { month: mo, year: yr, dateStr: `${yr}-${String(mo).padStart(2, '0')}-01` }
   231→   231→      }
   232→   232→    }
   233→   233→  }
   234→   234→
   235→   235→  return null
   236→   236→}

   277→   277→/**
   278→   278→ * RF-04: Detect if a description is a structural title and return its context.
   279→   279→ * Returns { type: 'section' | 'field', fieldKey?, sectionKey? } or null.
   280→   280→ */
   281→   281→function detectStructuralTitle(description: string): { type: 'section' | 'field'; fieldKey?: string; sectionKey?: string } | null {
   282→   282→  const norm = normalizeAccents(description)
   283→   283→
   284→   284→  // Check for section-level matches first
   285→   285→  for (const entry of STRUCTURAL_TITLES) {
   286→   286→    for (const pattern of entry.patterns) {
   287→   287→      if (norm === normalizeAccents(pattern) || norm.includes(normalizeAccents(pattern))) {
   288→   288→        if (entry.sectionKey && !entry.fieldKey) {
   289→   289→          return { type: 'section', sectionKey: entry.sectionKey }
   290→   290→        }
   291→   291→        if (entry.fieldKey) {
   292→   292→          return { type: 'field', fieldKey: entry.fieldKey, sectionKey: DRE_FIELD_SECTIONS[entry.fieldKey] }
   293→   293→        }
   294→   294→      }
   295→   295→    }
   296→   296→  }
   297→   297→  return null
   298→   298→}

   300→   300→/**
   301→   301→ * When a structural section title row has a value, map it to a reasonable DRE field.
   302→   302→ * For receita section: maps to 'prestacaoNf' (main revenue field for law firms).
   303→   303→ * For desp sections: maps to the first field in that section.
   304→   304→ */
   305→   305→function sectionTitleToField(sectionKey: string): string {
   306→   306→  if (sectionKey === 'receita') return 'prestacaoNf'
   307→   307→  // For expense sections, find the first field in that section
   308→   308→  const sectionFields = Object.entries(DRE_FIELD_SECTIONS)
   309→   309→    .filter(([, section]) => section === sectionKey)
   310→   310→    .map(([key]) => key)
   311→   311→  return sectionFields[0] || 'naoIdentificado'
   312→   312→}

   314→   314→/**
   315→   315→ * RF-04: Given a section context, try to auto-classify a description into the first matching field in that section.
   316→   316→ */
   317→   317→function autoClassifyInSection(description: string, sectionKey: string): string | null {
   318→   318→  const norm = normalizeAccents(description)
   319→   319→  const sectionFields = Object.entries(DRE_FIELD_SECTIONS)
   320→   320→    .filter(([, section]) => section === sectionKey)
   321→   321→    .map(([key]) => key)
   322→   322→
   323→   323→  for (const fieldKey of sectionFields) {
   324→   324→    const label = DRE_FIELD_LABELS[fieldKey]
   325→   325→    if (!label) continue
   326→   326→    const normLabel = normalizeAccents(label)
   327→   327→    // Check if description contains field label keywords
   328→   328→    const labelWords = normLabel.split(/\s+/).filter(w => w.length > 2)
   329→   329→    for (const word of labelWords) {
   330→   330→      if (norm.includes(word)) return fieldKey
   331→   331→    }
   332→   332→  }
   333→   333→  return null
   334→   334→}

   336→   336→/**
   337→   337→ * Try to match a category string from column D against DRE field labels.
   338→   338→ */
   339→   339→function classifyByCategory(category: string): string | null {
   340→   340→  if (!category || !category.trim()) return null
   341→   341→  const norm = normalizeAccents(category)
   342→   342→
   343→   343→  for (const key of Object.keys(DRE_FIELD_LABELS)) {
   344→   344→    if (normalizeAccents(key) === norm) return key
   345→   345→  }
   346→   346→  for (const [key, value] of Object.entries(DRE_FIELD_LABELS)) {
   347→   347→    if (normalizeAccents(value) === norm) return key
   348→   348→  }
   349→   349→  return null
   350→   350→}

   354→   354→export async function POST(req: NextRequest) {
   355→   355→  try {
   356→   356→    const session = await getServerSession(authOptions)
   357→   357→    if (!session?.user?.id) {
   358→   358→      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
   359→   359→    }
   360→   360→
   361→   361→    const userId = (session.user as any).id
   362→   362→    const formData = await req.formData()
   363→   363→    const file = formData.get('file') as File
   364→   364→    const companyId = formData.get('companyId') as string
   365→   365→    const month = parseInt(formData.get('month') as string)
   366→   366→    const year = parseInt(formData.get('year') as string)
   367→   367→
   368→   368→    if (!file || !companyId || !month || !year) {
   369→   369→      return NextResponse.json({ error: 'Parâmetros incompletos' }, { status: 400 })
   370→   370→    }
   371→   371→
   372→   372→    const hasAccess = await verifyAccess(userId, companyId)
   373→   373→    if (!hasAccess) {
   374→   374→      return NextResponse.json({ error: 'Sem permissão' }, { status: 403 })
   375→   375→    }
   376→   376→
   377→   377→    const buffer = Buffer.from(await file.arrayBuffer())
   378→   378→    const workbook = new ExcelJS.Workbook()
   379→   379→    await workbook.xlsx.load(buffer)
   380→   380→
   381→   381→    if (workbook.worksheets.length === 0) {
   382→   382→      return NextResponse.json({ error: 'Planilha vazia' }, { status: 400 })
   383→   383→    }
   384→   384→
   385→   385→    // ─── Step 1: Look for "Caixa YYYY" worksheet to extract Receita Bruta from C3 ───
   386→   386→    let receitaBrutaValue = 0
   387→   387→    let receitaBrutaYear = year
   388→   388→    const caixaPattern = /^caixa\s+(\d{4})$/i
   389→   389→    for (const ws of workbook.worksheets) {
   390→   390→      const match = ws.name?.trim().match(caixaPattern)
   391→   391→      if (match) {
   392→   392→        receitaBrutaYear = parseInt(match[1])
   393→   393→        // Read specifically cell C3 (row 3, column 3)
   394→   394→        if (ws.rowCount >= 3 && ws.getColumn(3).values) {
   395→   395→          const cell = ws.getCell(3, 3)
   396→   396→          const val = parseCellToNumber(cell.value, { font: cell.font } as ExcelCellLike)
   397→   397→          if (val !== 0) {
   398→   398→            receitaBrutaValue = val
   399→   399→          }
   400→   400→        }
   401→   401→        break
   402→   402→      }
   403→   403→    }
   404→   404→
   405→   405→    const worksheet = workbook.worksheets[0]
   406→   406→    if (!worksheet) {
   407→   407→      return NextResponse.json({ error: 'Planilha vazia' }, { status: 400 })
   408→   408→    }
   409→   409→
   410→   410→    // RF-04: Track structural title context during parsing
   411→   411→    let currentSectionContext: string | null = null
   412→   412→    let currentFieldContext: string | null = null
   413→   413→
   414→   414→    const parsedRows: Array<{
   415→   415→      description: string
   416→   416→      date: string
   417→   417→      totalValue: number
   418→   418→      itemMonth: number
   419→   419→      itemYear: number
   420→   420→      dreField: string
   421→   421→    }> = []
   422→   422→
   423→   423→    let structuralTitlesFound = 0
   424→   424→
   425→   425→    worksheet.eachRow((row, rowNumber) => {
   426→   426→      if (rowNumber === 1) return
   427→   427→
   428→   428→      // Column A: Description
   429→   429→      const description = row.getCell(1).value?.toString()?.trim() || ''
   430→   430→      if (!description) return
   431→   431→
   432→   432→      // Column B: Date → parse to get month/year
   433→   433→      const rawDate = row.getCell(2).value
   434→   434→      const parsedDate = parseRowDate(rawDate)
   435→   435→      // If date is parsed, use it; otherwise default to selected month/year
   436→   436→      const itemMonth = parsedDate?.month ?? month
   437→   437→      const itemYear = parsedDate?.year ?? year
   438→   438→      const dateStr = parsedDate?.dateStr ?? ''
   439→   439→
   440→   440→      // Column C: Value
   441→   441→      const valueCell = row.getCell(3)
   442→   442→      const value = parseCellToNumber(valueCell.value, { font: valueCell.font } as ExcelCellLike)
   443→   443→
   444→   444→      // RF-04: Check if this row is a structural title
   445→   445→      const titleInfo = detectStructuralTitle(description)
   446→   446→
   447→   447→      if (titleInfo) {
   448→   448→        structuralTitlesFound++
   449→   449→        if (titleInfo.type === 'section') {
   450→   450→          currentSectionContext = titleInfo.sectionKey ?? null
   451→   451→          currentFieldContext = null
   452→   452→        } else if (titleInfo.type === 'field') {
   453→   453→          currentFieldContext = titleInfo.fieldKey
   454→   454→          currentSectionContext = titleInfo.sectionKey ?? null
   455→   455→        }
   456→   456→        // Structural titles with no value are skipped as data rows
   457→   457→        if (value === 0) return
   458→   458→      }
   459→   459→
   460→   460→      // Column D: Category (explicit auto-classification, takes priority)
   461→   461→      const categoryCell = row.getCell(4)
   462→   462→      let dreField = ''
   463→   463→      if (categoryCell && categoryCell.value) {
   464→   464→        const categoryText = String(categoryCell.value).trim()
   465→   465→        dreField = classifyByCategory(categoryText) || ''
   466→   466→      }
   467→   467→
   468→   468→      // RF-04: If no category, use structural context for auto-classification
   469→   469→      if (!dreField) {
   470→   470→        if (currentFieldContext) {
   471→   471→          dreField = currentFieldContext
   472→   472→        } else if (titleInfo?.type === 'section' && titleInfo.sectionKey) {
   473→   473→          // Section-level title with a value: map to a catch-all field in that section
   474→   474→          // e.g. "Receita Bruta" with value → naoIdentificado
   475→   475→          dreField = sectionTitleToField(titleInfo.sectionKey)
   476→   476→        } else if (currentSectionContext) {
   477→   477→          const autoField = autoClassifyInSection(description, currentSectionContext)
   478→   478→          if (autoField) dreField = autoField
   479→   479→        }
   480→   480→      }
   481→   481→
   482→   482→      if (value !== 0) {
   483→   483→        parsedRows.push({ description, date: dateStr, totalValue: value, itemMonth, itemYear, dreField })
   484→   484→      }
   485→   485→    })
   486→   486→
   487→   487→    if (parsedRows.length === 0 && receitaBrutaValue === 0) {
   488→   488→      return NextResponse.json(
   489→   489→        { error: 'Nenhum dado válido encontrado. Verifique: Descrição (A), Data (B), Valor (C).' },
   490→   490→        { status: 400 }
   491→   491→      )
   492→   492→    }
   493→   493→
   494→   494→    // Create uploaded file record (use month=0, year=0 to indicate multi-month file)
   495→   495→    const uploadedFile = await db.uploadedFile.create({
   496→   496→      data: { companyId, month: 0, year: 0, fileName: file.name },
   497→   497→    })
   498→   498→
   499→   499→    // If "Caixa YYYY" worksheet was found, add Receita Bruta as a qualification item
   500→   500→    // The auto-update below will inject it into the DRE
   501→   501→    let receitaBrutaInserted = false
   502→   502→    if (receitaBrutaValue !== 0) {
   503→   503→      await db.expenseQualification.create({
   504→   504→        data: {
   505→   505→          companyId,
   506→   506→          month: 1,
   507→   507→          year: receitaBrutaYear,
   508→   508→          description: 'Receita Bruta',
   509→   509→          date: `${receitaBrutaYear}-01-01`,
   510→   510→          totalValue: receitaBrutaValue,
   511→   511→          dreField: 'prestacaoNf',
   512→   512→          uploadedFileId: uploadedFile.id,
   513→   513→        },
   514→   514→      })
   515→   515→      receitaBrutaInserted = true
   516→   516→    }
   517→   517→
   518→   518→    // Save all other qualifications
   519→   519→    const qualifications = parsedRows.map((item) => ({
   520→   520→      companyId,
   521→   521→      month: item.itemMonth,
   522→   522→      year: item.itemYear,
   523→   523→      description: item.description,
   524→   524→      date: item.date,
   525→   525→      totalValue: item.totalValue,
   526→   526→      dreField: item.dreField,
   527→   527→      uploadedFileId: uploadedFile.id,
   528→   528→    }))
   529→   529→
   530→   530→    if (qualifications.length > 0) {
   531→   531→      await db.expenseQualification.createMany({ data: qualifications })
   532→   532→    }
   533→   533→
   534→   534→    const savedItems = await db.expenseQualification.findMany({
   535→   535→      where: { uploadedFileId: uploadedFile.id },
   536→   536→      orderBy: { createdAt: 'asc' },
   537→   537→    })
   538→   538→
   539→   539→    // Build distributed months list
   540→   540→    const groupedByMonth = new Map<string, number[]>()
   541→   541→    for (const row of parsedRows) {
   542→   542→      const key = `${row.itemMonth}-${row.itemYear}`
   543→   543→      if (!groupedByMonth.has(key)) groupedByMonth.set(key, [])
   544→   544→      groupedByMonth.get(key)!.push(row.itemMonth)
   545→   545→    }
   546→   546→    const distributedMonths = Array.from(groupedByMonth.keys())
   547→   547→      .map(key => {
   548→   548→        const [m, y] = key.split('-')
   549→   549→        return `${String(parseInt(m)).padStart(2, '0')}/${y}`
   550→   550→      })
   551→   551→      .sort((a, b) => {
   552→   552→        const [am, ay] = a.split('/').map(Number)
   553→   553→        const [bm, by] = b.split('/').map(Number)
   554→   554→        return ay !== by ? ay - by : am - bm
   555→   555→      })
   556→   556→
   557→   557→    // Count auto-classified items
   558→   558→    const autoClassified = savedItems.filter(i => i.dreField).length
   559→   559→
   560→   560→    // Auto-update DRE entries for months that have auto-classified items (especially C3/Receita Bruta)
   561→   561→    const autoClassifiedItems = savedItems.filter(i => i.dreField)
   562→   562→    const autoMonthKeys = new Set(autoClassifiedItems.map(i => `${i.month}-${i.year}`))
   563→   563→    let updatedDREForMonths: Record<string, any> = {}
   564→   564→    for (const monthKey of autoMonthKeys) {
   565→   565→      const [mStr, yStr] = monthKey.split('-')
   566→   566→      const m = parseInt(mStr)
   567→   567→      const y = parseInt(yStr)
   568→   568→      const monthItems = autoClassifiedItems.filter(i => i.month === m && i.year === y)
   569→   569→      
   570→   570→      const fieldTotals: Record<string, number> = {}
   571→   571→      for (const item of monthItems) {
   572→   572→        if (item.dreField && !item.dreField.startsWith('custom_')) {
   573→   573→          fieldTotals[item.dreField] = (fieldTotals[item.dreField] || 0) + item.totalValue
   574→   574→        }
   575→   575→      }
   576→   576→
   577→   577→      if (Object.keys(fieldTotals).length > 0) {
   578→   578→        const existing = await db.dREEntry.findUnique({
   579→   579→          where: { companyId_month_year: { companyId, month: m, year: y } },
   580→   580→        })
   581→   581→        if (existing) {
   582→   582→          // Merge: add to existing values
   583→   583→          const mergedData: Record<string, number> = {}
   584→   584→          for (const [field, total] of Object.entries(fieldTotals)) {
   585→   585→            mergedData[field] = ((existing as any)[field] || 0) + total
   586→   586→          }
   587→   587→          await db.dREEntry.update({
   588→   588→            where: { id: existing.id },
   589→   589→            data: mergedData,
   590→   590→          })
   591→   591→          // Fetch updated entry for response
   592→   592→          const updated = await db.dREEntry.findUnique({ where: { id: existing.id } })
   593→   593→          if (updated) {
   594→   594→            const { id: _id, companyId: _c, month: _m2, year: _y2, createdAt, updatedAt, company, ...fields } = updated as any
   595→   595→            updatedDREForMonths[monthKey] = fields
   596→   596→          }
   597→   597→        } else {
   598→   598→          const created = await db.dREEntry.create({
   599→   599→            data: { companyId, month: m, year: y, ...fieldTotals },
   600→   600→          })
   601→   601→          const { id: _id, companyId: _c, month: _m2, year: _y2, createdAt, updatedAt, company, ...fields } = created as any
   602→   602→          updatedDREForMonths[monthKey] = fields
   603→   603→        }
   604→   604→      }
   605→   605→    }
   606→   606→
   607→   607→    return NextResponse.json({
   608→   608→      items: savedItems.map(i => ({
   609→   609→        id: i.id, description: i.description, date: i.date,
   610→   610→        totalValue: i.totalValue, dreField: i.dreField,
   611→   611→        month: i.month, year: i.year,
   612→   612→      })),
   613→   613→      file: { id: uploadedFile.id, fileName: uploadedFile.fileName, uploadedAt: uploadedFile.uploadedAt },
   614→   614→      distributedMonths,
   615→   615→      structuralTitlesFound,
   616→   616→      autoClassified,
   617→   617→      updatedDREForMonths,
   618→   618→      receitaBrutaInserted,
   619→   619→      receitaBrutaValue,
   620→   620→      receitaBrutaYear,
   621→   621→    })
   622→   622→  } catch (error) {
   623→   623→    console.error('Expense qualification error:', error)
   624→   624→    return NextResponse.json({ error: 'Erro ao processar planilha de despesas' }, { status: 500 })
   625→   625→  }
   626→   626→}

   630→   630→export async function PUT(req: NextRequest) {
   631→   631→  try {
   632→   632→    const session = await getServerSession(authOptions)
   633→   633→    if (!session?.user?.id) {
   634→   634→      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
   635→   635→    }
   636→   636→
   637→   637→    const userId = (session.user as any).id
   638→   638→    const body = await req.json()
   639→   639→    const { companyId, month, year, items } = body
   640→   640→
   641→   641→    if (!companyId || !month || !year || !items) {
   642→   642→      return NextResponse.json({ error: 'Parâmetros incompletos' }, { status: 400 })
   643→   643→    }
   644→   644→
   645→   645→    const hasAccess = await verifyAccess(userId, companyId)
   646→   646→    if (!hasAccess) {
   647→   647→      return NextResponse.json({ error: 'Sem permissão' }, { status: 403 })
   648→   648→    }
   649→   649→
   650→   650→    // Update each item's dreField mapping in parallel
   651→   651→    const itemsToUpdate = items.filter((item: any) => item.id && item.dreField)
   652→   652→    await Promise.all(
   653→   653→      itemsToUpdate.map((item: any) =>
   654→   654→        db.expenseQualification.update({
   655→   655→          where: { id: item.id },
   656→   656→          data: { dreField: item.dreField },
   657→   657→        })
   658→   658→      )
   659→   659→    )
   660→   660→
   661→   661→    // Find all distinct month/year combos that have classified items for this company
   662→   662→    const allClassified = await db.expenseQualification.findMany({
   663→   663→      where: { companyId, dreField: { not: '' } },
   664→   664→      select: { month: true, year: true },
   665→   665→      distinct: ['month', 'year'],
   666→   666→    })
   667→   667→
   668→   668→    // Update DRE entries for ALL months that have classified items
   669→   669→    let currentMonthDreEntryId: string | null = null
   670→   670→    let currentMonthFieldTotals: Record<string, number> = {}
   671→   671→    for (const { month: m, year: y } of allClassified) {
   672→   672→      const monthQualifications = await db.expenseQualification.findMany({
   673→   673→        where: { companyId, month: m, year: y, dreField: { not: '' } },
   674→   674→      })
   675→   675→
   676→   676→      const fieldTotals: Record<string, number> = {}
   677→   677→      const customFieldTotals: Record<string, number> = {}
   678→   678→      for (const q of monthQualifications) {
   679→   679→        if (q.dreField) {
   680→   680→          if (q.dreField.startsWith('custom_')) {
   681→   681→            const customFieldId = q.dreField.replace('custom_', '')
   682→   682→            customFieldTotals[customFieldId] = (customFieldTotals[customFieldId] || 0) + q.totalValue
   683→   683→          } else {
   684→   684→            fieldTotals[q.dreField] = (fieldTotals[q.dreField] || 0) + q.totalValue
   685→   685→          }
   686→   686→        }
   687→   687→      }
   688→   688→
   689→   689→      // Save field totals for the currently selected month
   690→   690→      if (m === month && y === year) {
   691→   691→        currentMonthFieldTotals = fieldTotals
   692→   692→      }
   693→   693→
   694→   694→      const existing = await db.dREEntry.findUnique({
   695→   695→        where: { companyId_month_year: { companyId, month: m, year: y } },
   696→   696→      })
   697→   697→
   698→   698→      const updateData: Record<string, number> = {}
   699→   699→      for (const [field, total] of Object.entries(fieldTotals)) {
   700→   700→        updateData[field] = total
   701→   701→      }
   702→   702→
   703→   703→      let dreEntryId: string | null = null
   704→   704→      if (Object.keys(updateData).length > 0 || Object.keys(customFieldTotals).length > 0) {
   705→   705→        if (existing) {
   706→   706→          dreEntryId = existing.id
   707→   707→          if (Object.keys(updateData).length > 0) {
   708→   708→            await db.dREEntry.update({
   709→   709→              where: { id: existing.id },
   710→   710→              data: updateData,
   711→   711→            })
   712→   712→          }
   713→   713→        } else {
   714→   714→          const created = await db.dREEntry.create({
   715→   715→            data: { companyId, month: m, year: y, ...updateData },
   716→   716→          })
   717→   717→          dreEntryId = created.id
   718→   718→        }
   719→   719→      } else {
   720→   720→        dreEntryId = existing?.id || null
   721→   721→      }
   722→   722→
   723→   723→      // Handle custom field totals (with sign correction)
   724→   724→      if (dreEntryId && Object.keys(customFieldTotals).length > 0) {
   725→   725→        const customFieldIds = Object.keys(customFieldTotals)
   726→   726→        const customFields = await db.customDREField.findMany({
   727→   727→          where: { id: { in: customFieldIds } },
   728→   728→        })
   729→   729→        const fieldSignMap: Record<string, string> = {}

   733→   733→          Object.entries(customFieldTotals).map(([customFieldId, total]) => {

   736→   736→            return db.customDREValue.upsert({
   737→   737→              where: { customFieldId_dREEntryId: { customFieldId, dREEntryId } },
   738→   738→              create: { customFieldId, dREEntryId, value: adjustedTotal },
   739→   739→              update: { value: adjustedTotal },
   740→   740→            })
   741→   741→          })
   742→   742→        )
   743→   743→      }
   744→   744→
   745→   745→      // Track the entry ID for the currently selected month
   746→   746→      if (m === month && y === year) {
   747→   747→        currentMonthDreEntryId = dreEntryId
   748→   748→      }
   749→   749→    }
   750→   750→
   751→   751→    // RF-07: Fetch the updated DRE entry to return for reactive sync
   752→   752→    let updatedDREData: Record<string, any> | null = null
   753→   753→    if (currentMonthDreEntryId) {
   754→   754→      const updatedEntry = await db.dREEntry.findUnique({ where: { id: currentMonthDreEntryId } })
   755→   755→      if (updatedEntry) {
   756→   756→        const { id, companyId: _cid, month: _m, year: _y, createdAt, updatedAt, company, ...fields } = updatedEntry as any
   757→   757→        updatedDREData = fields
   758→   758→
   759→   759→        // Also fetch custom field values
   760→   760→        const customValues = await db.customDREValue.findMany({
   761→   761→          where: { dREEntryId: currentMonthDreEntryId },
   762→   762→          include: { customField: true },
   763→   763→        })
   764→   764→        if (customValues.length > 0) {
   765→   765→          updatedDREData._customFieldValues = {}
   766→   766→          for (const cv of customValues) {
   767→   767→            const sign = (cv.customField as any).sign || '+'
   768→   768→            updatedDREData._customFieldValues[cv.customFieldId] = sign === '-' ? -cv.value : cv.value
   769→   769→          }
   770→   770→        }
   771→   771→      }
   772→   772→    }
   773→   773→
   774→   774→    return NextResponse.json({
   775→   775→      success: true,
   776→   776→      fieldTotals: currentMonthFieldTotals,
   777→   777→      dreEntryId: currentMonthDreEntryId,
   778→   778→      updatedDREData, // RF-07: reactive sync payload
   779→   779→    })
   780→   780→  } catch (error) {
   781→   781→    console.error('Error updating qualification:', error)
   782→   782→    return NextResponse.json({ error: 'Erro ao atualizar qualificação' }, { status: 500 })
   783→   783→  }
   784→   784→}

   838→   838→export async function GET(req: NextRequest) {
