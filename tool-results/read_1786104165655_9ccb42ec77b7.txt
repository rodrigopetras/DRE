    10→async function verifyAccess(userId: string, companyId: string) {
    11→  const company = await db.company.findUnique({ where: { id: companyId } })
    12→  if (!company) return false
    13→  if (company.userId === userId) return true
    14→  const share = await db.companyShare.findUnique({
    15→    where: { companyId_userId: { companyId, userId } },
    16→  })
    17→  return share?.status === 'accepted'
    18→}

    22→interface ExcelCellLike {
    23→  value: unknown
    24→  font?: { color?: { argb?: string } }
    25→}
    26→
    27→/**
    28→ * Detect if an Excel cell's font color is red.
    29→ * Red = alpha byte 0xFF (high byte), R > 150, G < 100, B < 100.
    30→ */
    31→function isRedFont(cell?: ExcelCellLike): boolean {
    32→  if (!cell?.font?.color?.argb) return false
    33→  const argb = cell.font.color.argb
    34→  if (argb.length < 8) return false
    35→  const alpha = parseInt(argb.substring(0, 2), 16)
    36→  if (alpha !== 0xFF) return false
    37→  const r = parseInt(argb.substring(2, 4), 16)
    38→  const g = parseInt(argb.substring(4, 6), 16)
    39→  const b = parseInt(argb.substring(6, 8), 16)
    40→  return r > 150 && g < 100 && b < 100
    41→}
    42→
    43→/**
    44→ * Parse any ExcelJS cell value into a number.
    45→ * Handles: number, formula, RichText, Hyperlink, string, Date.
    46→ * Parenthesized values like (780,00) are converted to negative.
    47→ * Red font color also makes the value negative.
    48→ */
    49→function parseCellToNumber(cellValue: unknown, cell?: ExcelCellLike): number {
    50→  let result = 0
    51→
    52→  if (typeof cellValue === 'number') {
    53→    result = cellValue
    54→  }
    55→  else if (cellValue && typeof cellValue === 'object' && 'result' in cellValue) {
    56→    const formulaResult = (cellValue as any).result
    57→    if (typeof formulaResult === 'number') result = formulaResult
    58→    else if (typeof formulaResult === 'string') result = parseTextToNumber(formulaResult)
    59→  }
    60→  else if (cellValue && typeof cellValue === 'object' && 'richText' in cellValue) {
    61→    const parts = (cellValue as any).richText as Array<{ text: string }>
    62→    const text = parts.map(p => p.text).join('')
    63→    result = parseTextToNumber(text)
    64→  }
    65→  else if (cellValue && typeof cellValue === 'object' && 'text' in cellValue && 'hyperlink' in cellValue) {
    66→    result = parseTextToNumber((cellValue as any).text)
    67→  }
    68→  else if (typeof cellValue === 'string') {
    69→    result = parseTextToNumber(cellValue)
    70→  }
    71→
    72→  if (result > 0 && isRedFont(cell)) {
    73→    result = -result
    74→  }
    75→
    76→  return result
    77→}
    78→
    79→/**
    80→ * Parse a text string like "1.234,56" or "(780,00)" into a number.
    81→ * Parenthesized values become negative.
    82→ */
    83→function parseTextToNumber(text: string): number {
    84→  const trimmed = text.trim()
    85→  if (!trimmed) return 0
    86→  const parenMatch = trimmed.match(/^\((.+)\)$/)
    87→  let numStr: string
    88→  let sign = 1
    89→  if (parenMatch) {
    90→    numStr = parenMatch[1]
    91→    sign = -1
    92→  } else {
    93→    numStr = trimmed
    94→    if (numStr.startsWith('-')) {
    95→      sign = -1
    96→      numStr = numStr.substring(1)
    97→    }
    98→  }
    99→  const cleaned = numStr.replace(/\./g, '').replace(',', '.')
   100→  const parsed = parseFloat(cleaned)
   101→  if (isNaN(parsed)) return 0
   102→  return sign * Math.abs(parsed)
   103→}

   107→function excelSerialToDate(serial: number): Date {
   108→  if (serial < 1) return new Date(NaN)
   109→  const epoch = new Date(1899, 11, 30)
   110→  const msPerDay = 24 * 60 * 60 * 1000
   111→  return new Date(epoch.getTime() + serial * msPerDay)
   112→}

   114→/** Remove accents/diacritics from a string for comparison */
   115→function normalizeAccents(str: string): string {
   116→  return str.normalize('NFD').replace(/[\u0300-\u036f]/g, '').toLowerCase().trim()
   117→}

   119→interface ParsedDateInfo {
   120→  month: number
   121→  year: number
   122→  dateStr: string
   123→}
   124→
   125→/**
   126→ * Parse a date value from an Excel cell (column B) into month/year.
   127→ * Supports: JS Date, DD/MM/YYYY, YYYY-MM-DD, MM/DD/YYYY, DD-MM-YYYY, Excel serial numbers.
   128→ */
   129→function parseRowDate(rawValue: unknown): ParsedDateInfo | null {
   130→  if (!rawValue) return null
   131→
   132→  if (rawValue instanceof Date && !isNaN(rawValue.getTime())) {
   133→    return {
   134→      month: rawValue.getMonth() + 1,
   135→      year: rawValue.getFullYear(),
   136→      dateStr: rawValue.toISOString().split('T')[0],
   137→    }
   138→  }
   139→
   140→  if (typeof rawValue === 'number' && rawValue > 25569 && rawValue < 100000) {
   141→    const d = excelSerialToDate(rawValue)
   142→    if (!isNaN(d.getTime())) {
   143→      return {
   144→        month: d.getMonth() + 1,
   145→        year: d.getFullYear(),
   146→        dateStr: d.toISOString().split('T')[0],
   147→      }
   148→    }
   149→  }
   150→
   151→  if (typeof rawValue === 'string') {
   152→    const s = rawValue.trim()
   153→    if (!s) return null
   154→
   155→    // YYYY-MM-DD
   156→    let m = s.match(/^(\d{4})-(\d{1,2})-(\d{1,2})/)
   157→    if (m) {
   158→      const yr = parseInt(m[1]); const mo = parseInt(m[2]); const dy = parseInt(m[3])
   159→      if (mo >= 1 && mo <= 12 && dy >= 1 && dy <= 31) {
   160→        return { month: mo, year: yr, dateStr: `${yr}-${String(mo).padStart(2, '0')}-${String(dy).padStart(2, '0')}` }
   161→      }
   162→    }
   163→
   164→    // DD/MM/YYYY or MM/DD/YYYY
   165→    m = s.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/)
   166→    if (m) {
   167→      const a = parseInt(m[1]); const b = parseInt(m[2]); const yr = parseInt(m[3])
   168→      let mo: number, dy: number
   169→      if (a > 12) { dy = a; mo = b }
   170→      else if (b > 12) { mo = a; dy = b }
   171→      else { dy = a; mo = b }
   172→      if (mo >= 1 && mo <= 12 && dy >= 1 && dy <= 31) {
   173→        return { month: mo, year: yr, dateStr: `${yr}-${String(mo).padStart(2, '0')}-${String(dy).padStart(2, '0')}` }
   174→      }
   175→    }
   176→
   177→    // DD-MM-YYYY
   178→    m = s.match(/^(\d{1,2})-(\d{1,2})-(\d{4})$/)
   179→    if (m) {
   180→      const a = parseInt(m[1]); const b = parseInt(m[2]); const yr = parseInt(m[3])
   181→      let mo: number, dy: number
   182→      if (a > 12) { dy = a; mo = b }
   183→      else if (b > 12) { mo = a; dy = b }
   184→      else { dy = a; mo = b }
   185→      if (mo >= 1 && mo <= 12 && dy >= 1 && dy <= 31) {
   186→        return { month: mo, year: yr, dateStr: `${yr}-${String(mo).padStart(2, '0')}-${String(dy).padStart(2, '0')}` }
   187→      }
   188→    }
   189→
   190→    // DD/MM/YY
   191→    m = s.match(/^(\d{1,2})\/(\d{1,2})\/(\d{2})$/)
   192→    if (m) {
   193→      const a = parseInt(m[1]); const b = parseInt(m[2])
   194→      let yr = parseInt(m[3])
   195→      yr = yr < 100 ? 2000 + yr : yr
   196→      let mo: number, dy: number
   197→      if (a > 12) { dy = a; mo = b }
   198→      else if (b > 12) { mo = a; dy = b }
   199→      else { dy = a; mo = b }
   200→      if (mo >= 1 && mo <= 12 && dy >= 1 && dy <= 31) {
   201→        return { month: mo, year: yr, dateStr: `${yr}-${String(mo).padStart(2, '0')}-${String(dy).padStart(2, '0')}` }
   202→      }
   203→    }
   204→  }
   205→
   206→  return null
   207→}

   248→/**
   249→ * RF-04: Detect if a description is a structural title and return its context.
   250→ * Returns { type: 'section' | 'field', fieldKey?, sectionKey? } or null.
   251→ */
   252→function detectStructuralTitle(description: string): { type: 'section' | 'field'; fieldKey?: string; sectionKey?: string } | null {
   253→  const norm = normalizeAccents(description)
   254→
   255→  // Check for section-level matches first
   256→  for (const entry of STRUCTURAL_TITLES) {
   257→    for (const pattern of entry.patterns) {
   258→      if (norm === normalizeAccents(pattern) || norm.includes(normalizeAccents(pattern))) {
   259→        if (entry.sectionKey && !entry.fieldKey) {
   260→          return { type: 'section', sectionKey: entry.sectionKey }
   261→        }
   262→        if (entry.fieldKey) {
   263→          return { type: 'field', fieldKey: entry.fieldKey, sectionKey: DRE_FIELD_SECTIONS[entry.fieldKey] }
   264→        }
   265→      }
   266→    }
   267→  }
   268→  return null
   269→}
   270→
   271→/**
   272→ * RF-04: Given a section context, try to auto-classify a description into the first matching field in that section.
   273→ */
   274→function autoClassifyInSection(description: string, sectionKey: string): string | null {
   275→  const norm = normalizeAccents(description)
   276→  const sectionFields = Object.entries(DRE_FIELD_SECTIONS)
   277→    .filter(([, section]) => section === sectionKey)
   278→    .map(([key]) => key)
   279→
   280→  for (const fieldKey of sectionFields) {
   281→    const label = DRE_FIELD_LABELS[fieldKey]
   282→    if (!label) continue
   283→    const normLabel = normalizeAccents(label)
   284→    // Check if description contains field label keywords
   285→    const labelWords = normLabel.split(/\s+/).filter(w => w.length > 2)
   286→    for (const word of labelWords) {
   287→      if (norm.includes(word)) return fieldKey
   288→    }
   289→  }
   290→  return null
   291→}
   292→
   293→/**
   294→ * Try to match a category string from column D against DRE field labels.
   295→ */
   296→function classifyByCategory(category: string): string | null {
   297→  if (!category || !category.trim()) return null
   298→  const norm = normalizeAccents(category)
   299→
   300→  for (const key of Object.keys(DRE_FIELD_LABELS)) {
   301→    if (normalizeAccents(key) === norm) return key
   302→  }
   303→  for (const [key, value] of Object.entries(DRE_FIELD_LABELS)) {
   304→    if (normalizeAccents(value) === norm) return key
   305→  }
   306→  return null
   307→}

   309→// ─── POST: Upload spreadsheet ───
   310→
   311→export async function POST(req: NextRequest) {
   312→  try {
   313→    const session = await getServerSession(authOptions)
   314→    if (!session?.user?.id) {
   315→      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
   316→    }
   317→
   318→    const userId = (session.user as any).id
   319→    const formData = await req.formData()
   320→    const file = formData.get('file') as File
   321→    const companyId = formData.get('companyId') as string
   322→    const month = parseInt(formData.get('month') as string)
   323→    const year = parseInt(formData.get('year') as string)
   324→
   325→    if (!file || !companyId || !month || !year) {
   326→      return NextResponse.json({ error: 'Parâmetros incompletos' }, { status: 400 })
   327→    }
   328→
   329→    const hasAccess = await verifyAccess(userId, companyId)
   330→    if (!hasAccess) {
   331→      return NextResponse.json({ error: 'Sem permissão' }, { status: 403 })
   332→    }
   333→
   334→    const buffer = Buffer.from(await file.arrayBuffer())
   335→    const workbook = new ExcelJS.Workbook()
   336→    await workbook.xlsx.load(buffer)
   337→
   338→    const worksheet = workbook.worksheets[0]
   339→    if (!worksheet) {
   340→      return NextResponse.json({ error: 'Planilha vazia' }, { status: 400 })
   341→    }
   342→
   343→    // RF-04: Track structural title context during parsing
   344→    let currentSectionContext: string | null = null
   345→    let currentFieldContext: string | null = null
   346→
   347→    const parsedRows: Array<{
   348→      description: string
   349→      date: string
   350→      totalValue: number
   351→      itemMonth: number
   352→      itemYear: number
   353→      dreField: string
   354→    }> = []
   355→
   356→    let structuralTitlesFound = 0
   357→
   358→    worksheet.eachRow((row, rowNumber) => {
   359→      if (rowNumber === 1) return
   360→
   361→      // Column A: Description
   362→      const description = row.getCell(1).value?.toString()?.trim() || ''
   363→      if (!description) return
   364→
   365→      // Column B: Date → parse to get month/year
   366→      const rawDate = row.getCell(2).value
   367→      const parsedDate = parseRowDate(rawDate)
   368→      const itemMonth = parsedDate?.month ?? month
   369→      const itemYear = parsedDate?.year ?? year
   370→      const dateStr = parsedDate?.dateStr ?? ''
   371→
   372→      // Column C: Value
   373→      const valueCell = row.getCell(3)
   374→      const value = parseCellToNumber(valueCell.value, { font: valueCell.font } as ExcelCellLike)
   375→
   376→      // RF-04: Check if this row is a structural title
   377→      const titleInfo = detectStructuralTitle(description)
   378→
   379→      if (titleInfo) {
   380→        structuralTitlesFound++
   381→        if (titleInfo.type === 'section') {
   382→          currentSectionContext = titleInfo.sectionKey ?? null
   383→          currentFieldContext = null
   384→        } else if (titleInfo.type === 'field') {
   385→          currentFieldContext = titleInfo.fieldKey
   386→          currentSectionContext = titleInfo.sectionKey ?? null
   387→        }
   388→        // Structural titles with no value are skipped as data rows
   389→        if (value === 0) return
   390→      }
   391→
   392→      // Column D: Category (explicit auto-classification, takes priority)
   393→      const categoryCell = row.getCell(4)
   394→      let dreField = ''
   395→      if (categoryCell && categoryCell.value) {
   396→        const categoryText = String(categoryCell.value).trim()
   397→        dreField = classifyByCategory(categoryText) || ''
   398→      }
   399→
   400→      // RF-04: If no category, use structural context for auto-classification
   401→      if (!dreField) {
   402→        if (currentFieldContext) {
   403→          dreField = currentFieldContext
   404→        } else if (currentSectionContext) {
   405→          const autoField = autoClassifyInSection(description, currentSectionContext)
   406→          if (autoField) dreField = autoField
   407→        }
   408→      }
   409→
   410→      if (value !== 0) {
   411→        parsedRows.push({ description, date: dateStr, totalValue: value, itemMonth, itemYear, dreField })
   412→      }
   413→    })
   414→
   415→    if (parsedRows.length === 0) {
   416→      return NextResponse.json(
   417→        { error: 'Nenhum dado válido encontrado. Verifique: Descrição (A), Data (B), Valor (C).' },
   418→        { status: 400 }
   419→      )
   420→    }
   421→
   422→    // Create uploaded file record
   423→    const uploadedFile = await db.uploadedFile.create({
   424→      data: { companyId, month, year, fileName: file.name },
   425→    })
   426→
   427→    // Save all qualifications
   428→    const qualifications = parsedRows.map((item) => ({
   429→      companyId,
   430→      month: item.itemMonth,
   431→      year: item.itemYear,
   432→      description: item.description,
   433→      date: item.date,
   434→      totalValue: item.totalValue,
   435→      dreField: item.dreField,
   436→      uploadedFileId: uploadedFile.id,
   437→    }))
   438→
   439→    await db.expenseQualification.createMany({ data: qualifications })
   440→
   441→    const savedItems = await db.expenseQualification.findMany({
   442→      where: { uploadedFileId: uploadedFile.id },
   443→      orderBy: { createdAt: 'asc' },
   444→    })
   445→
   446→    // Build distributed months list
   447→    const groupedByMonth = new Map<string, number[]>()
   448→    for (const row of parsedRows) {
   449→      const key = `${row.itemMonth}-${row.itemYear}`
   450→      if (!groupedByMonth.has(key)) groupedByMonth.set(key, [])
   451→      groupedByMonth.get(key)!.push(row.itemMonth)
   452→    }
   453→    const distributedMonths = Array.from(groupedByMonth.keys())
   454→      .map(key => {
   455→        const [m, y] = key.split('-')
   456→        return `${String(parseInt(m)).padStart(2, '0')}/${y}`
   457→      })
   458→      .sort((a, b) => {
   459→        const [am, ay] = a.split('/').map(Number)
   460→        const [bm, by] = b.split('/').map(Number)
   461→        return ay !== by ? ay - by : am - bm
   462→      })
   463→
   464→    // Count auto-classified items
   465→    const autoClassified = savedItems.filter(i => i.dreField).length
   466→
   467→    return NextResponse.json({
   468→      items: savedItems.map(i => ({
   469→        id: i.id, description: i.description, date: i.date,
   470→        totalValue: i.totalValue, dreField: i.dreField,
   471→        month: i.month, year: i.year,
   472→      })),
   473→      file: { id: uploadedFile.id, fileName: uploadedFile.fileName, uploadedAt: uploadedFile.uploadedAt },
   474→      distributedMonths,
   475→      structuralTitlesFound,
   476→      autoClassified,
   477→    })
   478→  } catch (error) {
   479→    console.error('Expense qualification error:', error)
   480→    return NextResponse.json({ error: 'Erro ao processar planilha de despesas' }, { status: 500 })
   481→  }
   482→}

   484→// ─── PUT: Save expense classification and update DRE (RF-07: returns updated DRE data) ───
   485→
   486→export async function PUT(req: NextRequest) {
   487→  try {
   488→    const session = await getServerSession(authOptions)
   489→    if (!session?.user?.id) {
   490→      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
   491→    }
   492→
   493→    const userId = (session.user as any).id
   494→    const body = await req.json()
   495→    const { companyId, month, year, items } = body
   496→
   497→    if (!companyId || !month || !year || !items) {
   498→      return NextResponse.json({ error: 'Parâmetros incompletos' }, { status: 400 })
   499→    }
   500→
   501→    const hasAccess = await verifyAccess(userId, companyId)
   502→    if (!hasAccess) {
   503→      return NextResponse.json({ error: 'Sem permissão' }, { status: 403 })
   504→    }
   505→
   506→    // Update each item's dreField mapping in parallel
   507→    const itemsToUpdate = items.filter((item: any) => item.id && item.dreField)
   508→    await Promise.all(
   509→      itemsToUpdate.map((item: any) =>
   510→        db.expenseQualification.update({
   511→          where: { id: item.id },
   512→          data: { dreField: item.dreField },
   513→        })
   514→      )
   515→    )
   516→
   517→    // Calculate totals grouped by dreField for THIS month/year only
   518→    const qualifications = await db.expenseQualification.findMany({
   519→      where: { companyId, month, year, dreField: { not: '' } },
   520→    })
   521→
   522→    const fieldTotals: Record<string, number> = {}
   523→    const customFieldTotals: Record<string, number> = {}
   524→    for (const q of qualifications) {
   525→      if (q.dreField) {
   526→        if (q.dreField.startsWith('custom_')) {
   527→          const customFieldId = q.dreField.replace('custom_', '')
   528→          customFieldTotals[customFieldId] = (customFieldTotals[customFieldId] || 0) + q.totalValue
   529→        } else {
   530→          fieldTotals[q.dreField] = (fieldTotals[q.dreField] || 0) + q.totalValue
   531→        }
   532→      }
   533→    }
   534→
   535→    // Update DRE entry with classified totals
   536→    const existing = await db.dREEntry.findUnique({
   537→      where: { companyId_month_year: { companyId, month, year } },
   538→    })
   539→
   540→    const updateData: Record<string, number> = {}
   541→    for (const [field, total] of Object.entries(fieldTotals)) {
   542→      updateData[field] = total
   543→    }
   544→
   545→    let dreEntryId: string | null = null
   546→    if (Object.keys(updateData).length > 0 || Object.keys(customFieldTotals).length > 0) {
   547→      if (existing) {
   548→        dreEntryId = existing.id
   549→        if (Object.keys(updateData).length > 0) {
   550→          await db.dREEntry.update({
   551→            where: { id: existing.id },
   552→            data: updateData,
   553→          })
   554→        }
   555→      } else {
   556→        const created = await db.dREEntry.create({
   557→          data: { companyId, month, year, ...updateData },
   558→        })
   559→        dreEntryId = created.id
   560→      }
   561→    } else {
   562→      dreEntryId = existing?.id || null
   563→    }
   564→
   565→    // Handle custom field totals (with sign correction)
   566→    if (dreEntryId && Object.keys(customFieldTotals).length > 0) {
   567→      const customFieldIds = Object.keys(customFieldTotals)
   568→      const customFields = await db.customDREField.findMany({
   569→        where: { id: { in: customFieldIds } },
   570→      })
   571→      const fieldSignMap: Record<string, string> = {}
   572→      for (const cf of customFields) fieldSignMap[cf.id] = cf.sign
   573→
   574→      await Promise.all(
   575→        Object.entries(customFieldTotals).map(([customFieldId, total]) => {
   576→          const sign = fieldSignMap[customFieldId]
   577→          const adjustedTotal = sign === '-' ? -total : total
   578→          return db.customDREValue.upsert({
   579→            where: { customFieldId_dREEntryId: { customFieldId, dREEntryId } },
   580→            create: { customFieldId, dREEntryId, value: adjustedTotal },
   581→            update: { value: adjustedTotal },
   582→          })
   583→        })
   584→      )
   585→    }
   586→
   587→    // RF-07: Fetch the updated DRE entry to return for reactive sync
   588→    let updatedDREData: Record<string, any> | null = null
   589→    if (dreEntryId) {
   590→      const updatedEntry = await db.dREEntry.findUnique({ where: { id: dreEntryId } })
   591→      if (updatedEntry) {
   592→        const { id, companyId: _cid, month: _m, year: _y, createdAt, updatedAt, company, ...fields } = updatedEntry as any
   593→        updatedDREData = fields
   594→
   595→        // Also fetch custom field values
   596→        const customValues = await db.customDREValue.findMany({
   597→          where: { dREEntryId },
   598→          include: { customField: true },
   599→        })
   600→        if (customValues.length > 0) {
   601→          updatedDREData._customFieldValues = {}
   602→          for (const cv of customValues) {
   603→            const sign = (cv.customField as any).sign || '+'
   604→            updatedDREData._customFieldValues[cv.customFieldId] = sign === '-' ? -cv.value : cv.value
   605→          }
   606→        }
   607→      }
   608→    }
   609→
   610→    return NextResponse.json({
   611→      success: true,
   612→      fieldTotals,
   613→      dreEntryId,
   614→      updatedDREData, // RF-07: reactive sync payload
   615→    })
   616→  } catch (error) {
   617→    console.error('Error updating qualification:', error)
   618→    return NextResponse.json({ error: 'Erro ao atualizar qualificação' }, { status: 500 })
   619→  }
   620→}

   622→// ─── DELETE: Remove uploaded files ───
   623→
   624→export async function DELETE(req: NextRequest) {
   625→  try {
   626→    const session = await getServerSession(authOptions)
   627→    if (!session?.user?.id) {
   628→      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
   629→    }
   630→
   631→    const userId = (session.user as any).id
   632→    const { searchParams } = new URL(req.url)
   633→    const fileIdsParam = searchParams.get('fileIds')
   634→
   635→    if (!fileIdsParam) {
   636→      return NextResponse.json({ error: 'fileIds obrigatório' }, { status: 400 })
   637→    }
   638→
   639→    const ids = fileIdsParam.split(',').filter(Boolean)
   640→    if (ids.length === 0) {
   641→      return NextResponse.json({ error: 'Nenhum ID válido' }, { status: 400 })
   642→    }
   643→
   644→    const files = await db.uploadedFile.findMany({ where: { id: { in: ids } } })
   645→    const uniqueCompanyIds = [...new Set(files.map(f => f.companyId))]
   646→    const accessChecks = await Promise.all(
   647→      uniqueCompanyIds.map(cid => verifyAccess(userId, cid))
   648→    )
   649→    const accessMap = new Map(uniqueCompanyIds.map((cid, i) => [cid, accessChecks[i]]))
   650→
   651→    for (const file of files) {
   652→      if (!accessMap.get(file.companyId)) {
   653→        return NextResponse.json({ error: 'Sem permissão para excluir arquivo' }, { status: 403 })
   654→      }
   655→    }
   656→
   657→    const deleteResult = await db.expenseQualification.deleteMany({
   658→      where: { uploadedFileId: { in: ids } },
   659→    })
   660→
   661→    const fileDeleteResult = await db.uploadedFile.deleteMany({
   662→      where: { id: { in: ids } },
   663→    })
   664→
   665→    return NextResponse.json({ success: true, deletedFiles: fileDeleteResult.count, deletedItems: deleteResult.count })
   666→  } catch (error) {
   667→    console.error('Error deleting uploaded files:', error)
   668→    return NextResponse.json({ error: 'Erro ao excluir arquivos' }, { status: 500 })
   669→  }
   670→}
