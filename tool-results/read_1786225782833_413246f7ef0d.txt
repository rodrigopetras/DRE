    22→    22→interface ExcelCellLike {
    23→    23→  value: unknown
    24→    24→  font?: { color?: { argb?: string } }
    25→    25→}
    26→    26→
    27→    27→/**
    28→    28→ * Detect if an Excel cell's font color is red.
    29→    29→ * Red = alpha byte 0xFF (high byte), R > 150, G < 100, B < 100.
    30→    30→ */
    31→    31→function isRedFont(cell?: ExcelCellLike): boolean {
    32→    32→  if (!cell?.font?.color?.argb) return false
    33→    33→  const argb = cell.font.color.argb
    34→    34→  if (argb.length < 8) return false
    35→    35→  const alpha = parseInt(argb.substring(0, 2), 16)
    36→    36→  if (alpha !== 0xFF) return false
    37→    37→  const r = parseInt(argb.substring(2, 4), 16)
    38→    38→  const g = parseInt(argb.substring(4, 6), 16)
    39→    39→  const b = parseInt(argb.substring(6, 8), 16)
    40→    40→  return r > 150 && g < 100 && b < 100
    41→    41→}
    42→    42→
    43→    43→/**
    44→    44→ * Parse any ExcelJS cell value into a number.
    45→    45→ * Handles: number, formula, RichText, Hyperlink, string, Date.
    46→    46→ * Parenthesized values like (780,00) are converted to negative.
    47→    47→ * Red font color also makes the value negative.
    48→    48→ */
    49→    49→function parseCellToNumber(cellValue: unknown, cell?: ExcelCellLike): number {
    50→    50→  let result = 0
    51→    51→
    52→    52→  if (typeof cellValue === 'number') {
    53→    53→    result = cellValue
    54→    54→  }
    55→    55→  else if (cellValue && typeof cellValue === 'object' && 'result' in cellValue) {
    56→    56→    const formulaResult = (cellValue as any).result
    57→    57→    if (typeof formulaResult === 'number') result = formulaResult
    58→    58→    else if (typeof formulaResult === 'string') result = parseTextToNumber(formulaResult)
    59→    59→  }
    60→    60→  else if (cellValue && typeof cellValue === 'object' && 'richText' in cellValue) {
    61→    61→    const parts = (cellValue as any).richText as Array<{ text: string }>
    62→    62→    const text = parts.map(p => p.text).join('')
    63→    63→    result = parseTextToNumber(text)
    64→    64→  }
    65→    65→  else if (cellValue && typeof cellValue === 'object' && 'text' in cellValue && 'hyperlink' in cellValue) {
    66→    66→    result = parseTextToNumber((cellValue as any).text)
    67→    67→  }
    68→    68→  else if (typeof cellValue === 'string') {
    69→    69→    result = parseTextToNumber(cellValue)
    70→    70→  }
    71→    71→
    72→    72→  if (result > 0 && isRedFont(cell)) {
    73→    73→    result = -result
    74→    74→  }
    75→    75→
    76→    76→  return result
    77→    77→}

    79→    79→/**
    80→    80→ * Parse a text string like "1.234,56" or "(780,00)" into a number.
    81→    81→ * Parenthesized values become negative.
    82→    82→ */
    83→    83→function parseTextToNumber(text: string): number {
    84→    84→  const trimmed = text.trim()
    85→    85→  if (!trimmed) return 0
    86→    86→  const parenMatch = trimmed.match(/^\((.+)\)$/)
    87→    87→  let numStr: string
    88→    88→  let sign = 1
    89→    89→  if (parenMatch) {
    90→    90→    numStr = parenMatch[1]
    91→    91→    sign = -1
    92→    92→  } else {
    93→    93→    numStr = trimmed
    94→    94→    if (numStr.startsWith('-')) {
    95→    95→      sign = -1
    96→    96→      numStr = numStr.substring(1)
    97→    97→    }
    98→    98→  }
    99→    99→  const cleaned = numStr.replace(/\./g, '').replace(',', '.')
   100→   100→  const parsed = parseFloat(cleaned)
   101→   101→  if (isNaN(parsed)) return 0
   102→   102→  return sign * Math.abs(parsed)
   103→   103→}

   107→   107→function excelSerialToDate(serial: number): Date {
   108→   108→  if (serial < 1) return new Date(NaN)
   109→   109→  const epoch = new Date(1899, 11, 30)
   110→   110→  const msPerDay = 24 * 60 * 60 * 1000
   111→   111→  return new Date(epoch.getTime() + serial * msPerDay)
   112→   112→}

   115→   115→function normalizeAccents(str: string): string {

   127→   127→ * Supports: JS Date, DD/MM/YYYY, YYYY-MM-DD, MM/DD/YYYY, DD-MM-YYYY, Excel serial numbers.

   129→   129→const MONTH_NAMES_PT: Record<string, number> = {

   136→   136→function parseRowDate(rawValue: unknown): ParsedDateInfo | null {
   137→   137→  if (!rawValue) return null
   138→   138→
   139→   139→  if (rawValue instanceof Date && !isNaN(rawValue.getTime())) {
   140→   140→    return {
   141→   141→      month: rawValue.getMonth() + 1,
   142→   142→      year: rawValue.getFullYear(),
   143→   143→      dateStr: rawValue.toISOString().split('T')[0],
   144→   144→    }
   145→   145→  }
   146→   146→
   147→   147→  if (typeof rawValue === 'number' && rawValue > 25569 && rawValue < 100000) {
   148→   148→    const d = excelSerialToDate(rawValue)
   149→   149→    if (!isNaN(d.getTime())) {
   150→   150→      return {
   151→   151→        month: d.getMonth() + 1,
   152→   152→        year: d.getFullYear(),
   153→   153→        dateStr: d.toISOString().split('T')[0],
   154→   154→      }
   155→   155→    }
   156→   156→  }
   157→   157→
   158→   158→  if (typeof rawValue === 'string') {
   159→   159→    const s = rawValue.trim()
   160→   160→    if (!s) return null
   161→   161→
   162→   162→    // YYYY-MM-DD
   163→   163→    let m = s.match(/^(\d{4})-(\d{1,2})-(\d{1,2})/)
   164→   164→    if (m) {
   165→   165→      const yr = parseInt(m[1]); const mo = parseInt(m[2]); const dy = parseInt(m[3])
   166→   166→      if (mo >= 1 && mo <= 12 && dy >= 1 && dy <= 31) {
   167→   167→        return { month: mo, year: yr, dateStr: `${yr}-${String(mo).padStart(2, '0')}-${String(dy).padStart(2, '0')}` }
   168→   168→      }
   169→   169→    }
   170→   170→
   171→   171→    // DD/MM/YYYY or MM/DD/YYYY
   172→   172→    m = s.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/)
   173→   173→    if (m) {
   174→   174→      const a = parseInt(m[1]); const b = parseInt(m[2]); const yr = parseInt(m[3])
   175→   175→      let mo: number, dy: number
   176→   176→      if (a > 12) { dy = a; mo = b }
   177→   177→      else if (b > 12) { mo = a; dy = b }
   178→   178→      else { dy = a; mo = b }
   179→   179→      if (mo >= 1 && mo <= 12 && dy >= 1 && dy <= 31) {
   180→   180→        return { month: mo, year: yr, dateStr: `${yr}-${String(mo).padStart(2, '0')}-${String(dy).padStart(2, '0')}` }
   181→   181→      }
   182→   182→    }
   183→   183→
   184→   184→    // DD-MM-YYYY
   185→   185→    m = s.match(/^(\d{1,2})-(\d{1,2})-(\d{4})$/)
   186→   186→    if (m) {
   187→   187→      const a = parseInt(m[1]); const b = parseInt(m[2]); const yr = parseInt(m[3])
   188→   188→      let mo: number, dy: number
   189→   189→      if (a > 12) { dy = a; mo = b }
   190→   190→      else if (b > 12) { mo = a; dy = b }
   191→   191→      else { dy = a; mo = b }
   192→   192→      if (mo >= 1 && mo <= 12 && dy >= 1 && dy <= 31) {
   193→   193→        return { month: mo, year: yr, dateStr: `${yr}-${String(mo).padStart(2, '0')}-${String(dy).padStart(2, '0')}` }
   194→   194→      }
   195→   195→    }
   196→   196→
   197→   197→    // DD/MM/YY
   198→   198→    m = s.match(/^(\d{1,2})\/(\d{1,2})\/(\d{2})$/)
   199→   199→    if (m) {
   200→   200→      const a = parseInt(m[1]); const b = parseInt(m[2])
   201→   201→      let yr = parseInt(m[3])
   202→   202→      yr = yr < 100 ? 2000 + yr : yr
   203→   203→      let mo: number, dy: number
   204→   204→      if (a > 12) { dy = a; mo = b }
   205→   205→      else if (b > 12) { mo = a; dy = b }
   206→   206→      else { dy = a; mo = b }
   207→   207→      if (mo >= 1 && mo <= 12 && dy >= 1 && dy <= 31) {
   208→   208→        return { month: mo, year: yr, dateStr: `${yr}-${String(mo).padStart(2, '0')}-${String(dy).padStart(2, '0')}` }
   209→   209→      }
   210→   210→    }
   211→   211→
   212→   212→    // Month name in Portuguese (e.g. "Janeiro 2025", "Jan/2025", "janeiro")
   213→   213→    const normS = normalizeAccents(s)
   214→   214→    for (const [monthName, monthNum] of Object.entries(MONTH_NAMES_PT)) {
   215→   215→      const normName = normalizeAccents(monthName)
   216→   216→      const regex = new RegExp(`\\b${normName}\\b`, 'i')
   217→   217→      if (regex.test(normS)) {
   218→   218→        // Try to extract year from the string
   219→   219→        const yearMatch = s.match(/(\d{4})/)
   220→   220→        const yr = yearMatch ? parseInt(yearMatch[1]) : new Date().getFullYear()
   221→   221→        return { month: monthNum, year: yr, dateStr: `${yr}-${String(monthNum).padStart(2, '0')}-01` }
   222→   222→      }
   223→   223→    }
   224→   224→
   225→   225→    // MM/YYYY (just month and year)
   226→   226→    m = s.match(/^(\d{1,2})\/(\d{4})$/)
   227→   227→    if (m) {
   228→   228→      const mo = parseInt(m[1]); const yr = parseInt(m[2])
   229→   229→      if (mo >= 1 && mo <= 12) {
   230→   230→        return { month: mo, year: yr, dateStr: `${yr}-${String(mo).padStart(2, '0')}-01` }
   231→   231→      }
   232→   232→    }
   233→   233→  }
   234→   234→
   235→   235→  return null
   236→   236→}

   289→   289→/**
   290→   290→ * RF-04: Detect if a description is a structural title and return its context.
   291→   291→ * Returns { type: 'section' | 'field', fieldKey?, sectionKey? } or null.
   292→   292→ */
   293→   293→function detectStructuralTitle(description: string): { type: 'section' | 'field'; fieldKey?: string; sectionKey?: string } | null {
   294→   294→  const norm = normalizeAccents(description)
   295→   295→
   296→   296→  // Check for section-level matches first
   297→   297→  for (const entry of STRUCTURAL_TITLES) {
   298→   298→    for (const pattern of entry.patterns) {
   299→   299→      if (norm === normalizeAccents(pattern) || norm.includes(normalizeAccents(pattern))) {
   300→   300→        if (entry.sectionKey && !entry.fieldKey) {
   301→   301→          return { type: 'section', sectionKey: entry.sectionKey }
   302→   302→        }
   303→   303→        if (entry.fieldKey) {
   304→   304→          return { type: 'field', fieldKey: entry.fieldKey, sectionKey: DRE_FIELD_SECTIONS[entry.fieldKey] }
   305→   305→        }
   306→   306→      }
   307→   307→    }
   308→   308→  }
   309→   309→  return null
   310→   310→}

   312→   312→/**
   313→   313→ * When a structural section title row has a value, map it to a reasonable DRE field.
   314→   314→ * For receita section: maps to 'prestacaoNf' (main revenue field for law firms).
   315→   315→ * For desp sections: maps to the first field in that section.
   316→   316→ */
   317→   317→function sectionTitleToField(sectionKey: string): string {
   318→   318→  if (sectionKey === 'receita') return 'prestacaoNf'
   319→   319→  // For expense sections, find the first field in that section
   320→   320→  const sectionFields = Object.entries(DRE_FIELD_SECTIONS)
   321→   321→    .filter(([, section]) => section === sectionKey)
   322→   322→    .map(([key]) => key)
   323→   323→  return sectionFields[0] || 'naoIdentificado'
   324→   324→}

   326→   326→/**
   327→   327→ * RF-04: Given a section context, try to auto-classify a description into the first matching field in that section.
   328→   328→ */
   329→   329→function autoClassifyInSection(description: string, sectionKey: string): string | null {
   330→   330→  const norm = normalizeAccents(description)
   331→   331→  const sectionFields = Object.entries(DRE_FIELD_SECTIONS)
   332→   332→    .filter(([, section]) => section === sectionKey)
   333→   333→    .map(([key]) => key)
   334→   334→
   335→   335→  for (const fieldKey of sectionFields) {
   336→   336→    const label = DRE_FIELD_LABELS[fieldKey]
   337→   337→    if (!label) continue
   338→   338→    const normLabel = normalizeAccents(label)
   339→   339→    // Check if description contains field label keywords
   340→   340→    const labelWords = normLabel.split(/\s+/).filter(w => w.length > 2)
   341→   341→    for (const word of labelWords) {
   342→   342→      if (norm.includes(word)) return fieldKey
   343→   343→    }
   344→   344→  }
   345→   345→  return null
   346→   346→}

   348→   348→/**
   349→   349→ * Try to match a category string from column D against DRE field labels.
   350→   350→ */
   351→   351→function classifyByCategory(category: string): string | null {
   352→   352→  if (!category || !category.trim()) return null
   353→   353→  const norm = normalizeAccents(category)
   354→   354→
   355→   355→  for (const key of Object.keys(DRE_FIELD_LABELS)) {
   356→   356→    if (normalizeAccents(key) === norm) return key
   357→   357→  }
   358→   358→  for (const [key, value] of Object.entries(DRE_FIELD_LABELS)) {
   359→   359→    if (normalizeAccents(value) === norm) return key
   360→   360→  }
   361→   361→  return null
   362→   362→}

   366→   366→export async function POST(req: NextRequest) {
   367→   367→  try {
   368→   368→    const session = await getServerSession(authOptions)
   369→   369→    if (!session?.user?.id) {
   370→   370→      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
   371→   371→    }
   372→   372→
   373→   373→    const userId = (session.user as any).id
   374→   374→    const formData = await req.formData()
   375→   375→    const file = formData.get('file') as File
   376→   376→    const companyId = formData.get('companyId') as string
   377→   377→    const month = parseInt(formData.get('month') as string)
   378→   378→    const year = parseInt(formData.get('year') as string)
   379→   379→
   380→   380→    if (!file || !companyId || !month || !year) {
   381→   381→      return NextResponse.json({ error: 'Parâmetros incompletos' }, { status: 400 })
   382→   382→    }
   383→   383→
   384→   384→    const hasAccess = await verifyAccess(userId, companyId)
   385→   385→    if (!hasAccess) {
   386→   386→      return NextResponse.json({ error: 'Sem permissão' }, { status: 403 })
   387→   387→    }
   388→   388→
   389→   389→    const buffer = Buffer.from(await file.arrayBuffer())
   390→   390→    const workbook = new ExcelJS.Workbook()
   391→   391→    await workbook.xlsx.load(buffer)
   392→   392→
   393→   393→    if (workbook.worksheets.length === 0) {
   394→   394→      return NextResponse.json({ error: 'Planilha vazia' }, { status: 400 })
   395→   395→    }
   396→   396→
   397→   397→    // ─── Step 1: Look for "Caixa YYYY" worksheet to extract Receita Bruta from C3 ───
   398→   398→    let receitaBrutaValue = 0
   399→   399→    let receitaBrutaYear = year
   400→   400→    const caixaPattern = /^caixa\s+(\d{4})$/i
   401→   401→    for (const ws of workbook.worksheets) {
   402→   402→      const match = ws.name?.trim().match(caixaPattern)
   403→   403→      if (match) {
   404→   404→        receitaBrutaYear = parseInt(match[1])
   405→   405→        // Read specifically cell C3 (row 3, column 3)
   406→   406→        if (ws.rowCount >= 3 && ws.getColumn(3).values) {
   407→   407→          const cell = ws.getCell(3, 3)
   408→   408→          const val = parseCellToNumber(cell.value, { font: cell.font } as ExcelCellLike)
   409→   409→          if (val !== 0) {
   410→   410→            receitaBrutaValue = val
   411→   411→          }
   412→   412→        }
   413→   413→        break
   414→   414→      }
   415→   415→    }
   416→   416→
   417→   417→    const worksheet = workbook.worksheets[0]
   418→   418→    if (!worksheet) {
   419→   419→      return NextResponse.json({ error: 'Planilha vazia' }, { status: 400 })
   420→   420→    }
   421→   421→
   422→   422→    // RF-04: Track structural title context during parsing
   423→   423→    let currentSectionContext: string | null = null
   424→   424→    let currentFieldContext: string | null = null
   425→   425→
   426→   426→    const parsedRows: Array<{
   427→   427→      description: string
   428→   428→      date: string
   429→   429→      totalValue: number
   430→   430→      itemMonth: number
   431→   431→      itemYear: number
   432→   432→      dreField: string
   433→   433→    }> = []
   434→   434→
   435→   435→    let structuralTitlesFound = 0
   436→   436→
   437→   437→    worksheet.eachRow((row, rowNumber) => {
   438→   438→      if (rowNumber === 1) return
   439→   439→
   440→   440→      // Column A: Description
   441→   441→      const description = row.getCell(1).value?.toString()?.trim() || ''
   442→   442→      if (!description) return
   443→   443→
   444→   444→      // Column B: Date → parse to get month/year
   445→   445→      const rawDate = row.getCell(2).value
   446→   446→      const parsedDate = parseRowDate(rawDate)
   447→   447→      // If date is parsed, use it; otherwise default to selected month/year
   448→   448→      const itemMonth = parsedDate?.month ?? month
   449→   449→      const itemYear = parsedDate?.year ?? year
   450→   450→      const dateStr = parsedDate?.dateStr ?? ''
   451→   451→
   452→   452→      // Column C: Value
   453→   453→      const valueCell = row.getCell(3)
   454→   454→      const value = parseCellToNumber(valueCell.value, { font: valueCell.font } as ExcelCellLike)
   455→   455→
   456→   456→      // RF-04: Check if this row is a structural title
   457→   457→      const titleInfo = detectStructuralTitle(description)
   458→   458→
   459→   459→      if (titleInfo) {
   460→   460→        structuralTitlesFound++
   461→   461→        if (titleInfo.type === 'section') {
   462→   462→          currentSectionContext = titleInfo.sectionKey ?? null
   463→   463→          currentFieldContext = null
   464→   464→        } else if (titleInfo.type === 'field') {
   465→   465→          currentFieldContext = titleInfo.fieldKey
   466→   466→          currentSectionContext = titleInfo.sectionKey ?? null
   467→   467→        }
   468→   468→        // Structural titles with no value are skipped as data rows
   469→   469→        if (value === 0) return
   470→   470→      }
   471→   471→
   472→   472→      // Column D: Category (explicit auto-classification, takes priority)
   473→   473→      const categoryCell = row.getCell(4)
   474→   474→      let dreField = ''
   475→   475→      if (categoryCell && categoryCell.value) {
   476→   476→        const categoryText = String(categoryCell.value).trim()
   477→   477→        dreField = classifyByCategory(categoryText) || ''
   478→   478→      }
   479→   479→
   480→   480→      // RF-04: If no category, use structural context for auto-classification
   481→   481→      if (!dreField) {
   482→   482→        if (currentFieldContext) {
   483→   483→          dreField = currentFieldContext
   484→   484→        } else if (titleInfo?.type === 'section' && titleInfo.sectionKey) {
   485→   485→          // Section-level title with a value: map to a catch-all field in that section
   486→   486→          // e.g. "Receita Bruta" with value → naoIdentificado
   487→   487→          dreField = sectionTitleToField(titleInfo.sectionKey)
   488→   488→        } else if (currentSectionContext) {
   489→   489→          const autoField = autoClassifyInSection(description, currentSectionContext)
   490→   490→          if (autoField) dreField = autoField
   491→   491→        }
   492→   492→      }
   493→   493→
   494→   494→      if (value !== 0) {
   495→   495→        parsedRows.push({ description, date: dateStr, totalValue: value, itemMonth, itemYear, dreField })
   496→   496→      }
   497→   497→    })
   498→   498→
   499→   499→    if (parsedRows.length === 0 && receitaBrutaValue === 0) {
   500→   500→      return NextResponse.json(
   501→   501→        { error: 'Nenhum dado válido encontrado. Verifique: Descrição (A), Data (B), Valor (C).' },
   502→   502→        { status: 400 }
   503→   503→      )
   504→   504→    }
   505→   505→
   506→   506→    // Create uploaded file record (use month=0, year=0 to indicate multi-month file)
   507→   507→    const uploadedFile = await db.uploadedFile.create({
   508→   508→      data: { companyId, month: 0, year: 0, fileName: file.name },
   509→   509→    })
   510→   510→
   511→   511→    // If "Caixa YYYY" worksheet was found, add Receita Bruta as a qualification item
   512→   512→    // Use the upload year (ano corrente) — NOT the year from the sheet name
   513→   513→    // The auto-update below will inject it into the DRE
   514→   514→    let receitaBrutaInserted = false
   515→   515→    if (receitaBrutaValue !== 0) {
   516→   516→      await db.expenseQualification.create({
   517→   517→        data: {
   518→   518→          companyId,
   519→   519→          month: 1,
   520→   520→          year: year,
   521→   521→          description: 'Receita Bruta',
   522→   522→          date: `${year}-01-01`,
   523→   523→          totalValue: receitaBrutaValue,
   524→   524→          dreField: 'prestacaoNf',
   525→   525→          uploadedFileId: uploadedFile.id,
   526→   526→        },
   527→   527→      })
   528→   528→      receitaBrutaInserted = true
   529→   529→    }
   530→   530→
   531→   531→    // Save all other qualifications
   532→   532→    const qualifications = parsedRows.map((item) => ({
   533→   533→      companyId,
   534→   534→      month: item.itemMonth,
   535→   535→      year: item.itemYear,
   536→   536→      description: item.description,
   537→   537→      date: item.date,
   538→   538→      totalValue: item.totalValue,
   539→   539→      dreField: item.dreField,
   540→   540→      uploadedFileId: uploadedFile.id,
   541→   541→    }))
   542→   542→
   543→   543→    if (qualifications.length > 0) {
   544→   544→      await db.expenseQualification.createMany({ data: qualifications })
   545→   545→    }
   546→   546→
   547→   547→    const savedItems = await db.expenseQualification.findMany({
   548→   548→      where: { uploadedFileId: uploadedFile.id },
   549→   549→      orderBy: { createdAt: 'asc' },
   550→   550→    })
   551→   551→
   552→   552→    // Build distributed months list
   553→   553→    const groupedByMonth = new Map<string, number[]>()
   554→   554→    for (const row of parsedRows) {
   555→   555→      const key = `${row.itemMonth}-${row.itemYear}`
   556→   556→      if (!groupedByMonth.has(key)) groupedByMonth.set(key, [])
   557→   557→      groupedByMonth.get(key)!.push(row.itemMonth)
   558→   558→    }
   559→   559→    const distributedMonths = Array.from(groupedByMonth.keys())
   560→   560→      .map(key => {
   561→   561→        const [m, y] = key.split('-')
   562→   562→        return `${String(parseInt(m)).padStart(2, '0')}/${y}`
   563→   563→      })
   564→   564→      .sort((a, b) => {
   565→   565→        const [am, ay] = a.split('/').map(Number)
   566→   566→        const [bm, by] = b.split('/').map(Number)
   567→   567→        return ay !== by ? ay - by : am - bm
   568→   568→      })
   569→   569→
   570→   570→    // Count auto-classified items
   571→   571→    const autoClassified = savedItems.filter(i => i.dreField).length
   572→   572→
   573→   573→    // Auto-update DRE entries for months that have auto-classified items (especially C3/Receita Bruta)
   574→   574→    const autoClassifiedItems = savedItems.filter(i => i.dreField)
   575→   575→    const autoMonthKeys = new Set(autoClassifiedItems.map(i => `${i.month}-${i.year}`))
   576→   576→    let updatedDREForMonths: Record<string, any> = {}
   577→   577→    for (const monthKey of autoMonthKeys) {
   578→   578→      const [mStr, yStr] = monthKey.split('-')
   579→   579→      const m = parseInt(mStr)
   580→   580→      const y = parseInt(yStr)
   581→   581→      const monthItems = autoClassifiedItems.filter(i => i.month === m && i.year === y)
   582→   582→      
   583→   583→      const fieldTotals: Record<string, number> = {}
   584→   584→      for (const item of monthItems) {
   585→   585→        if (item.dreField && !item.dreField.startsWith('custom_')) {
   586→   586→          fieldTotals[item.dreField] = (fieldTotals[item.dreField] || 0) + item.totalValue
   587→   587→        }
   588→   588→      }
   589→   589→
   590→   590→      if (Object.keys(fieldTotals).length > 0) {
   591→   591→        const existing = await db.dREEntry.findUnique({
   592→   592→          where: { companyId_month_year: { companyId, month: m, year: y } },
   593→   593→        })
   594→   594→        if (existing) {
   595→   595→          // Merge: add to existing values
   596→   596→          const mergedData: Record<string, number> = {}
   597→   597→          for (const [field, total] of Object.entries(fieldTotals)) {
   598→   598→            mergedData[field] = ((existing as any)[field] || 0) + total
   599→   599→          }
   600→   600→          await db.dREEntry.update({
   601→   601→            where: { id: existing.id },
   602→   602→            data: mergedData,
   603→   603→          })
   604→   604→          // Fetch updated entry for response
   605→   605→          const updated = await db.dREEntry.findUnique({ where: { id: existing.id } })
   606→   606→          if (updated) {
   607→   607→            const { id: _id, companyId: _c, month: _m2, year: _y2, createdAt, updatedAt, company, ...fields } = updated as any
   608→   608→            updatedDREForMonths[monthKey] = fields
   609→   609→          }
   610→   610→        } else {
   611→   611→          const created = await db.dREEntry.create({
   612→   612→            data: { companyId, month: m, year: y, ...fieldTotals },
   613→   613→          })
   614→   614→          const { id: _id, companyId: _c, month: _m2, year: _y2, createdAt, updatedAt, company, ...fields } = created as any
   615→   615→          updatedDREForMonths[monthKey] = fields
   616→   616→        }
   617→   617→      }
   618→   618→    }
   619→   619→
   620→   620→    // Exclude the auto-generated "Receita Bruta" C3 item from the items list
   621→   621→    // since it's already applied directly to the DRE as prestacaoNf
   622→   622→    const receitaBrutaItemId = receitaBrutaInserted
   623→   623→      ? savedItems.find(i => i.description === 'Receita Bruta' && i.dreField === 'prestacaoNf' && i.month === 1)?.id
   624→   624→      : null
   625→   625→
   626→   626→    return NextResponse.json({
   627→   627→      items: savedItems
   628→   628→        .filter(i => i.id !== receitaBrutaItemId)
   629→   629→        .map(i => ({
   630→   630→          id: i.id, description: i.description, date: i.date,
   631→   631→          totalValue: i.totalValue, dreField: i.dreField,
   632→   632→          month: i.month, year: i.year,
   633→   633→        })),
   634→   634→      file: { id: uploadedFile.id, fileName: uploadedFile.fileName, uploadedAt: uploadedFile.uploadedAt },
   635→   635→      distributedMonths,
   636→   636→      structuralTitlesFound,
   637→   637→      autoClassified,
   638→   638→      updatedDREForMonths,
   639→   639→      receitaBrutaInserted,
   640→   640→      receitaBrutaValue,
   641→   641→      receitaBrutaYear,
   642→   642→    })
   643→   643→  } catch (error) {
   644→   644→    console.error('Expense qualification error:', error)
   645→   645→    return NextResponse.json({ error: 'Erro ao processar planilha de despesas' }, { status: 500 })
   646→   646→  }
   647→   647→}

   649→   649→// ─── PUT: Save expense classification and update DRE (RF-07: returns updated DRE data) ───
   650→   650→
   651→   651→export async function PUT(req: NextRequest) {
   652→   652→  try {
   653→   653→    const session = await getServerSession(authOptions)
   654→   654→    if (!session?.user?.id) {
   655→   655→      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
   656→   656→    }
   657→   657→
   658→   658→    const userId = (session.user as any).id
   659→   659→    const body = await req.json()
   660→   660→    const { companyId, month, year, items } = body
   661→   661→
   662→   662→    if (!companyId || !month || !year || !items) {
   663→   663→      return NextResponse.json({ error: 'Parâmetros incompletos' }, { status: 400 })
   664→   664→    }
   665→   665→
   666→   666→    const hasAccess = await verifyAccess(userId, companyId)
   667→   667→    if (!hasAccess) {
   668→   668→      return NextResponse.json({ error: 'Sem permissão' }, { status: 403 })
   669→   669→    }
   670→   670→
   671→   671→    // Update each item's dreField mapping in parallel
   672→   672→    const itemsToUpdate = items.filter((item: any) => item.id && item.dreField)
   673→   673→    await Promise.all(
   674→   674→      itemsToUpdate.map((item: any) =>
   675→   675→        db.expenseQualification.update({
   676→   676→          where: { id: item.id },
   677→   677→          data: { dreField: item.dreField },
   678→   678→        })
   679→   679→      )
   680→   680→    )
   681→   681→
   682→   682→    // Find all distinct month/year combos that have classified items for this company
   683→   683→    const allClassified = await db.expenseQualification.findMany({
   684→   684→      where: { companyId, dreField: { not: '' } },
   685→   685→      select: { month: true, year: true },
   686→   686→      distinct: ['month', 'year'],
   687→   687→    })
   688→   688→
   689→   689→    // Update DRE entries for ALL months that have classified items
   690→   690→    let currentMonthDreEntryId: string | null = null
   691→   691→    let currentMonthFieldTotals: Record<string, number> = {}
   692→   692→    for (const { month: m, year: y } of allClassified) {
   693→   693→      const monthQualifications = await db.expenseQualification.findMany({
   694→   694→        where: { companyId, month: m, year: y, dreField: { not: '' } },
   695→   695→      })
   696→   696→
   697→   697→      const fieldTotals: Record<string, number> = {}
   698→   698→      const customFieldTotals: Record<string, number> = {}
   699→   699→      for (const q of monthQualifications) {
   700→   700→        if (q.dreField) {
   701→   701→          if (q.dreField.startsWith('custom_')) {
   702→   702→            const customFieldId = q.dreField.replace('custom_', '')
   703→   703→            customFieldTotals[customFieldId] = (customFieldTotals[customFieldId] || 0) + q.totalValue
   704→   704→          } else {
   705→   705→            fieldTotals[q.dreField] = (fieldTotals[q.dreField] || 0) + q.totalValue
   706→   706→          }
   707→   707→        }
   708→   708→      }
   709→   709→
   710→   710→      // Save field totals for the currently selected month
   711→   711→      if (m === month && y === year) {
   712→   712→        currentMonthFieldTotals = fieldTotals
   713→   713→      }
   714→   714→
   715→   715→      const existing = await db.dREEntry.findUnique({
   716→   716→        where: { companyId_month_year: { companyId, month: m, year: y } },
   717→   717→      })
   718→   718→
   719→   719→      const updateData: Record<string, number> = {}
   720→   720→      for (const [field, total] of Object.entries(fieldTotals)) {
   721→   721→        updateData[field] = total
   722→   722→      }
   723→   723→
   724→   724→      let dreEntryId: string | null = null
   725→   725→      if (Object.keys(updateData).length > 0 || Object.keys(customFieldTotals).length > 0) {
   726→   726→        if (existing) {
   727→   727→          dreEntryId = existing.id
   728→   728→          if (Object.keys(updateData).length > 0) {
   729→   729→            await db.dREEntry.update({
   730→   730→              where: { id: existing.id },
   731→   731→              data: updateData,
   732→   732→            })
   733→   733→          }
   734→   734→        } else {
   735→   735→          const created = await db.dREEntry.create({
   736→   736→            data: { companyId, month: m, year: y, ...updateData },
   737→   737→          })
   738→   738→          dreEntryId = created.id
   739→   739→        }
   740→   740→      } else {
   741→   741→        dreEntryId = existing?.id || null
   742→   742→      }
   743→   743→
   744→   744→      // Handle custom field totals (with sign correction)
   745→   745→      if (dreEntryId && Object.keys(customFieldTotals).length > 0) {
   746→   746→        const customFieldIds = Object.keys(customFieldTotals)
   747→   747→        const customFields = await db.customDREField.findMany({
   748→   748→          where: { id: { in: customFieldIds } },
   749→   749→        })
   750→   750→        const fieldSignMap: Record<string, string> = {}
   751→   751→        for (const cf of customFields) fieldSignMap[cf.id] = cf.sign
   752→   752→
   753→   753→        await Promise.all(
   754→   754→          Object.entries(customFieldTotals).map(([customFieldId, total]) => {
   755→   755→            const sign = fieldSignMap[customFieldId]
   756→   756→            const adjustedTotal = sign === '-' ? -total : total
   757→   757→            return db.customDREValue.upsert({
   758→   758→              where: { customFieldId_dREEntryId: { customFieldId, dREEntryId } },
   759→   759→              create: { customFieldId, dREEntryId, value: adjustedTotal },
   760→   760→              update: { value: adjustedTotal },
   761→   761→            })
   762→   762→          })
   763→   763→        )
   764→   764→      }
   765→   765→
   766→   766→      // Track the entry ID for the currently selected month
   767→   767→      if (m === month && y === year) {
   768→   768→        currentMonthDreEntryId = dreEntryId
   769→   769→      }
   770→   770→    }
   771→   771→
   772→   772→    // RF-07: Fetch the updated DRE entry to return for reactive sync
   773→   773→    let updatedDREData: Record<string, any> | null = null
   774→   774→    if (currentMonthDreEntryId) {
   775→   775→      const updatedEntry = await db.dREEntry.findUnique({ where: { id: currentMonthDreEntryId } })
   776→   776→      if (updatedEntry) {
   777→   777→        const { id, companyId: _cid, month: _m, year: _y, createdAt, updatedAt, company, ...fields } = updatedEntry as any
   778→   778→        updatedDREData = fields
   779→   779→
   780→   780→        // Also fetch custom field values
   781→   781→        const customValues = await db.customDREValue.findMany({
   782→   782→          where: { dREEntryId: currentMonthDreEntryId },
   783→   783→          include: { customField: true },
   784→   784→        })
   785→   785→        if (customValues.length > 0) {
   786→   786→          updatedDREData._customFieldValues = {}
   787→   787→          for (const cv of customValues) {
   788→   788→            const sign = (cv.customField as any).sign || '+'
   789→   789→            updatedDREData._customFieldValues[cv.customFieldId] = sign === '-' ? -cv.value : cv.value
   790→   790→          }
   791→   791→        }
   792→   792→      }
   793→   793→    }
   794→   794→
   795→   795→    return NextResponse.json({
   796→   796→      success: true,
   797→   797→      fieldTotals: currentMonthFieldTotals,
   798→   798→      dreEntryId: currentMonthDreEntryId,
   799→   799→      updatedDREData, // RF-07: reactive sync payload
   800→   800→    })
   801→   801→  } catch (error) {
   802→   802→    console.error('Error updating qualification:', error)
   803→   803→    return NextResponse.json({ error: 'Erro ao atualizar qualificação' }, { status: 500 })
   804→   804→  }
   805→   805→}

   809→   809→export async function DELETE(req: NextRequest) {
   810→   810→  try {
   811→   811→    const session = await getServerSession(authOptions)
   812→   812→    if (!session?.user?.id) {
   813→   813→      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
   814→   814→    }
   815→   815→
   816→   816→    const userId = (session.user as any).id
   817→   817→    const { searchParams } = new URL(req.url)
   818→   818→    const fileIdsParam = searchParams.get('fileIds')

   821→   821→      return NextResponse.json({ error: 'fileIds obrigatório' }, { status: 400 })
   822→   822→    }

   827→   827→    }
   828→   828→
   829→   829→    const files = await db.uploadedFile.findMany({ where: { id: { in: ids } } })
   830→   830→    const uniqueCompanyIds = [...new Set(files.map(f => f.companyId))]
   831→   831→    const accessChecks = await Promise.all(
   832→   832→      uniqueCompanyIds.map(cid => verifyAccess(userId, cid))
   833→   833→    )
   834→   834→    const accessMap = new Map(uniqueCompanyIds.map((cid, i) => [cid, accessChecks[i]]))
   835→   835→
   836→   836→    for (const file of files) {
   837→   837→      if (!accessMap.get(file.companyId)) {
   838→   838→        return NextResponse.json({ error: 'Sem permissão para excluir arquivo' }, { status: 403 })
   839→   839→      }
   840→   840→    }
   841→   841→
   842→   842→    const deleteResult = await db.expenseQualification.deleteMany({
   843→   843→      where: { uploadedFileId: { in: ids } },
   844→   844→    })
   845→   845→
   846→   846→    const fileDeleteResult = await db.uploadedFile.deleteMany({
   847→   847→      where: { id: { in: ids } },
   848→   848→    })
   849→   849→
   850→   850→    return NextResponse.json({ success: true, deletedFiles: fileDeleteResult.count, deletedItems: deleteResult.count })
   851→   851→  } catch (error) {
   852→   852→    console.error('Error deleting uploaded files:', error)
   853→   853→    return NextResponse.json({ error: 'Erro ao excluir arquivos' }, { status: 500 })
   854→   854→  }
   855→   855→}

   859→   859→export async function GET(req: NextRequest) {
   860→   860→  try {
   861→   861→    const session = await getServerSession(authOptions)
   862→   862→    if (!session?.user?.id) {
   863→   863→      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
   864→   864→    }
   865→   865→
   866→   866→    const userId = (session.user as any).id
   867→   867→    const { searchParams } = new URL(req.url)
   868→   868→    const companyId = searchParams.get('companyId')
   869→   869→    const month = searchParams.get('month')
   870→   870→    const year = searchParams.get('year')
   871→   871→
   872→   872→    if (!companyId || !month || !year) {
   873→   873→      return NextResponse.json({ error: 'Parâmetros incompletos' }, { status: 400 })
   874→   874→    }
   875→   875→
   876→   876→    const hasAccess = await verifyAccess(userId, companyId)
   877→   877→    if (!hasAccess) {
   878→   878→      return NextResponse.json({ error: 'Sem permissão' }, { status: 403 })
   879→   879→    }
   880→   880→
   881→   881→    const qualifications = await db.expenseQualification.findMany({
   882→   882→      where: { companyId, month: parseInt(month), year: parseInt(year) },
   883→   883→      orderBy: { createdAt: 'asc' },
   884→   884→    })
   885→   885→
   886→   886→    // Show uploaded files that have expense items in the current month/year
   887→   887→    const distinctFileIds = qualifications
   888→   888→      .map(q => q.uploadedFileId)
   889→   889→      .filter((id): id is string => id !== null && id !== undefined)
   890→   890→    const uniqueFileIds = [...new Set(distinctFileIds)]
   891→   891→    let uploadedFiles: any[] = []
   892→   892→    if (uniqueFileIds.length > 0) {
   893→   893→      uploadedFiles = await db.uploadedFile.findMany({
   894→   894→        where: { id: { in: uniqueFileIds } },
   895→   895→        orderBy: { uploadedAt: 'desc' },
   896→   896→      })
   897→   897→    }
   898→   898→
   899→   899→    return NextResponse.json({
   900→   900→      // Exclude auto-generated "Receita Bruta" C3 items from the list (already in DRE)
   901→   901→      items: qualifications
   902→   902→        .filter(q => !(q.description === 'Receita Bruta' && q.dreField === 'prestacaoNf' && q.month === 1))
   903→   903→        .map(q => ({
   904→   904→          id: q.id, description: q.description, date: q.date,
   905→   905→          totalValue: q.totalValue, dreField: q.dreField,
   906→   906→          uploadedFileId: q.uploadedFileId,
   907→   907→        })),
   908→   908→      uploadedFiles: uploadedFiles.map(f => ({
   909→   909→        id: f.id, fileName: f.fileName, uploadedAt: f.uploadedAt,
   910→   910→      })),
   911→   911→    })
   912→   912→  } catch (error) {
   913→   913→    return NextResponse.json({ error: 'Erro ao buscar qualificações' }, { status: 500 })
   914→   914→  }
   915→   915→}
