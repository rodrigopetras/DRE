   179→   179→// ─── DRE Calculations ───
   180→   180→function calcDRE(d: DREData, customFieldValues?: Record<string, number>, customFields?: CustomFieldDef[]) {
   181→   181→  const receitaBruta = d.prestacaoNf + d.honorariosSucumbenciais + d.honorarios + d.reembolsoDespesas + d.naoIdentificado
   182→   182→  const receitaLiquida = receitaBruta - d.impostosCpp
   183→   183→  const despesasVariaveis =
   184→   184→    d.combustivel + d.insumos + d.materialEscritorio + d.servicosContratados +
   185→   185→    d.materiaisUsoConsumo + d.taxas + d.tarifas + d.energiaEletrica + d.agua
   186→   186→  const despesasFixas =
   187→   187→    d.aluguel + d.salarios + d.fgts + d.prolabore + d.bonus +
   188→   188→    d.inss + d.iptu + d.telefoneInternet + d.valeTransporte + d.planoSaude
   189→   189→
   190→   190→  // Add custom field values
   191→   191→  let customReceita = 0
   192→   192→  let customDespVar = 0
   193→   193→  let customDespFixas = 0
   194→   194→  if (customFieldValues && customFields) {
   195→   195→    for (const cf of customFields) {
   196→   196→      const val = customFieldValues[cf.id] || 0
   197→   197→      if (cf.sign === '+') {
   198→   198→        if (cf.section === 'receita') customReceita += val
   199→   199→        else if (cf.section === 'desp_variaveis') customDespVar += val
   200→   200→        else if (cf.section === 'desp_fixas') customDespFixas += val
   201→   201→      } else {
   202→   202→        if (cf.section === 'receita') customReceita -= val
   203→   203→        else if (cf.section === 'desp_variaveis') customDespVar -= val
   204→   204→        else if (cf.section === 'desp_fixas') customDespFixas -= val
   205→   205→      }
   206→   206→    }
   207→   207→  }
   208→   208→
   209→   209→  const totalReceitaBruta = receitaBruta + customReceita
   210→   210→  const totalReceitaLiquida = totalReceitaBruta - d.impostosCpp
   211→   211→  const totalDespesasVariaveis = despesasVariaveis + customDespVar
   212→   212→  const totalDespesasFixas = despesasFixas + customDespFixas
   213→   213→  const resultadoLiquido = totalReceitaLiquida - totalDespesasVariaveis - totalDespesasFixas
   214→   214→  return { receitaBruta: totalReceitaBruta, receitaLiquida: totalReceitaLiquida, despesasVariaveis: totalDespesasVariaveis, despesasFixas: totalDespesasFixas, resultadoLiquido }
   215→   215→}

   218→   218→function ProjNumInput({ value, onChange, className }: {
   219→   219→  value: number; onChange: (v: number) => void; className?: string
   220→   220→}) {
   221→   221→  const [focused, setFocused] = useState(false)
   222→   222→  const [local, setLocal] = useState('')
   223→   223→  const inputRef = useRef<HTMLInputElement>(null)
   224→   224→
   225→   225→  const displayValue = focused
   226→   226→    ? local
   227→   227→    : (value === 0 ? '' : value.toFixed(2).replace('.', ','))
   228→   228→
   229→   229→  const handleFocus = () => {
   230→   230→    setFocused(true)
   231→   231→    setLocal(value === 0 ? '' : value.toFixed(2).replace('.', ','))
   232→   232→    // Move cursor to start
   233→   233→    setTimeout(() => {
   234→   234→      if (inputRef.current) {
   235→   235→        inputRef.current.setSelectionRange(0, 0)
   236→   236→      }
   237→   237→    }, 0)
   238→   238→  }
   239→   239→
   240→   240→  const handleBlur = () => {
   241→   241→    setFocused(false)
   242→   242→    const parsed = parseFloat(local.replace(/\./g, '').replace(',', '.')) || 0
   243→   243→    onChange(parsed)
   244→   244→  }
   245→   245→
   246→   246→  const handleChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   247→   247→    const val = e.target.value
   248→   248→    // Allow only numbers, commas, dots
   249→   249→    const cleaned = val.replace(/[^\d.,-]/g, '')
   250→   250→    setLocal(cleaned)
   251→   251→    const parsed = parseFloat(cleaned.replace(/\./g, '').replace(',', '.')) || 0
   252→   252→    if (!isNaN(parsed)) onChange(parsed)
   253→   253→  }
   254→   254→
   255→   255→  return (
   256→   256→    <input
   257→   257→      ref={inputRef}
   258→   258→      type="text"
   259→   259→      inputMode="decimal"
   260→   260→      className={`w-28 text-right font-mono bg-white border rounded px-1.5 py-0.5 text-xs ${className || ''}`}
   261→   261→      value={displayValue}
   262→   262→      onFocus={handleFocus}
   263→   263→      onBlur={handleBlur}
   264→   264→      onChange={handleChange}
   265→   265→    />
   266→   266→  )
   267→   267→}

   270→   270→function NumInput({ label, value, onChange, prefix, isViewer, onSuggestChange, isSuggested, dreField }: {
   271→   271→  label: string; value: number; onChange: (v: number) => void; prefix?: string
   272→   272→  isViewer?: boolean; onSuggestChange?: (field: string, newValue: number) => void; isSuggested?: boolean; dreField?: string
   273→   273→}) {
   274→   274→  const [editing, setEditing] = useState(false)
   275→   275→  const [local, setLocal] = useState('')
   276→   276→
   277→   277→  const commit = () => {
   278→   278→    setEditing(false)
   279→   279→    const parsed = parseFloat(local.replace(/\./g, '').replace(',', '.')) || 0
   280→   280→    if (isViewer && onSuggestChange && dreField && parsed !== value) {
   281→   281→      onSuggestChange(dreField, parsed)
   282→   282→      return
   283→   283→    }
   284→   284→    onChange(parsed)
   285→   285→  }
   286→   286→
   287→   287→  return (
   288→   288→    <div className="flex items-center justify-between py-1.5 group">
   289→   289→      <label className="text-sm text-muted-foreground flex items-center gap-2">
   290→   290→        {isSuggested && <span className="h-2 w-2 rounded-full bg-amber-400" />}
   291→   291→        {label}
   292→   292→      </label>
   293→   293→      <div className="flex items-center gap-1 w-44 justify-end">
   294→   294→        {prefix && <span className="text-xs text-muted-foreground">{prefix}</span>}
   295→   295→        <Input
   296→   296→          type="text"
   297→   297→          inputMode="decimal"
   298→   298→          className={`h-8 text-right text-sm font-mono ${editing ? 'ring-2 ring-primary/30' : ''} ${isSuggested ? 'bg-amber-100 border-amber-300' : ''} transition-all`}
   299→   299→          value={editing ? local : (value === 0 ? '' : brl(value))}
   300→   300→          onFocus={() => { setEditing(true); setLocal(value === 0 ? '' : value.toFixed(2).replace('.', ',')) }}
   301→   301→          onBlur={commit}
   302→   302→          onChange={(e) => {
   303→   303→            setLocal(e.target.value)
   304→   304→            const parsed = parseFloat(e.target.value.replace(/\./g, '').replace(',', '.'))
   305→   305→            if (!isNaN(parsed)) onChange(parsed)
   306→   306→          }}
   307→   307→          onKeyDown={(e) => e.key === 'Enter' && commit()}
   308→   308→          placeholder="0,00"
   309→   309→        />
   310→   310→      </div>
   311→   311→    </div>
   312→   312→  )
   313→   313→}

   361→   361→          email: email.trim(),

   373→   373→        const err = await res.json()

   375→   375→      }
   376→   376→    } catch {
   377→   377→      toast.error('Erro ao fazer login')

   394→   394→        </DialogHeader>
   395→   395→        <form onSubmit={handleSubmit} className="space-y-4 py-2">
   396→   396→          <div>
   397→   397→            <Label className="mb-1.5 block">Nome</Label>
   398→   398→            <Input placeholder="Seu nome" value={name} onChange={(e) => setName(e.target.value)} required />
   399→   399→          </div>
   400→   400→          <div>
   401→   401→            <Label className="mb-1.5 block">Email</Label>
   402→   402→            <Input type="email" placeholder="seu@email.com" value={email} onChange={(e) => setEmail(e.target.value)} required />
   403→   403→          </div>
   404→   404→          <DialogFooter>
   405→   405→            <Button type="button" variant="outline" onClick={onClose}>Cancelar</Button>
   406→   406→            <Button type="submit" disabled={loading}>
   407→   407→              {loading ? <Loader2 className="h-4 w-4 mr-2 animate-spin" /> : null}
   408→   408→              Continuar
   409→   409→            </Button>
   410→   410→          </DialogFooter>
   411→   411→        </form>
   412→   412→      </DialogContent>
   413→   413→    </Dialog>
   414→   414→  )
