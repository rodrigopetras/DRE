   161→// ─── Formatters ───
   162→function brl(value: number) {
   163→  return new Intl.NumberFormat('pt-BR', { style: 'currency', currency: 'BRL' }).format(value || 0)
   164→}
   165→function pct(value: number) {
   166→  if (!isFinite(value) || isNaN(value)) return '0,00%'
   167→  return new Intl.NumberFormat('pt-BR', { style: 'percent', minimumFractionDigits: 2, maximumFractionDigits: 2 }).format(value || 0)
   168→}
   169→function safeDiv(a: number, b: number) {
   170→  return b === 0 ? 0 : a / b
   171→}

   173→// ─── DRE Calculations ───
   174→function calcDRE(d: DREData) {
   175→  const receitaBruta = d.prestacaoNf + d.honorariosSucumbenciais
   176→  const receitaLiquida = receitaBruta - d.impostosCpp
   177→  const despesasVariaveis =
   178→    d.combustivel + d.insumos + d.materialEscritorio + d.servicosContratados +
   179→    d.materiaisUsoConsumo + d.taxas + d.tarifas + d.energiaEletrica + d.agua
   180→  const despesasFixas =
   181→    d.aluguel + d.salarios + d.fgts + d.prolabore + d.bonus +
   182→    d.inss + d.iptu + d.telefoneInternet + d.valeTransporte + d.planoSaude
   183→  const resultadoLiquido = receitaLiquida - despesasVariaveis - despesasFixas
   184→  return { receitaBruta, receitaLiquida, despesasVariaveis, despesasFixas, resultadoLiquido }
   185→}

   187→// ─── Number Input Component ───
   188→function NumInput({ label, value, onChange, prefix, isViewer, onSuggestChange, isSuggested, dreField }: {
   189→  label: string; value: number; onChange: (v: number) => void; prefix?: string
   190→  isViewer?: boolean; onSuggestChange?: (field: string, newValue: number) => void; isSuggested?: boolean; dreField?: string
   191→}) {
   192→  const [editing, setEditing] = useState(false)
   193→  const [local, setLocal] = useState('')
   194→
   195→  const commit = () => {
   196→    setEditing(false)
   197→    const parsed = parseFloat(local.replace(/\./g, '').replace(',', '.')) || 0
   198→    if (isViewer && onSuggestChange && dreField && parsed !== value) {
   199→      onSuggestChange(dreField, parsed)
   200→      return
   201→    }
   202→    onChange(parsed)
   203→  }
   204→
   205→  return (
   206→    <div className="flex items-center justify-between py-1.5 group">
   207→      <label className="text-sm text-muted-foreground flex items-center gap-2">
   208→        {isSuggested && <span className="h-2 w-2 rounded-full bg-amber-400" />}
   209→        {label}
   210→      </label>
   211→      <div className="flex items-center gap-1 w-44 justify-end">
   212→        {prefix && <span className="text-xs text-muted-foreground">{prefix}</span>}
   213→        <Input
   214→          type="text"
   215→          inputMode="decimal"
   216→          className={`h-8 text-right text-sm font-mono ${editing ? 'ring-2 ring-primary/30' : ''} ${isSuggested ? 'bg-amber-100 border-amber-300' : ''} transition-all`}
   217→          value={editing ? local : (value === 0 ? '' : brl(value))}
   218→          onFocus={() => { setEditing(true); setLocal(value === 0 ? '' : value.toFixed(2).replace('.', ',')) }}
   219→          onBlur={commit}
   220→          onChange={(e) => {
   221→            setLocal(e.target.value)
   222→            const parsed = parseFloat(e.target.value.replace(/\./g, '').replace(',', '.'))
   223→            if (!isNaN(parsed)) onChange(parsed)
   224→          }}
   225→          onKeyDown={(e) => e.key === 'Enter' && commit()}
   226→          placeholder="0,00"
   227→        />
   228→      </div>
   229→    </div>
   230→  )
   231→}
