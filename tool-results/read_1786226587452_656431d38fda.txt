    86→// RF-12: Hydration guard component - prevents flash of default state
    87→function HydrationGuard({ children }: { children: React.ReactNode }) {
    88→  const hydrated = useAppStore(s => s._hydrated)
    89→  if (!hydrated) {
    90→    return (
    91→      <div className="min-h-screen flex items-center justify-center bg-background">
    92→        <Loader2 className="h-8 w-8 animate-spin text-muted-foreground" />
    93→      </div>
    94→    )
    95→  }
    96→  return <>{children}</>
    97→}

   100→function AppContent() {
   101→  const { data: session, status } = useSession()
   102→  const userId = (session?.user as any)?.id
   103→
   104→  if (status === 'loading') {
   105→    return (
   106→      <div className="min-h-screen flex items-center justify-center bg-background">
   107→        <Loader2 className="h-8 w-8 animate-spin text-muted-foreground" />
   108→      </div>
   109→    )
   110→  }
   111→
   112→  if (!session) {
   113→    return <LoginScreen />
   114→  }
   115→
   116→  return <HydrationGuard><DREApp session={session} userId={userId} /></HydrationGuard>
   117→}
