   119→// ─── Formatters ───
   120→function brl(value: number) {
   121→  return new Intl.NumberFormat('pt-BR', { style: 'currency', currency: 'BRL' }).format(value || 0)
   122→}
   123→function pct(value: number) {
   124→  if (!isFinite(value) || isNaN(value)) return '0,00%'
   125→  return new Intl.NumberFormat('pt-BR', { style: 'percent', minimumFractionDigits: 2, maximumFractionDigits: 2 }).format(value || 0)
   126→}
   127→function safeDiv(a: number, b: number) {
   128→  return b === 0 ? 0 : a / b
   129→}

   131→// ─── DRE Calculations ───
   132→function calcDRE(d: DREData) {
   133→  const receitaBruta = d.prestacaoNf + d.honorariosSucumbenciais
   134→  const receitaLiquida = receitaBruta - d.impostosCpp
   135→  const despesasVariaveis =
   136→    d.combustivel + d.insumos + d.materialEscritorio + d.servicosContratados +
   137→    d.materiaisUsoConsumo + d.taxas + d.tarifas + d.energiaEletrica + d.agua
   138→  const despesasFixas =
   139→    d.aluguel + d.salarios + d.fgts + d.prolabore + d.bonus +
   140→    d.inss + d.iptu + d.telefoneInternet + d.valeTransporte + d.planoSaude
   141→  const resultadoLiquido = receitaLiquida - despesasVariaveis - despesasFixas
   142→  return { receitaBruta, receitaLiquida, despesasVariaveis, despesasFixas, resultadoLiquido }
   143→}

   145→// ─── Number Input Component ───
   146→function NumInput({ label, value, onChange, prefix }: {
   147→  label: string; value: number; onChange: (v: number) => void; prefix?: string
   148→}) {
   149→  const [editing, setEditing] = useState(false)
   150→  const [local, setLocal] = useState('')
   151→
   152→  const formattedValue = value === 0 ? '' : brl(value)
   153→
   154→  const commit = () => {
   155→    setEditing(false)
   156→    const parsed = parseFloat(local.replace(/\./g, '').replace(',', '.')) || 0
   157→    onChange(parsed)
   158→  }
   159→
   160→  return (
   161→    <div className="flex items-center justify-between py-1.5 group">
   162→      <label className="text-sm text-muted-foreground flex items-center gap-2">
   163→        {label}
   164→      </label>
   165→      <div className="flex items-center gap-1 w-44 justify-end">
   166→        {prefix && <span className="text-xs text-muted-foreground">{prefix}</span>}
   167→        <Input
   168→          type="text"
   169→          inputMode="decimal"
   170→          className={`h-8 text-right text-sm font-mono ${editing ? 'ring-2 ring-primary/30' : ''} transition-all`}
   171→          value={editing ? local : (value === 0 ? '' : brl(value))}
   172→          onFocus={() => { setEditing(true); setLocal(value === 0 ? '' : value.toFixed(2).replace('.', ',')) }}
   173→          onBlur={commit}
   174→          onChange={(e) => {
   175→            setLocal(e.target.value)
   176→            const parsed = parseFloat(e.target.value.replace(/\./g, '').replace(',', '.'))
   177→            if (!isNaN(parsed)) onChange(parsed)
   178→          }}
   179→          onKeyDown={(e) => e.key === 'Enter' && commit()}
   180→          placeholder="0,00"
   181→        />
   182→      </div>
   183→    </div>
   184→  )
   185→}
