    10→async function verifyAccess(userId: string, companyId: string) {
    11→  const company = await db.company.findUnique({ where: { id: companyId } })
    12→  if (!company) return false
    13→  if (company.userId === userId) return true
    14→  const share = await db.companyShare.findUnique({
    15→    where: { companyId_userId: { companyId, userId } },
    16→  })
    17→  return share?.status === 'accepted'
    18→}

    22→interface ExcelCellLike {
    23→  value: unknown
    24→  font?: { color?: { argb?: string } }
    25→}
    26→
    27→/**
    28→ * Detect if an Excel cell's font color is red.
    29→ * Red = alpha byte 0xFF (high byte), R > 150, G < 100, B < 100.
    30→ */
    31→function isRedFont(cell?: ExcelCellLike): boolean {
    32→  if (!cell?.font?.color?.argb) return false
    33→  const argb = cell.font.color.argb
    34→  if (argb.length < 8) return false
    35→  const alpha = parseInt(argb.substring(0, 2), 16)
    36→  if (alpha !== 0xFF) return false
    37→  const r = parseInt(argb.substring(2, 4), 16)
    38→  const g = parseInt(argb.substring(4, 6), 16)
    39→  const b = parseInt(argb.substring(6, 8), 16)
    40→  return r > 150 && g < 100 && b < 100
    41→}
    42→
    43→/**
    44→ * Parse any ExcelJS cell value into a number.
    45→ * Handles: number, formula, RichText, Hyperlink, string, Date.
    46→ * Parenthesized values like (780,00) are converted to negative.
    47→ * Red font color also makes the value negative.
    48→ */
    49→function parseCellToNumber(cellValue: unknown, cell?: ExcelCellLike): number {
    50→  let result = 0
    51→
    52→  if (typeof cellValue === 'number') {
    53→    result = cellValue
    54→  }
    55→  else if (cellValue && typeof cellValue === 'object' && 'result' in cellValue) {
    56→    const formulaResult = (cellValue as any).result
    57→    if (typeof formulaResult === 'number') result = formulaResult
    58→    else if (typeof formulaResult === 'string') result = parseTextToNumber(formulaResult)
    59→  }
    60→  else if (cellValue && typeof cellValue === 'object' && 'richText' in cellValue) {
    61→    const parts = (cellValue as any).richText as Array<{ text: string }>
    62→    const text = parts.map(p => p.text).join('')
    63→    result = parseTextToNumber(text)
    64→  }
    65→  else if (cellValue && typeof cellValue === 'object' && 'text' in cellValue && 'hyperlink' in cellValue) {
    66→    result = parseTextToNumber((cellValue as any).text)
    67→  }
    68→  else if (typeof cellValue === 'string') {
    69→    result = parseTextToNumber(cellValue)
    70→  }
    71→
    72→  if (result > 0 && isRedFont(cell)) {
    73→    result = -result
    74→  }
    75→
    76→  return result
    77→}
    78→
    79→/**
    80→ * Parse a text string like "1.234,56" or "(780,00)" into a number.
    81→ * Parenthesized values become negative.
    82→ */
    83→function parseTextToNumber(text: string): number {
    84→  const trimmed = text.trim()
    85→  if (!trimmed) return 0
    86→  const parenMatch = trimmed.match(/^\((.+)\)$/)
    87→  let numStr: string
    88→  let sign = 1
    89→  if (parenMatch) {
    90→    numStr = parenMatch[1]
    91→    sign = -1
    92→  } else {
    93→    numStr = trimmed
    94→    if (numStr.startsWith('-')) {
    95→      sign = -1
    96→      numStr = numStr.substring(1)
    97→    }
    98→  }
    99→  const cleaned = numStr.replace(/\./g, '').replace(',', '.')
   100→  const parsed = parseFloat(cleaned)
   101→  if (isNaN(parsed)) return 0
   102→  return sign * Math.abs(parsed)
   103→}

   107→function excelSerialToDate(serial: number): Date {
   108→  if (serial < 1) return new Date(NaN)
   109→  const epoch = new Date(1899, 11, 30)
   110→  const msPerDay = 24 * 60 * 60 * 1000
   111→  return new Date(epoch.getTime() + serial * msPerDay)
   112→}

   114→/** Remove accents/diacritics from a string for comparison */
   115→function normalizeAccents(str: string): string {
   116→  return str.normalize('NFD').replace(/[\u0300-\u036f]/g, '').toLowerCase().trim()
   117→}

   119→interface ParsedDateInfo {
   120→  month: number
   121→  year: number
   122→  dateStr: string
   123→}
   124→
   125→/**
   126→ * Parse a date value from an Excel cell (column B) into month/year.
   127→ * Supports: JS Date, DD/MM/YYYY, YYYY-MM-DD, MM/DD/YYYY, DD-MM-YYYY, Excel serial numbers.
   128→ */
   129→const MONTH_NAMES_PT: Record<string, number> = {
   130→  'janeiro': 1, 'fevereiro': 2, 'marco': 3, 'março': 3, 'abril': 4, 'maio': 5, 'junho': 6,
   131→  'julho': 7, 'agosto': 8, 'setembro': 9, 'outubro': 10, 'novembro': 11, 'dezembro': 12,
   132→  'jan': 1, 'fev': 2, 'mar': 3, 'abr': 4, 'mai': 5, 'jun': 6,
   133→  'jul': 7, 'ago': 8, 'set': 9, 'out': 10, 'nov': 11, 'dez': 12,
   134→}
   135→
   136→function parseRowDate(rawValue: unknown): ParsedDateInfo | null {
   137→  if (!rawValue) return null
   138→
   139→  if (rawValue instanceof Date && !isNaN(rawValue.getTime())) {
   140→    return {
   141→      month: rawValue.getMonth() + 1,
   142→      year: rawValue.getFullYear(),
   143→      dateStr: rawValue.toISOString().split('T')[0],
   144→    }
   145→  }
   146→
   147→  if (typeof rawValue === 'number' && rawValue > 25569 && rawValue < 100000) {
   148→    const d = excelSerialToDate(rawValue)
   149→    if (!isNaN(d.getTime())) {
   150→      return {
   151→        month: d.getMonth() + 1,
   152→        year: d.getFullYear(),
   153→        dateStr: d.toISOString().split('T')[0],
   154→      }
   155→    }
   156→  }
   157→
   158→  if (typeof rawValue === 'string') {
   159→    const s = rawValue.trim()
   160→    if (!s) return null
   161→
   162→    // YYYY-MM-DD
   163→    let m = s.match(/^(\d{4})-(\d{1,2})-(\d{1,2})/)
   164→    if (m) {
   165→      const yr = parseInt(m[1]); const mo = parseInt(m[2]); const dy = parseInt(m[3])
   166→      if (mo >= 1 && mo <= 12 && dy >= 1 && dy <= 31) {
   167→        return { month: mo, year: yr, dateStr: `${yr}-${String(mo).padStart(2, '0')}-${String(dy).padStart(2, '0')}` }
   168→      }
   169→    }
   170→
   171→    // DD/MM/YYYY or MM/DD/YYYY
   172→    m = s.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/)
   173→    if (m) {
   174→      const a = parseInt(m[1]); const b = parseInt(m[2]); const yr = parseInt(m[3])
   175→      let mo: number, dy: number
   176→      if (a > 12) { dy = a; mo = b }
   177→      else if (b > 12) { mo = a; dy = b }
   178→      else { dy = a; mo = b }
   179→      if (mo >= 1 && mo <= 12 && dy >= 1 && dy <= 31) {
   180→        return { month: mo, year: yr, dateStr: `${yr}-${String(mo).padStart(2, '0')}-${String(dy).padStart(2, '0')}` }
   181→      }
   182→    }
   183→
   184→    // DD-MM-YYYY
   185→    m = s.match(/^(\d{1,2})-(\d{1,2})-(\d{4})$/)
   186→    if (m) {
   187→      const a = parseInt(m[1]); const b = parseInt(m[2]); const yr = parseInt(m[3])
   188→      let mo: number, dy: number
   189→      if (a > 12) { dy = a; mo = b }
   190→      else if (b > 12) { mo = a; dy = b }
   191→      else { dy = a; mo = b }
   192→      if (mo >= 1 && mo <= 12 && dy >= 1 && dy <= 31) {
   193→        return { month: mo, year: yr, dateStr: `${yr}-${String(mo).padStart(2, '0')}-${String(dy).padStart(2, '0')}` }
   194→      }
   195→    }
   196→
   197→    // DD/MM/YY
   198→    m = s.match(/^(\d{1,2})\/(\d{1,2})\/(\d{2})$/)
   199→    if (m) {
   200→      const a = parseInt(m[1]); const b = parseInt(m[2])
   201→      let yr = parseInt(m[3])
   202→      yr = yr < 100 ? 2000 + yr : yr
   203→      let mo: number, dy: number
   204→      if (a > 12) { dy = a; mo = b }
   205→      else if (b > 12) { mo = a; dy = b }
   206→      else { dy = a; mo = b }
   207→      if (mo >= 1 && mo <= 12 && dy >= 1 && dy <= 31) {
   208→        return { month: mo, year: yr, dateStr: `${yr}-${String(mo).padStart(2, '0')}-${String(dy).padStart(2, '0')}` }
   209→      }
   210→    }
   211→
   212→    // Month name in Portuguese (e.g. "Janeiro 2025", "Jan/2025", "janeiro")
   213→    const normS = normalizeAccents(s)
   214→    for (const [monthName, monthNum] of Object.entries(MONTH_NAMES_PT)) {
   215→      const normName = normalizeAccents(monthName)
   216→      const regex = new RegExp(`\\b${normName}\\b`, 'i')
   217→      if (regex.test(normS)) {
   218→        // Try to extract year from the string
   219→        const yearMatch = s.match(/(\d{4})/)
   220→        const yr = yearMatch ? parseInt(yearMatch[1]) : new Date().getFullYear()
   221→        return { month: monthNum, year: yr, dateStr: `${yr}-${String(monthNum).padStart(2, '0')}-01` }
   222→      }
   223→    }
   224→
   225→    // MM/YYYY (just month and year)
   226→    m = s.match(/^(\d{1,2})\/(\d{4})$/)
   227→    if (m) {
   228→      const mo = parseInt(m[1]); const yr = parseInt(m[2])
   229→      if (mo >= 1 && mo <= 12) {
   230→        return { month: mo, year: yr, dateStr: `${yr}-${String(mo).padStart(2, '0')}-01` }
   231→      }
   232→    }
   233→  }
   234→
   235→  return null
   236→}

   289→/**
   290→ * RF-04: Detect if a description is a structural title and return its context.
   291→ * Returns { type: 'section' | 'field', fieldKey?, sectionKey? } or null.
   292→ */
   293→function detectStructuralTitle(description: string): { type: 'section' | 'field'; fieldKey?: string; sectionKey?: string } | null {
   294→  const norm = normalizeAccents(description)
   295→
   296→  // Check for section-level matches first
   297→  for (const entry of STRUCTURAL_TITLES) {
   298→    for (const pattern of entry.patterns) {
   299→      if (norm === normalizeAccents(pattern) || norm.includes(normalizeAccents(pattern))) {
   300→        if (entry.sectionKey && !entry.fieldKey) {
   301→          return { type: 'section', sectionKey: entry.sectionKey }
   302→        }
   303→        if (entry.fieldKey) {
   304→          return { type: 'field', fieldKey: entry.fieldKey, sectionKey: DRE_FIELD_SECTIONS[entry.fieldKey] }
   305→        }
   306→      }
   307→    }
   308→  }
   309→  return null
   310→}

   312→/**
   313→ * When a structural section title row has a value, map it to a reasonable DRE field.
   314→ * For receita section: maps to 'prestacaoNf' (main revenue field for law firms).
   315→ * For desp sections: maps to the first field in that section.
   316→ */
   317→function sectionTitleToField(sectionKey: string): string {
   318→  if (sectionKey === 'receita') return 'prestacaoNf'
   319→  // For expense sections, find the first field in that section
   320→  const sectionFields = Object.entries(DRE_FIELD_SECTIONS)
   321→    .filter(([, section]) => section === sectionKey)
   322→    .map(([key]) => key)
   323→  return sectionFields[0] || 'naoIdentificado'
   324→}

   326→/**
   327→ * RF-04: Given a section context, try to auto-classify a description into the first matching field in that section.
   328→ */
   329→function autoClassifyInSection(description: string, sectionKey: string): string | null {
   330→  const norm = normalizeAccents(description)
   331→  const sectionFields = Object.entries(DRE_FIELD_SECTIONS)
   332→    .filter(([, section]) => section === sectionKey)
   333→    .map(([key]) => key)
   334→
   335→  for (const fieldKey of sectionFields) {
   336→    const label = DRE_FIELD_LABELS[fieldKey]
   337→    if (!label) continue
   338→    const normLabel = normalizeAccents(label)
   339→    // Check if description contains field label keywords
   340→    const labelWords = normLabel.split(/\s+/).filter(w => w.length > 2)
   341→    for (const word of labelWords) {
   342→      if (norm.includes(word)) return fieldKey
   343→    }
   344→  }
   345→  return null
   346→}

   348→/**
   349→ * Try to match a category string from column D against DRE field labels.
   350→ */
   351→function classifyByCategory(category: string): string | null {
   352→  if (!category || !category.trim()) return null
   353→  const norm = normalizeAccents(category)
   354→
   355→  for (const key of Object.keys(DRE_FIELD_LABELS)) {
   356→    if (normalizeAccents(key) === norm) return key
   357→  }
   358→  for (const [key, value] of Object.entries(DRE_FIELD_LABELS)) {
   359→    if (normalizeAccents(value) === norm) return key
   360→  }
   361→  return null
   362→}

   364→// ─── POST: Upload spreadsheet ───
   365→
   366→export async function POST(req: NextRequest) {
   367→  try {
   368→    const session = await getServerSession(authOptions)
   369→    if (!session?.user?.id) {
   370→      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
   371→    }
   372→
   373→    const userId = (session.user as any).id
   374→    const formData = await req.formData()
   375→    const file = formData.get('file') as File
   376→    const companyId = formData.get('companyId') as string
   377→    const month = parseInt(formData.get('month') as string)
   378→    const year = parseInt(formData.get('year') as string)
   379→
   380→    if (!file || !companyId || !month || !year) {
   381→      return NextResponse.json({ error: 'Parâmetros incompletos' }, { status: 400 })
   382→    }
   383→
   384→    const hasAccess = await verifyAccess(userId, companyId)
   385→    if (!hasAccess) {
   386→      return NextResponse.json({ error: 'Sem permissão' }, { status: 403 })
   387→    }
   388→
   389→    const buffer = Buffer.from(await file.arrayBuffer())
   390→    const workbook = new ExcelJS.Workbook()
   391→    await workbook.xlsx.load(buffer)
   392→
   393→    if (workbook.worksheets.length === 0) {
   394→      return NextResponse.json({ error: 'Planilha vazia' }, { status: 400 })
   395→    }
   396→
   397→    // ─── Step 1: Look for "Caixa YYYY" worksheet to extract Receita Bruta from C3 ───
   398→    let receitaBrutaValue = 0
   399→    let receitaBrutaYear = year
   400→    const caixaPattern = /^caixa\s+(\d{4})$/i
   401→    for (const ws of workbook.worksheets) {
   402→      const match = ws.name?.trim().match(caixaPattern)
   403→      if (match) {
   404→        receitaBrutaYear = parseInt(match[1])
   405→        // Read specifically cell C3 (row 3, column 3)
   406→        if (ws.rowCount >= 3 && ws.getColumn(3).values) {
   407→          const cell = ws.getCell(3, 3)
   408→          const val = parseCellToNumber(cell.value, { font: cell.font } as ExcelCellLike)
   409→          if (val !== 0) {
   410→            receitaBrutaValue = val
   411→          }
   412→        }
   413→        break
   414→      }
   415→    }
   416→
   417→    const worksheet = workbook.worksheets[0]
   418→    if (!worksheet) {
   419→      return NextResponse.json({ error: 'Planilha vazia' }, { status: 400 })
   420→    }
   421→
   422→    // RF-04: Track structural title context during parsing
   423→    let currentSectionContext: string | null = null
   424→    let currentFieldContext: string | null = null
   425→
   426→    const parsedRows: Array<{
   427→      description: string
   428→      date: string
   429→      totalValue: number
   430→      itemMonth: number
   431→      itemYear: number
   432→      dreField: string
   433→    }> = []
   434→
   435→    let structuralTitlesFound = 0
   436→
   437→    worksheet.eachRow((row, rowNumber) => {
   438→      if (rowNumber === 1) return
   439→
   440→      // Column A: Description
   441→      const description = row.getCell(1).value?.toString()?.trim() || ''
   442→      if (!description) return
   443→
   444→      // Column B: Date → parse to get month/year
   445→      const rawDate = row.getCell(2).value
   446→      const parsedDate = parseRowDate(rawDate)
   447→      // If date is parsed, use it; otherwise default to selected month/year
   448→      const itemMonth = parsedDate?.month ?? month
   449→      const itemYear = parsedDate?.year ?? year
   450→      const dateStr = parsedDate?.dateStr ?? ''
   451→
   452→      // Column C: Value
   453→      const valueCell = row.getCell(3)
   454→      const value = parseCellToNumber(valueCell.value, { font: valueCell.font } as ExcelCellLike)
   455→
   456→      // RF-04: Check if this row is a structural title
   457→      const titleInfo = detectStructuralTitle(description)
   458→
   459→      if (titleInfo) {
   460→        structuralTitlesFound++
   461→        if (titleInfo.type === 'section') {
   462→          currentSectionContext = titleInfo.sectionKey ?? null
   463→          currentFieldContext = null
   464→        } else if (titleInfo.type === 'field') {
   465→          currentFieldContext = titleInfo.fieldKey
   466→          currentSectionContext = titleInfo.sectionKey ?? null
   467→        }
   468→        // Structural titles with no value are skipped as data rows
   469→        if (value === 0) return
   470→      }
   471→
   472→      // Column D: Category (explicit auto-classification, takes priority)
   473→      const categoryCell = row.getCell(4)
   474→      let dreField = ''
   475→      if (categoryCell && categoryCell.value) {
   476→        const categoryText = String(categoryCell.value).trim()
   477→        dreField = classifyByCategory(categoryText) || ''
   478→      }
   479→
   480→      // RF-04: If no category, use structural context for auto-classification
   481→      if (!dreField) {
   482→        if (currentFieldContext) {
   483→          dreField = currentFieldContext
   484→        } else if (titleInfo?.type === 'section' && titleInfo.sectionKey) {
   485→          // Section-level title with a value: map to a catch-all field in that section
   486→          // e.g. "Receita Bruta" with value → naoIdentificado
   487→          dreField = sectionTitleToField(titleInfo.sectionKey)
   488→        } else if (currentSectionContext) {
   489→          const autoField = autoClassifyInSection(description, currentSectionContext)
   490→          if (autoField) dreField = autoField
   491→        }
   492→      }
   493→
   494→      if (value !== 0) {
   495→        parsedRows.push({ description, date: dateStr, totalValue: value, itemMonth, itemYear, dreField })
   496→      }
   497→    })
   498→
   499→    if (parsedRows.length === 0 && receitaBrutaValue === 0) {
   500→      return NextResponse.json(
   501→        { error: 'Nenhum dado válido encontrado. Verifique: Descrição (A), Data (B), Valor (C).' },
   502→        { status: 400 }
   503→      )
   504→    }
   505→
   506→    // Create uploaded file record (use month=0, year=0 to indicate multi-month file)
   507→    const uploadedFile = await db.uploadedFile.create({
   508→      data: { companyId, month: 0, year: 0, fileName: file.name },
   509→    })
   510→
   511→    // If "Caixa YYYY" worksheet was found, add Receita Bruta as a qualification item
   512→    // Use the upload year (ano corrente) — NOT the year from the sheet name
   513→    // The auto-update below will inject it into the DRE
   514→    let receitaBrutaInserted = false
   515→    if (receitaBrutaValue !== 0) {
   516→      await db.expenseQualification.create({
   517→        data: {
   518→          companyId,
   519→          month: 1,
   520→          year: year,
   521→          description: 'Receita Bruta',
   522→          date: `${year}-01-01`,
   523→          totalValue: receitaBrutaValue,
   524→          dreField: 'prestacaoNf',
   525→          uploadedFileId: uploadedFile.id,
   526→        },
   527→      })
   528→      receitaBrutaInserted = true
   529→    }
   530→
   531→    // Save all other qualifications
   532→    const qualifications = parsedRows.map((item) => ({
   533→      companyId,
   534→      month: item.itemMonth,
   535→      year: item.itemYear,
   536→      description: item.description,
   537→      date: item.date,
   538→      totalValue: item.totalValue,
   539→      dreField: item.dreField,
   540→      uploadedFileId: uploadedFile.id,
   541→    }))
   542→
   543→    if (qualifications.length > 0) {
   544→      await db.expenseQualification.createMany({ data: qualifications })
   545→    }
   546→
   547→    const savedItems = await db.expenseQualification.findMany({
   548→      where: { uploadedFileId: uploadedFile.id },
   549→      orderBy: { createdAt: 'asc' },
   550→    })
   551→
   552→    // Build distributed months list
   553→    const groupedByMonth = new Map<string, number[]>()
   554→    for (const row of parsedRows) {
   555→      const key = `${row.itemMonth}-${row.itemYear}`
   556→      if (!groupedByMonth.has(key)) groupedByMonth.set(key, [])
   557→      groupedByMonth.get(key)!.push(row.itemMonth)
   558→    }
   559→    const distributedMonths = Array.from(groupedByMonth.keys())
   560→      .map(key => {
   561→        const [m, y] = key.split('-')
   562→        return `${String(parseInt(m)).padStart(2, '0')}/${y}`
   563→      })
   564→      .sort((a, b) => {
   565→        const [am, ay] = a.split('/').map(Number)
   566→        const [bm, by] = b.split('/').map(Number)
   567→        return ay !== by ? ay - by : am - bm
   568→      })
   569→
   570→    // Count auto-classified items
   571→    const autoClassified = savedItems.filter(i => i.dreField).length
   572→
   573→    // Auto-update DRE entries for months that have auto-classified items (especially C3/Receita Bruta)
   574→    const autoClassifiedItems = savedItems.filter(i => i.dreField)
   575→    const autoMonthKeys = new Set(autoClassifiedItems.map(i => `${i.month}-${i.year}`))
   576→    let updatedDREForMonths: Record<string, any> = {}
   577→    for (const monthKey of autoMonthKeys) {
   578→      const [mStr, yStr] = monthKey.split('-')
   579→      const m = parseInt(mStr)
   580→      const y = parseInt(yStr)
   581→      const monthItems = autoClassifiedItems.filter(i => i.month === m && i.year === y)
   582→      
   583→      const fieldTotals: Record<string, number> = {}
   584→      for (const item of monthItems) {
   585→        if (item.dreField && !item.dreField.startsWith('custom_')) {
   586→          fieldTotals[item.dreField] = (fieldTotals[item.dreField] || 0) + item.totalValue
   587→        }
   588→      }
   589→
   590→      if (Object.keys(fieldTotals).length > 0) {
   591→        const existing = await db.dREEntry.findUnique({
   592→          where: { companyId_month_year: { companyId, month: m, year: y } },
   593→        })
   594→        if (existing) {
   595→          // Merge: add to existing values
   596→          const mergedData: Record<string, number> = {}
   597→          for (const [field, total] of Object.entries(fieldTotals)) {
   598→            mergedData[field] = ((existing as any)[field] || 0) + total
   599→          }
   600→          await db.dREEntry.update({
   601→            where: { id: existing.id },
   602→            data: mergedData,
   603→          })
   604→          // Fetch updated entry for response
   605→          const updated = await db.dREEntry.findUnique({ where: { id: existing.id } })
   606→          if (updated) {
   607→            const { id: _id, companyId: _c, month: _m2, year: _y2, createdAt, updatedAt, company, ...fields } = updated as any
   608→            updatedDREForMonths[monthKey] = fields
   609→          }
   610→        } else {
   611→          const created = await db.dREEntry.create({
   612→            data: { companyId, month: m, year: y, ...fieldTotals },
   613→          })
   614→          const { id: _id, companyId: _c, month: _m2, year: _y2, createdAt, updatedAt, company, ...fields } = created as any
   615→          updatedDREForMonths[monthKey] = fields
   616→        }
   617→      }
   618→    }
   619→
   620→    // Exclude the auto-generated "Receita Bruta" C3 item from the items list
   621→    // since it's already applied directly to the DRE as prestacaoNf
   622→    const receitaBrutaItemId = receitaBrutaInserted
   623→      ? savedItems.find(i => i.description === 'Receita Bruta' && i.dreField === 'prestacaoNf' && i.month === 1)?.id
   624→      : null
   625→
   626→    return NextResponse.json({
   627→      items: savedItems
   628→        .filter(i => i.id !== receitaBrutaItemId)
   629→        .map(i => ({
   630→          id: i.id, description: i.description, date: i.date,
   631→          totalValue: i.totalValue, dreField: i.dreField,
   632→          month: i.month, year: i.year,
   633→        })),
   634→      file: { id: uploadedFile.id, fileName: uploadedFile.fileName, uploadedAt: uploadedFile.uploadedAt },
   635→      distributedMonths,
   636→      structuralTitlesFound,
   637→      autoClassified,
   638→      updatedDREForMonths,
   639→      receitaBrutaInserted,
   640→      receitaBrutaValue,
   641→      receitaBrutaYear,
   642→    })
   643→  } catch (error) {
   644→    console.error('Expense qualification error:', error)
   645→    return NextResponse.json({ error: 'Erro ao processar planilha de despesas' }, { status: 500 })
   646→  }
   647→}

   649→// ─── PUT: Save expense classification and update DRE (RF-07: returns updated DRE data) ───
   650→
   651→export async function PUT(req: NextRequest) {
   652→  try {
   653→    const session = await getServerSession(authOptions)
   654→    if (!session?.user?.id) {
   655→      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
   656→    }
   657→
   658→    const userId = (session.user as any).id
   659→    const body = await req.json()
   660→    const { companyId, month, year, items } = body
   661→
   662→    if (!companyId || !month || !year || !items) {
   663→      return NextResponse.json({ error: 'Parâmetros incompletos' }, { status: 400 })
   664→    }
   665→
   666→    const hasAccess = await verifyAccess(userId, companyId)
   667→    if (!hasAccess) {
   668→      return NextResponse.json({ error: 'Sem permissão' }, { status: 403 })
   669→    }
   670→
   671→    // Update each item's dreField mapping in parallel
   672→    const itemsToUpdate = items.filter((item: any) => item.id && item.dreField)
   673→    await Promise.all(
   674→      itemsToUpdate.map((item: any) =>
   675→        db.expenseQualification.update({
   676→          where: { id: item.id },
   677→          data: { dreField: item.dreField },
   678→        })
   679→      )
   680→    )
   681→
   682→    // Find all distinct month/year combos that have classified items for this company
   683→    const allClassified = await db.expenseQualification.findMany({
   684→      where: { companyId, dreField: { not: '' } },
   685→      select: { month: true, year: true },
   686→      distinct: ['month', 'year'],
   687→    })
   688→
   689→    // Update DRE entries for ALL months that have classified items
   690→    let currentMonthDreEntryId: string | null = null
   691→    let currentMonthFieldTotals: Record<string, number> = {}
   692→    for (const { month: m, year: y } of allClassified) {
   693→      const monthQualifications = await db.expenseQualification.findMany({
   694→        where: { companyId, month: m, year: y, dreField: { not: '' } },
   695→      })
   696→
   697→      const fieldTotals: Record<string, number> = {}
   698→      const customFieldTotals: Record<string, number> = {}
   699→      for (const q of monthQualifications) {
   700→        if (q.dreField) {
   701→          if (q.dreField.startsWith('custom_')) {
   702→            const customFieldId = q.dreField.replace('custom_', '')
   703→            customFieldTotals[customFieldId] = (customFieldTotals[customFieldId] || 0) + q.totalValue
   704→          } else {
   705→            fieldTotals[q.dreField] = (fieldTotals[q.dreField] || 0) + q.totalValue
   706→          }
   707→        }
   708→      }
   709→
   710→      // Save field totals for the currently selected month
   711→      if (m === month && y === year) {
   712→        currentMonthFieldTotals = fieldTotals
   713→      }
   714→
   715→      const existing = await db.dREEntry.findUnique({
   716→        where: { companyId_month_year: { companyId, month: m, year: y } },
   717→      })
   718→
   719→      const updateData: Record<string, number> = {}
   720→      for (const [field, total] of Object.entries(fieldTotals)) {
   721→        updateData[field] = total
   722→      }
   723→
   724→      let dreEntryId: string | null = null
   725→      if (Object.keys(updateData).length > 0 || Object.keys(customFieldTotals).length > 0) {
   726→        if (existing) {
   727→          dreEntryId = existing.id
   728→          if (Object.keys(updateData).length > 0) {
   729→            await db.dREEntry.update({
   730→              where: { id: existing.id },
   731→              data: updateData,
   732→            })
   733→          }
   734→        } else {
   735→          const created = await db.dREEntry.create({
   736→            data: { companyId, month: m, year: y, ...updateData },
   737→          })
   738→          dreEntryId = created.id
   739→        }
   740→      } else {
   741→        dreEntryId = existing?.id || null
   742→      }
   743→
   744→      // Handle custom field totals (with sign correction)
   745→      if (dreEntryId && Object.keys(customFieldTotals).length > 0) {
   746→        const customFieldIds = Object.keys(customFieldTotals)
   747→        const customFields = await db.customDREField.findMany({
   748→          where: { id: { in: customFieldIds } },
   749→        })
   750→        const fieldSignMap: Record<string, string> = {}
   751→        for (const cf of customFields) fieldSignMap[cf.id] = cf.sign
   752→
   753→        await Promise.all(
   754→          Object.entries(customFieldTotals).map(([customFieldId, total]) => {
   755→            const sign = fieldSignMap[customFieldId]
   756→            const adjustedTotal = sign === '-' ? -total : total
   757→            return db.customDREValue.upsert({
   758→              where: { customFieldId_dREEntryId: { customFieldId, dREEntryId } },
   759→              create: { customFieldId, dREEntryId, value: adjustedTotal },
   760→              update: { value: adjustedTotal },
   761→            })
   762→          })
   763→        )
   764→      }
   765→
   766→      // Track the entry ID for the currently selected month
   767→      if (m === month && y === year) {
   768→        currentMonthDreEntryId = dreEntryId
   769→      }
   770→    }
   771→
   772→    // RF-07: Fetch the updated DRE entry to return for reactive sync
   773→    let updatedDREData: Record<string, any> | null = null
   774→    if (currentMonthDreEntryId) {
   775→      const updatedEntry = await db.dREEntry.findUnique({ where: { id: currentMonthDreEntryId } })
   776→      if (updatedEntry) {
   777→        const { id, companyId: _cid, month: _m, year: _y, createdAt, updatedAt, company, ...fields } = updatedEntry as any
   778→        updatedDREData = fields
   779→
   780→        // Also fetch custom field values
   781→        const customValues = await db.customDREValue.findMany({
   782→          where: { dREEntryId: currentMonthDreEntryId },
   783→          include: { customField: true },
   784→        })
   785→        if (customValues.length > 0) {
   786→          updatedDREData._customFieldValues = {}
   787→          for (const cv of customValues) {
   788→            const sign = (cv.customField as any).sign || '+'
   789→            updatedDREData._customFieldValues[cv.customFieldId] = sign === '-' ? -cv.value : cv.value
   790→          }
   791→        }
   792→      }
   793→    }
   794→
   795→    return NextResponse.json({
   796→      success: true,
   797→      fieldTotals: currentMonthFieldTotals,
   798→      dreEntryId: currentMonthDreEntryId,
   799→      updatedDREData, // RF-07: reactive sync payload
   800→    })
   801→  } catch (error) {
   802→    console.error('Error updating qualification:', error)
   803→    return NextResponse.json({ error: 'Erro ao atualizar qualificação' }, { status: 500 })
   804→  }
   805→}

   807→// ─── DELETE: Remove uploaded files ───
   808→
   809→export async function DELETE(req: NextRequest) {
   810→  try {
   811→    const session = await getServerSession(authOptions)
   812→    if (!session?.user?.id) {
   813→      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
   814→    }
   815→
   816→    const userId = (session.user as any).id
   817→    const { searchParams } = new URL(req.url)
   818→    const fileIdsParam = searchParams.get('fileIds')
   819→
   820→    if (!fileIdsParam) {
   821→      return NextResponse.json({ error: 'fileIds obrigatório' }, { status: 400 })
   822→    }
   823→
   824→    const ids = fileIdsParam.split(',').filter(Boolean)
   825→    if (ids.length === 0) {
   826→      return NextResponse.json({ error: 'Nenhum ID válido' }, { status: 400 })
   827→    }
   828→
   829→    const files = await db.uploadedFile.findMany({ where: { id: { in: ids } } })
   830→    const uniqueCompanyIds = [...new Set(files.map(f => f.companyId))]
   831→    const accessChecks = await Promise.all(
   832→      uniqueCompanyIds.map(cid => verifyAccess(userId, cid))
   833→    )
   834→    const accessMap = new Map(uniqueCompanyIds.map((cid, i) => [cid, accessChecks[i]]))
   835→
   836→    for (const file of files) {
   837→      if (!accessMap.get(file.companyId)) {
   838→        return NextResponse.json({ error: 'Sem permissão para excluir arquivo' }, { status: 403 })
   839→      }
   840→    }
   841→
   842→    const deleteResult = await db.expenseQualification.deleteMany({
   843→      where: { uploadedFileId: { in: ids } },
   844→    })
   845→
   846→    const fileDeleteResult = await db.uploadedFile.deleteMany({
   847→      where: { id: { in: ids } },
   848→    })
   849→
   850→    return NextResponse.json({ success: true, deletedFiles: fileDeleteResult.count, deletedItems: deleteResult.count })
   851→  } catch (error) {
   852→    console.error('Error deleting uploaded files:', error)
   853→    return NextResponse.json({ error: 'Erro ao excluir arquivos' }, { status: 500 })
   854→  }
   855→}
