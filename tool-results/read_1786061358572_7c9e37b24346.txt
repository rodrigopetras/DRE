   535→  // ─── CNPJ mask ───
   536→  const formatCnpj = (value: string) => {
   537→    const digits = value.replace(/\D/g, '').slice(0, 14)
   538→    if (digits.length <= 2) return digits
   539→    if (digits.length <= 5) return `${digits.slice(0, 2)}.${digits.slice(2)}`
   540→    if (digits.length <= 8) return `${digits.slice(0, 2)}.${digits.slice(2, 5)}.${digits.slice(5)}`
   541→    if (digits.length <= 12) return `${digits.slice(0, 2)}.${digits.slice(2, 5)}.${digits.slice(5, 8)}/${digits.slice(8)}`
