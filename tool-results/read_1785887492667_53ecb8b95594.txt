    83→function AppContent() {
    84→  const { data: session, status } = useSession()
    85→  const userId = (session?.user as any)?.id
    86→
    87→  if (status === 'loading') {
    88→    return (
    89→      <div className="min-h-screen flex items-center justify-center bg-background">
    90→        <Loader2 className="h-8 w-8 animate-spin text-muted-foreground" />
    91→      </div>
    92→    )
    93→  }
    94→
    95→  if (!session) {
    96→    return <LoginScreen />
    97→  }
    98→
    99→  return <DREApp session={session} userId={userId} />
   100→}
