    22→    22→interface ExcelCellLike {
    23→    23→  value: unknown
    24→    24→  font?: { color?: { argb?: string } }
    25→    25→}
    26→    26→
    27→    27→/**
    28→    28→ * Detect if an Excel cell's font color is red.
    29→    29→ * Red = alpha byte 0xFF (high byte), R > 150, G < 100, B < 100.
    30→    30→ */
    31→    31→function isRedFont(cell?: ExcelCellLike): boolean {
    32→    32→  if (!cell?.font?.color?.argb) return false
    33→    33→  const argb = cell.font.color.argb
    34→    34→  if (argb.length < 8) return false
    35→    35→  const alpha = parseInt(argb.substring(0, 2), 16)
    36→    36→  if (alpha !== 0xFF) return false
    37→    37→  const r = parseInt(argb.substring(2, 4), 16)
    38→    38→  const g = parseInt(argb.substring(4, 6), 16)
    39→    39→  const b = parseInt(argb.substring(6, 8), 16)
    40→    40→  return r > 150 && g < 100 && b < 100
    41→    41→}
    42→    42→
    43→    43→/**
    44→    44→ * Parse any ExcelJS cell value into a number.
    45→    45→ * Handles: number, formula, RichText, Hyperlink, string, Date.
    46→    46→ * Parenthesized values like (780,00) are converted to negative.
    47→    47→ * Red font color also makes the value negative.
    48→    48→ */
    49→    49→function parseCellToNumber(cellValue: unknown, cell?: ExcelCellLike): number {
    50→    50→  let result = 0
    51→    51→
    52→    52→  if (typeof cellValue === 'number') {
    53→    53→    result = cellValue
    54→    54→  }
    55→    55→  else if (cellValue && typeof cellValue === 'object' && 'result' in cellValue) {
    56→    56→    const formulaResult = (cellValue as any).result
    57→    57→    if (typeof formulaResult === 'number') result = formulaResult
    58→    58→    else if (typeof formulaResult === 'string') result = parseTextToNumber(formulaResult)
    59→    59→  }
    60→    60→  else if (cellValue && typeof cellValue === 'object' && 'richText' in cellValue) {
    61→    61→    const parts = (cellValue as any).richText as Array<{ text: string }>
    62→    62→    const text = parts.map(p => p.text).join('')
    63→    63→    result = parseTextToNumber(text)
    64→    64→  }
    65→    65→  else if (cellValue && typeof cellValue === 'object' && 'text' in cellValue && 'hyperlink' in cellValue) {
    66→    66→    result = parseTextToNumber((cellValue as any).text)
    67→    67→  }
    68→    68→  else if (typeof cellValue === 'string') {
    69→    69→    result = parseTextToNumber(cellValue)
    70→    70→  }
    71→    71→
    72→    72→  if (result > 0 && isRedFont(cell)) {
    73→    73→    result = -result
    74→    74→  }
    75→    75→
    76→    76→  return result
    77→    77→}

    79→    79→/**
    80→    80→ * Parse a text string like "1.234,56" or "(780,00)" into a number.
    81→    81→ * Parenthesized values become negative.
    82→    82→ */
    83→    83→function parseTextToNumber(text: string): number {
    84→    84→  const trimmed = text.trim()
    85→    85→  if (!trimmed) return 0
    86→    86→  const parenMatch = trimmed.match(/^\((.+)\)$/)
    87→    87→  let numStr: string
    88→    88→  let sign = 1
    89→    89→  if (parenMatch) {
    90→    90→    numStr = parenMatch[1]
    91→    91→    sign = -1
    92→    92→  } else {
    93→    93→    numStr = trimmed
    94→    94→    if (numStr.startsWith('-')) {
    95→    95→      sign = -1
    96→    96→      numStr = numStr.substring(1)
    97→    97→    }
    98→    98→  }
    99→    99→  const cleaned = numStr.replace(/\./g, '').replace(',', '.')
   100→   100→  const parsed = parseFloat(cleaned)
   101→   101→  if (isNaN(parsed)) return 0
   102→   102→  return sign * Math.abs(parsed)
   103→   103→}

   107→   107→function excelSerialToDate(serial: number): Date {
   108→   108→  if (serial < 1) return new Date(NaN)
   109→   109→  const epoch = new Date(1899, 11, 30)
   110→   110→  const msPerDay = 24 * 60 * 60 * 1000
   111→   111→  return new Date(epoch.getTime() + serial * msPerDay)
   112→   112→}

   115→   115→function normalizeAccents(str: string): string {
   116→   116→  return str.normalize('NFD').replace(/[\u0300-\u036f]/g, '').toLowerCase().trim()

   129→   129→const MONTH_NAMES_PT: Record<string, number> = {

   136→   136→function parseRowDate(rawValue: unknown): ParsedDateInfo | null {
   137→   137→  if (!rawValue) return null
   138→   138→
   139→   139→  if (rawValue instanceof Date && !isNaN(rawValue.getTime())) {
   140→   140→    return {
   141→   141→      month: rawValue.getMonth() + 1,
   142→   142→      year: rawValue.getFullYear(),
   143→   143→      dateStr: rawValue.toISOString().split('T')[0],
   144→   144→    }
   145→   145→  }
   146→   146→
   147→   147→  if (typeof rawValue === 'number' && rawValue > 25569 && rawValue < 100000) {
   148→   148→    const d = excelSerialToDate(rawValue)
   149→   149→    if (!isNaN(d.getTime())) {
   150→   150→      return {
   151→   151→        month: d.getMonth() + 1,
   152→   152→        year: d.getFullYear(),
   153→   153→        dateStr: d.toISOString().split('T')[0],
   154→   154→      }
   155→   155→    }
   156→   156→  }
   157→   157→
   158→   158→  if (typeof rawValue === 'string') {
   159→   159→    const s = rawValue.trim()
   160→   160→    if (!s) return null
   161→   161→
   162→   162→    // YYYY-MM-DD
   163→   163→    let m = s.match(/^(\d{4})-(\d{1,2})-(\d{1,2})/)
   164→   164→    if (m) {
   165→   165→      const yr = parseInt(m[1]); const mo = parseInt(m[2]); const dy = parseInt(m[3])
   166→   166→      if (mo >= 1 && mo <= 12 && dy >= 1 && dy <= 31) {
   167→   167→        return { month: mo, year: yr, dateStr: `${yr}-${String(mo).padStart(2, '0')}-${String(dy).padStart(2, '0')}` }
   168→   168→      }
   169→   169→    }
   170→   170→
   171→   171→    // DD/MM/YYYY or MM/DD/YYYY
   172→   172→    m = s.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/)
   173→   173→    if (m) {
   174→   174→      const a = parseInt(m[1]); const b = parseInt(m[2]); const yr = parseInt(m[3])
   175→   175→      let mo: number, dy: number
   176→   176→      if (a > 12) { dy = a; mo = b }
   177→   177→      else if (b > 12) { mo = a; dy = b }
   178→   178→      else { dy = a; mo = b }
   179→   179→      if (mo >= 1 && mo <= 12 && dy >= 1 && dy <= 31) {
   180→   180→        return { month: mo, year: yr, dateStr: `${yr}-${String(mo).padStart(2, '0')}-${String(dy).padStart(2, '0')}` }
   181→   181→      }
   182→   182→    }
   183→   183→
   184→   184→    // DD-MM-YYYY
   185→   185→    m = s.match(/^(\d{1,2})-(\d{1,2})-(\d{4})$/)
   186→   186→    if (m) {
   187→   187→      const a = parseInt(m[1]); const b = parseInt(m[2]); const yr = parseInt(m[3])
   188→   188→      let mo: number, dy: number
   189→   189→      if (a > 12) { dy = a; mo = b }
   190→   190→      else if (b > 12) { mo = a; dy = b }
   191→   191→      else { dy = a; mo = b }
   192→   192→      if (mo >= 1 && mo <= 12 && dy >= 1 && dy <= 31) {
   193→   193→        return { month: mo, year: yr, dateStr: `${yr}-${String(mo).padStart(2, '0')}-${String(dy).padStart(2, '0')}` }
   194→   194→      }
   195→   195→    }
   196→   196→
   197→   197→    // DD/MM/YY
   198→   198→    m = s.match(/^(\d{1,2})\/(\d{1,2})\/(\d{2})$/)
   199→   199→    if (m) {
   200→   200→      const a = parseInt(m[1]); const b = parseInt(m[2])
   201→   201→      let yr = parseInt(m[3])
   202→   202→      yr = yr < 100 ? 2000 + yr : yr
   203→   203→      let mo: number, dy: number
   204→   204→      if (a > 12) { dy = a; mo = b }
   205→   205→      else if (b > 12) { mo = a; dy = b }
   206→   206→      else { dy = a; mo = b }
   207→   207→      if (mo >= 1 && mo <= 12 && dy >= 1 && dy <= 31) {
   208→   208→        return { month: mo, year: yr, dateStr: `${yr}-${String(mo).padStart(2, '0')}-${String(dy).padStart(2, '0')}` }
   209→   209→      }
   210→   210→    }
   211→   211→
   212→   212→    // Month name in Portuguese (e.g. "Janeiro 2025", "Jan/2025", "janeiro")
   213→   213→    const normS = normalizeAccents(s)
   214→   214→    for (const [monthName, monthNum] of Object.entries(MONTH_NAMES_PT)) {
   215→   215→      const normName = normalizeAccents(monthName)
   216→   216→      const regex = new RegExp(`\\b${normName}\\b`, 'i')
   217→   217→      if (regex.test(normS)) {
   218→   218→        // Try to extract year from the string
   219→   219→        const yearMatch = s.match(/(\d{4})/)
   220→   220→        const yr = yearMatch ? parseInt(yearMatch[1]) : new Date().getFullYear()
   221→   221→        return { month: monthNum, year: yr, dateStr: `${yr}-${String(monthNum).padStart(2, '0')}-01` }
   222→   222→      }
   223→   223→    }
   224→   224→
   225→   225→    // MM/YYYY (just month and year)
   226→   226→    m = s.match(/^(\d{1,2})\/(\d{4})$/)
   227→   227→    if (m) {
   228→   228→      const mo = parseInt(m[1]); const yr = parseInt(m[2])
   229→   229→      if (mo >= 1 && mo <= 12) {
   230→   230→        return { month: mo, year: yr, dateStr: `${yr}-${String(mo).padStart(2, '0')}-01` }
   231→   231→      }
   232→   232→    }
   233→   233→  }
   234→   234→
   235→   235→  return null
   236→   236→}

   277→   277→/**
   278→   278→ * RF-04: Detect if a description is a structural title and return its context.
   279→   279→ * Returns { type: 'section' | 'field', fieldKey?, sectionKey? } or null.
   280→   280→ */
   281→   281→function detectStructuralTitle(description: string): { type: 'section' | 'field'; fieldKey?: string; sectionKey?: string } | null {
   282→   282→  const norm = normalizeAccents(description)
   283→   283→
   284→   284→  // Check for section-level matches first
   285→   285→  for (const entry of STRUCTURAL_TITLES) {
   286→   286→    for (const pattern of entry.patterns) {
   287→   287→      if (norm === normalizeAccents(pattern) || norm.includes(normalizeAccents(pattern))) {
   288→   288→        if (entry.sectionKey && !entry.fieldKey) {
   289→   289→          return { type: 'section', sectionKey: entry.sectionKey }
   290→   290→        }
   291→   291→        if (entry.fieldKey) {
   292→   292→          return { type: 'field', fieldKey: entry.fieldKey, sectionKey: DRE_FIELD_SECTIONS[entry.fieldKey] }
   293→   293→        }
   294→   294→      }
   295→   295→    }
   296→   296→  }
   297→   297→  return null
   298→   298→}

   300→   300→/**
   301→   301→ * When a structural section title row has a value, map it to a reasonable DRE field.
   302→   302→ * For receita section: maps to 'naoIdentificado' (catch-all revenue field).
   303→   303→ * For desp sections: maps to the first field in that section.
   304→   304→ */
   305→   305→function sectionTitleToField(sectionKey: string): string {
   306→   306→  if (sectionKey === 'receita') return 'naoIdentificado'
   307→   307→  // For expense sections, find the first field in that section
   308→   308→  const sectionFields = Object.entries(DRE_FIELD_SECTIONS)
   309→   309→    .filter(([, section]) => section === sectionKey)
   310→   310→    .map(([key]) => key)
   311→   311→  return sectionFields[0] || 'naoIdentificado'
   312→   312→}

   314→   314→/**
   315→   315→ * RF-04: Given a section context, try to auto-classify a description into the first matching field in that section.
   316→   316→ */
   317→   317→function autoClassifyInSection(description: string, sectionKey: string): string | null {
   318→   318→  const norm = normalizeAccents(description)
   319→   319→  const sectionFields = Object.entries(DRE_FIELD_SECTIONS)
   320→   320→    .filter(([, section]) => section === sectionKey)
   321→   321→    .map(([key]) => key)
   322→   322→
   323→   323→  for (const fieldKey of sectionFields) {
   324→   324→    const label = DRE_FIELD_LABELS[fieldKey]
   325→   325→    if (!label) continue
   326→   326→    const normLabel = normalizeAccents(label)
   327→   327→    // Check if description contains field label keywords
   328→   328→    const labelWords = normLabel.split(/\s+/).filter(w => w.length > 2)
   329→   329→    for (const word of labelWords) {
   330→   330→      if (norm.includes(word)) return fieldKey
   331→   331→    }
   332→   332→  }
   333→   333→  return null
   334→   334→}

   336→   336→/**
   337→   337→ * Try to match a category string from column D against DRE field labels.
   338→   338→ */
   339→   339→function classifyByCategory(category: string): string | null {
   340→   340→  if (!category || !category.trim()) return null
   341→   341→  const norm = normalizeAccents(category)
   342→   342→
   343→   343→  for (const key of Object.keys(DRE_FIELD_LABELS)) {
   344→   344→    if (normalizeAccents(key) === norm) return key
   345→   345→  }
   346→   346→  for (const [key, value] of Object.entries(DRE_FIELD_LABELS)) {
   347→   347→    if (normalizeAccents(value) === norm) return key
   348→   348→  }
   349→   349→  return null
   350→   350→}

   354→   354→export async function POST(req: NextRequest) {
   355→   355→  try {
   356→   356→    const session = await getServerSession(authOptions)
   357→   357→    if (!session?.user?.id) {
   358→   358→      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
   359→   359→    }
   360→   360→
   361→   361→    const userId = (session.user as any).id
   362→   362→    const formData = await req.formData()
   363→   363→    const file = formData.get('file') as File
   364→   364→    const companyId = formData.get('companyId') as string
   365→   365→    const month = parseInt(formData.get('month') as string)
   366→   366→    const year = parseInt(formData.get('year') as string)
   367→   367→
   368→   368→    if (!file || !companyId || !month || !year) {
   369→   369→      return NextResponse.json({ error: 'Parâmetros incompletos' }, { status: 400 })
   370→   370→    }
   371→   371→
   372→   372→    const hasAccess = await verifyAccess(userId, companyId)
   373→   373→    if (!hasAccess) {
   374→   374→      return NextResponse.json({ error: 'Sem permissão' }, { status: 403 })
   375→   375→    }
   376→   376→
   377→   377→    const buffer = Buffer.from(await file.arrayBuffer())
   378→   378→    const workbook = new ExcelJS.Workbook()
   379→   379→    await workbook.xlsx.load(buffer)
   380→   380→
   381→   381→    const worksheet = workbook.worksheets[0]
   382→   382→    if (!worksheet) {
   383→   383→      return NextResponse.json({ error: 'Planilha vazia' }, { status: 400 })
   384→   384→    }
   385→   385→
   386→   386→    // RF-04: Track structural title context during parsing
   387→   387→    let currentSectionContext: string | null = null
   388→   388→    let currentFieldContext: string | null = null
   389→   389→
   390→   390→    const parsedRows: Array<{
   391→   391→      description: string
   392→   392→      date: string
   393→   393→      totalValue: number
   394→   394→      itemMonth: number
   395→   395→      itemYear: number
   396→   396→      dreField: string
   397→   397→    }> = []
   398→   398→
   399→   399→    let structuralTitlesFound = 0
   400→   400→
   401→   401→    worksheet.eachRow((row, rowNumber) => {
   402→   402→      if (rowNumber === 1) return
   403→   403→
   404→   404→      // Column A: Description
   405→   405→      const description = row.getCell(1).value?.toString()?.trim() || ''
   406→   406→      if (!description) return
   407→   407→
   408→   408→      // Column B: Date → parse to get month/year
   409→   409→      const rawDate = row.getCell(2).value
   410→   410→      const parsedDate = parseRowDate(rawDate)
   411→   411→      // If date is parsed, use it; otherwise default to selected month/year
   412→   412→      const itemMonth = parsedDate?.month ?? month
   413→   413→      const itemYear = parsedDate?.year ?? year
   414→   414→      const dateStr = parsedDate?.dateStr ?? ''
   415→   415→
   416→   416→      // Column C: Value
   417→   417→      const valueCell = row.getCell(3)
   418→   418→      const value = parseCellToNumber(valueCell.value, { font: valueCell.font } as ExcelCellLike)
   419→   419→
   420→   420→      // RF-04: Check if this row is a structural title
   421→   421→      const titleInfo = detectStructuralTitle(description)
   422→   422→
   423→   423→      if (titleInfo) {
   424→   424→        structuralTitlesFound++
   425→   425→        if (titleInfo.type === 'section') {
   426→   426→          currentSectionContext = titleInfo.sectionKey ?? null
   427→   427→          currentFieldContext = null
   428→   428→        } else if (titleInfo.type === 'field') {
   429→   429→          currentFieldContext = titleInfo.fieldKey
   430→   430→          currentSectionContext = titleInfo.sectionKey ?? null
   431→   431→        }
   432→   432→        // Structural titles with no value are skipped as data rows
   433→   433→        if (value === 0) return
   434→   434→      }
   435→   435→
   436→   436→      // Column D: Category (explicit auto-classification, takes priority)
   437→   437→      const categoryCell = row.getCell(4)
   438→   438→      let dreField = ''
   439→   439→      if (categoryCell && categoryCell.value) {
   440→   440→        const categoryText = String(categoryCell.value).trim()
   441→   441→        dreField = classifyByCategory(categoryText) || ''
   442→   442→      }
   443→   443→
   444→   444→      // RF-04: If no category, use structural context for auto-classification
   445→   445→      if (!dreField) {
   446→   446→        if (currentFieldContext) {
   447→   447→          dreField = currentFieldContext
   448→   448→        } else if (titleInfo?.type === 'section' && titleInfo.sectionKey) {
   449→   449→          // Section-level title with a value: map to a catch-all field in that section
   450→   450→          // e.g. "Receita Bruta" with value → naoIdentificado
   451→   451→          dreField = sectionTitleToField(titleInfo.sectionKey)
   452→   452→        } else if (currentSectionContext) {
   453→   453→          const autoField = autoClassifyInSection(description, currentSectionContext)
   454→   454→          if (autoField) dreField = autoField
   455→   455→        }
   456→   456→      }
   457→   457→
   458→   458→      if (value !== 0) {
   459→   459→        parsedRows.push({ description, date: dateStr, totalValue: value, itemMonth, itemYear, dreField })
   460→   460→      }
   461→   461→    })
   462→   462→
   463→   463→    if (parsedRows.length === 0) {
   464→   464→      return NextResponse.json(
   465→   465→        { error: 'Nenhum dado válido encontrado. Verifique: Descrição (A), Data (B), Valor (C).' },
   466→   466→        { status: 400 }
   467→   467→      )
   468→   468→    }
   469→   469→
   470→   470→    // Create uploaded file record
   471→   471→    const uploadedFile = await db.uploadedFile.create({
   472→   472→      data: { companyId, month, year, fileName: file.name },
   473→   473→    })
   474→   474→
   475→   475→    // Save all qualifications
   476→   476→    const qualifications = parsedRows.map((item) => ({
   477→   477→      companyId,
   478→   478→      month: item.itemMonth,
   479→   479→      year: item.itemYear,
   480→   480→      description: item.description,
   481→   481→      date: item.date,
   482→   482→      totalValue: item.totalValue,
   483→   483→      dreField: item.dreField,
   484→   484→      uploadedFileId: uploadedFile.id,
   485→   485→    }))
   486→   486→
   487→   487→    await db.expenseQualification.createMany({ data: qualifications })
   488→   488→
   489→   489→    const savedItems = await db.expenseQualification.findMany({
   490→   490→      where: { uploadedFileId: uploadedFile.id },
   491→   491→      orderBy: { createdAt: 'asc' },
   492→   492→    })
   493→   493→
   494→   494→    // Build distributed months list
   495→   495→    const groupedByMonth = new Map<string, number[]>()
   496→   496→    for (const row of parsedRows) {
   497→   497→      const key = `${row.itemMonth}-${row.itemYear}`
   498→   498→      if (!groupedByMonth.has(key)) groupedByMonth.set(key, [])
   499→   499→      groupedByMonth.get(key)!.push(row.itemMonth)
   500→   500→    }
   501→   501→    const distributedMonths = Array.from(groupedByMonth.keys())
   502→   502→      .map(key => {
   503→   503→        const [m, y] = key.split('-')
   504→   504→        return `${String(parseInt(m)).padStart(2, '0')}/${y}`
   505→   505→      })
   506→   506→      .sort((a, b) => {
   507→   507→        const [am, ay] = a.split('/').map(Number)
   508→   508→        const [bm, by] = b.split('/').map(Number)
   509→   509→        return ay !== by ? ay - by : am - bm
   510→   510→      })
   511→   511→
   512→   512→    // Count auto-classified items
   513→   513→    const autoClassified = savedItems.filter(i => i.dreField).length
   514→   514→
   515→   515→    return NextResponse.json({
   516→   516→      items: savedItems.map(i => ({
   517→   517→        id: i.id, description: i.description, date: i.date,
   518→   518→        totalValue: i.totalValue, dreField: i.dreField,
   519→   519→        month: i.month, year: i.year,
   520→   520→      })),
   521→   521→      file: { id: uploadedFile.id, fileName: uploadedFile.fileName, uploadedAt: uploadedFile.uploadedAt },
   522→   522→      distributedMonths,
   523→   523→      structuralTitlesFound,
   524→   524→      autoClassified,
   525→   525→    })
   526→   526→  } catch (error) {
   527→   527→    console.error('Expense qualification error:', error)
   528→   528→    return NextResponse.json({ error: 'Erro ao processar planilha de despesas' }, { status: 500 })
   529→   529→  }
   530→   530→}

   532→   532→// ─── PUT: Save expense classification and update DRE (RF-07: returns updated DRE data) ───
   533→   533→
   534→   534→export async function PUT(req: NextRequest) {
   535→   535→  try {
   536→   536→    const session = await getServerSession(authOptions)
   537→   537→    if (!session?.user?.id) {
   538→   538→      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
   539→   539→    }
   540→   540→
   541→   541→    const userId = (session.user as any).id
   542→   542→    const body = await req.json()
   543→   543→    const { companyId, month, year, items } = body
   544→   544→
   545→   545→    if (!companyId || !month || !year || !items) {
   546→   546→      return NextResponse.json({ error: 'Parâmetros incompletos' }, { status: 400 })
   547→   547→    }
   548→   548→
   549→   549→    const hasAccess = await verifyAccess(userId, companyId)
   550→   550→    if (!hasAccess) {
   551→   551→      return NextResponse.json({ error: 'Sem permissão' }, { status: 403 })
   552→   552→    }
   553→   553→
   554→   554→    // Update each item's dreField mapping in parallel
   555→   555→    const itemsToUpdate = items.filter((item: any) => item.id && item.dreField)
   556→   556→    await Promise.all(
   557→   557→      itemsToUpdate.map((item: any) =>
   558→   558→        db.expenseQualification.update({
   559→   559→          where: { id: item.id },
   560→   560→          data: { dreField: item.dreField },
   561→   561→        })
   562→   562→      )
   563→   563→    )
   564→   564→
   565→   565→    // Find all distinct month/year combos that have classified items for this company
   566→   566→    const allClassified = await db.expenseQualification.findMany({
   567→   567→      where: { companyId, dreField: { not: '' } },
   568→   568→      select: { month: true, year: true },
   569→   569→      distinct: ['month', 'year'],
   570→   570→    })
   571→   571→
   572→   572→    // Update DRE entries for ALL months that have classified items
   573→   573→    let currentMonthDreEntryId: string | null = null
   574→   574→    let currentMonthFieldTotals: Record<string, number> = {}
   575→   575→    for (const { month: m, year: y } of allClassified) {
   576→   576→      const monthQualifications = await db.expenseQualification.findMany({
   577→   577→        where: { companyId, month: m, year: y, dreField: { not: '' } },
   578→   578→      })
   579→   579→
   580→   580→      const fieldTotals: Record<string, number> = {}
   581→   581→      const customFieldTotals: Record<string, number> = {}
   582→   582→      for (const q of monthQualifications) {
   583→   583→        if (q.dreField) {
   584→   584→          if (q.dreField.startsWith('custom_')) {
   585→   585→            const customFieldId = q.dreField.replace('custom_', '')
   586→   586→            customFieldTotals[customFieldId] = (customFieldTotals[customFieldId] || 0) + q.totalValue
   587→   587→          } else {
   588→   588→            fieldTotals[q.dreField] = (fieldTotals[q.dreField] || 0) + q.totalValue
   589→   589→          }
   590→   590→        }
   591→   591→      }
   592→   592→
   593→   593→      // Save field totals for the currently selected month
   594→   594→      if (m === month && y === year) {
   595→   595→        currentMonthFieldTotals = fieldTotals
   596→   596→      }
   597→   597→
   598→   598→      const existing = await db.dREEntry.findUnique({
   599→   599→        where: { companyId_month_year: { companyId, month: m, year: y } },
   600→   600→      })
   601→   601→
   602→   602→      const updateData: Record<string, number> = {}
   603→   603→      for (const [field, total] of Object.entries(fieldTotals)) {
   604→   604→        updateData[field] = total
   605→   605→      }
   606→   606→
   607→   607→      let dreEntryId: string | null = null
   608→   608→      if (Object.keys(updateData).length > 0 || Object.keys(customFieldTotals).length > 0) {
   609→   609→        if (existing) {
   610→   610→          dreEntryId = existing.id
   611→   611→          if (Object.keys(updateData).length > 0) {
   612→   612→            await db.dREEntry.update({
   613→   613→              where: { id: existing.id },
   614→   614→              data: updateData,
   615→   615→            })
   616→   616→          }
   617→   617→        } else {
   618→   618→          const created = await db.dREEntry.create({
   619→   619→            data: { companyId, month: m, year: y, ...updateData },
   620→   620→          })
   621→   621→          dreEntryId = created.id
   622→   622→        }
   623→   623→      } else {
   624→   624→        dreEntryId = existing?.id || null
   625→   625→      }
   626→   626→
   627→   627→      // Handle custom field totals (with sign correction)
   628→   628→      if (dreEntryId && Object.keys(customFieldTotals).length > 0) {
   629→   629→        const customFieldIds = Object.keys(customFieldTotals)
   630→   630→        const customFields = await db.customDREField.findMany({
   631→   631→          where: { id: { in: customFieldIds } },
   632→   632→        })
   633→   633→        const fieldSignMap: Record<string, string> = {}
   634→   634→        for (const cf of customFields) fieldSignMap[cf.id] = cf.sign
   635→   635→
   636→   636→        await Promise.all(
   637→   637→          Object.entries(customFieldTotals).map(([customFieldId, total]) => {
   638→   638→            const sign = fieldSignMap[customFieldId]
   639→   639→            const adjustedTotal = sign === '-' ? -total : total
   640→   640→            return db.customDREValue.upsert({
   641→   641→              where: { customFieldId_dREEntryId: { customFieldId, dREEntryId } },
   642→   642→              create: { customFieldId, dREEntryId, value: adjustedTotal },
   643→   643→              update: { value: adjustedTotal },
   644→   644→            })
   645→   645→          })
   646→   646→        )
   647→   647→      }
   648→   648→
   649→   649→      // Track the entry ID for the currently selected month
   650→   650→      if (m === month && y === year) {
   651→   651→        currentMonthDreEntryId = dreEntryId
   652→   652→      }
   653→   653→    }
   654→   654→
   655→   655→    // RF-07: Fetch the updated DRE entry to return for reactive sync
   656→   656→    let updatedDREData: Record<string, any> | null = null
   657→   657→    if (currentMonthDreEntryId) {
   658→   658→      const updatedEntry = await db.dREEntry.findUnique({ where: { id: currentMonthDreEntryId } })
   659→   659→      if (updatedEntry) {
   660→   660→        const { id, companyId: _cid, month: _m, year: _y, createdAt, updatedAt, company, ...fields } = updatedEntry as any
   661→   661→        updatedDREData = fields
   662→   662→
   663→   663→        // Also fetch custom field values
   664→   664→        const customValues = await db.customDREValue.findMany({
   665→   665→          where: { dREEntryId: currentMonthDreEntryId },
   666→   666→          include: { customField: true },
   667→   667→        })
   668→   668→        if (customValues.length > 0) {
   669→   669→          updatedDREData._customFieldValues = {}
   670→   670→          for (const cv of customValues) {
   671→   671→            const sign = (cv.customField as any).sign || '+'
   672→   672→            updatedDREData._customFieldValues[cv.customFieldId] = sign === '-' ? -cv.value : cv.value
   673→   673→          }
   674→   674→        }
   675→   675→      }
   676→   676→    }
   677→   677→
   678→   678→    return NextResponse.json({
   679→   679→      success: true,
   680→   680→      fieldTotals: currentMonthFieldTotals,
   681→   681→      dreEntryId: currentMonthDreEntryId,
   682→   682→      updatedDREData, // RF-07: reactive sync payload
   683→   683→    })
   684→   684→  } catch (error) {
   685→   685→    console.error('Error updating qualification:', error)
   686→   686→    return NextResponse.json({ error: 'Erro ao atualizar qualificação' }, { status: 500 })
   687→   687→  }
   688→   688→}

   729→   729→    const fileDeleteResult = await db.uploadedFile.deleteMany({
   730→   730→      where: { id: { in: ids } },
