    10→async function verifyAccess(userId: string, companyId: string) {
    11→  const company = await db.company.findUnique({ where: { id: companyId } })
    12→  if (!company) return false
    13→  if (company.userId === userId) return true
    14→  const share = await db.companyShare.findUnique({
    15→    where: { companyId_userId: { companyId, userId } },
    16→  })
    17→  return share?.status === 'accepted'
    18→}

    22→interface ExcelCellLike {
    23→  value: unknown
    24→  font?: { color?: { argb?: string } }
    25→}
    26→
    27→/**
    28→ * Detect if an Excel cell's font color is red.
    29→ * Red = alpha byte 0xFF (high byte), R > 150, G < 100, B < 100.
    30→ */
    31→function isRedFont(cell?: ExcelCellLike): boolean {
    32→  if (!cell?.font?.color?.argb) return false
    33→  const argb = cell.font.color.argb
    34→  if (argb.length < 8) return false
    35→  const alpha = parseInt(argb.substring(0, 2), 16)
    36→  if (alpha !== 0xFF) return false
    37→  const r = parseInt(argb.substring(2, 4), 16)
    38→  const g = parseInt(argb.substring(4, 6), 16)
    39→  const b = parseInt(argb.substring(6, 8), 16)
    40→  return r > 150 && g < 100 && b < 100
    41→}
    42→
    43→/**
    44→ * Parse any ExcelJS cell value into a number.
    45→ * Handles: number, formula, RichText, Hyperlink, string, Date.
    46→ * Parenthesized values like (780,00) are converted to negative.
    47→ * Red font color also makes the value negative.
    48→ */
    49→function parseCellToNumber(cellValue: unknown, cell?: ExcelCellLike): number {
    50→  let result = 0
    51→
    52→  if (typeof cellValue === 'number') {
    53→    result = cellValue
    54→  }
    55→  else if (cellValue && typeof cellValue === 'object' && 'result' in cellValue) {
    56→    const formulaResult = (cellValue as any).result
    57→    if (typeof formulaResult === 'number') result = formulaResult
    58→    else if (typeof formulaResult === 'string') result = parseTextToNumber(formulaResult)
    59→  }
    60→  else if (cellValue && typeof cellValue === 'object' && 'richText' in cellValue) {
    61→    const parts = (cellValue as any).richText as Array<{ text: string }>
    62→    const text = parts.map(p => p.text).join('')
    63→    result = parseTextToNumber(text)
    64→  }
    65→  else if (cellValue && typeof cellValue === 'object' && 'text' in cellValue && 'hyperlink' in cellValue) {
    66→    result = parseTextToNumber((cellValue as any).text)
    67→  }
    68→  else if (typeof cellValue === 'string') {
    69→    result = parseTextToNumber(cellValue)
    70→  }
    71→
    72→  if (result > 0 && isRedFont(cell)) {
    73→    result = -result
    74→  }
    75→
    76→  return result
    77→}
    78→
    79→/**
    80→ * Parse a text string like "1.234,56" or "(780,00)" into a number.
    81→ * Parenthesized values become negative.
    82→ */
    83→function parseTextToNumber(text: string): number {
    84→  const trimmed = text.trim()
    85→  if (!trimmed) return 0
    86→  const parenMatch = trimmed.match(/^\((.+)\)$/)
    87→  let numStr: string
    88→  let sign = 1
    89→  if (parenMatch) {
    90→    numStr = parenMatch[1]
    91→    sign = -1
    92→  } else {
    93→    numStr = trimmed
    94→    if (numStr.startsWith('-')) {
    95→      sign = -1
    96→      numStr = numStr.substring(1)
    97→    }
    98→  }
    99→  const cleaned = numStr.replace(/\./g, '').replace(',', '.')
   100→  const parsed = parseFloat(cleaned)
   101→  if (isNaN(parsed)) return 0
   102→  return sign * Math.abs(parsed)
   103→}

   107→function excelSerialToDate(serial: number): Date {
   108→  if (serial < 1) return new Date(NaN)
   109→  const epoch = new Date(1899, 11, 30)
   110→  const msPerDay = 24 * 60 * 60 * 1000
   111→  return new Date(epoch.getTime() + serial * msPerDay)
   112→}

   114→/** Remove accents/diacritics from a string for comparison */
   115→function normalizeAccents(str: string): string {
   116→  return str.normalize('NFD').replace(/[\u0300-\u036f]/g, '').toLowerCase().trim()
   117→}

   119→interface ParsedDateInfo {
   120→  month: number
   121→  year: number
   122→  dateStr: string
   123→}
   124→
   125→/**
   126→ * Parse a date value from an Excel cell (column B) into month/year.
   127→ * Supports: JS Date, DD/MM/YYYY, YYYY-MM-DD, MM/DD/YYYY, DD-MM-YYYY, Excel serial numbers.
   128→ */
   129→const MONTH_NAMES_PT: Record<string, number> = {
   130→  'janeiro': 1, 'fevereiro': 2, 'marco': 3, 'março': 3, 'abril': 4, 'maio': 5, 'junho': 6,
   131→  'julho': 7, 'agosto': 8, 'setembro': 9, 'outubro': 10, 'novembro': 11, 'dezembro': 12,
   132→  'jan': 1, 'fev': 2, 'mar': 3, 'abr': 4, 'mai': 5, 'jun': 6,
   133→  'jul': 7, 'ago': 8, 'set': 9, 'out': 10, 'nov': 11, 'dez': 12,
   134→}
   135→
   136→function parseRowDate(rawValue: unknown): ParsedDateInfo | null {
   137→  if (!rawValue) return null
   138→
   139→  if (rawValue instanceof Date && !isNaN(rawValue.getTime())) {
   140→    return {
   141→      month: rawValue.getMonth() + 1,
   142→      year: rawValue.getFullYear(),
   143→      dateStr: rawValue.toISOString().split('T')[0],
   144→    }
   145→  }
   146→
   147→  if (typeof rawValue === 'number' && rawValue > 25569 && rawValue < 100000) {
   148→    const d = excelSerialToDate(rawValue)
   149→    if (!isNaN(d.getTime())) {
   150→      return {
   151→        month: d.getMonth() + 1,
   152→        year: d.getFullYear(),
   153→        dateStr: d.toISOString().split('T')[0],
   154→      }
   155→    }
   156→  }
   157→
   158→  if (typeof rawValue === 'string') {
   159→    const s = rawValue.trim()
   160→    if (!s) return null
   161→
   162→    // YYYY-MM-DD
   163→    let m = s.match(/^(\d{4})-(\d{1,2})-(\d{1,2})/)
   164→    if (m) {
   165→      const yr = parseInt(m[1]); const mo = parseInt(m[2]); const dy = parseInt(m[3])
   166→      if (mo >= 1 && mo <= 12 && dy >= 1 && dy <= 31) {
   167→        return { month: mo, year: yr, dateStr: `${yr}-${String(mo).padStart(2, '0')}-${String(dy).padStart(2, '0')}` }
   168→      }
   169→    }
   170→
   171→    // DD/MM/YYYY or MM/DD/YYYY
   172→    m = s.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/)
   173→    if (m) {
   174→      const a = parseInt(m[1]); const b = parseInt(m[2]); const yr = parseInt(m[3])
   175→      let mo: number, dy: number
   176→      if (a > 12) { dy = a; mo = b }
   177→      else if (b > 12) { mo = a; dy = b }
   178→      else { dy = a; mo = b }
   179→      if (mo >= 1 && mo <= 12 && dy >= 1 && dy <= 31) {
   180→        return { month: mo, year: yr, dateStr: `${yr}-${String(mo).padStart(2, '0')}-${String(dy).padStart(2, '0')}` }
   181→      }
   182→    }
   183→
   184→    // DD-MM-YYYY
   185→    m = s.match(/^(\d{1,2})-(\d{1,2})-(\d{4})$/)
   186→    if (m) {
   187→      const a = parseInt(m[1]); const b = parseInt(m[2]); const yr = parseInt(m[3])
   188→      let mo: number, dy: number
   189→      if (a > 12) { dy = a; mo = b }
   190→      else if (b > 12) { mo = a; dy = b }
   191→      else { dy = a; mo = b }
   192→      if (mo >= 1 && mo <= 12 && dy >= 1 && dy <= 31) {
   193→        return { month: mo, year: yr, dateStr: `${yr}-${String(mo).padStart(2, '0')}-${String(dy).padStart(2, '0')}` }
   194→      }
   195→    }
   196→
   197→    // DD/MM/YY
   198→    m = s.match(/^(\d{1,2})\/(\d{1,2})\/(\d{2})$/)
   199→    if (m) {
   200→      const a = parseInt(m[1]); const b = parseInt(m[2])
   201→      let yr = parseInt(m[3])
   202→      yr = yr < 100 ? 2000 + yr : yr
   203→      let mo: number, dy: number
   204→      if (a > 12) { dy = a; mo = b }
   205→      else if (b > 12) { mo = a; dy = b }
   206→      else { dy = a; mo = b }
   207→      if (mo >= 1 && mo <= 12 && dy >= 1 && dy <= 31) {
   208→        return { month: mo, year: yr, dateStr: `${yr}-${String(mo).padStart(2, '0')}-${String(dy).padStart(2, '0')}` }
   209→      }
   210→    }
   211→
   212→    // Month name in Portuguese (e.g. "Janeiro 2025", "Jan/2025", "janeiro")
   213→    const normS = normalizeAccents(s)
   214→    for (const [monthName, monthNum] of Object.entries(MONTH_NAMES_PT)) {
   215→      const normName = normalizeAccents(monthName)
   216→      const regex = new RegExp(`\\b${normName}\\b`, 'i')
   217→      if (regex.test(normS)) {
   218→        // Try to extract year from the string
   219→        const yearMatch = s.match(/(\d{4})/)
   220→        const yr = yearMatch ? parseInt(yearMatch[1]) : new Date().getFullYear()
   221→        return { month: monthNum, year: yr, dateStr: `${yr}-${String(monthNum).padStart(2, '0')}-01` }
   222→      }
   223→    }
   224→
   225→    // MM/YYYY (just month and year)
   226→    m = s.match(/^(\d{1,2})\/(\d{4})$/)
   227→    if (m) {
   228→      const mo = parseInt(m[1]); const yr = parseInt(m[2])
   229→      if (mo >= 1 && mo <= 12) {
   230→        return { month: mo, year: yr, dateStr: `${yr}-${String(mo).padStart(2, '0')}-01` }
   231→      }
   232→    }
   233→  }
   234→
   235→  return null
   236→}

   289→/**
   290→ * RF-04: Detect if a description is a structural title and return its context.
   291→ * Returns { type: 'section' | 'field', fieldKey?, sectionKey? } or null.
   292→ */
   293→function detectStructuralTitle(description: string): { type: 'section' | 'field'; fieldKey?: string; sectionKey?: string } | null {
   294→  const norm = normalizeAccents(description)
   295→
   296→  // Check for section-level matches first
   297→  for (const entry of STRUCTURAL_TITLES) {
   298→    for (const pattern of entry.patterns) {
   299→      if (norm === normalizeAccents(pattern) || norm.includes(normalizeAccents(pattern))) {
   300→        if (entry.sectionKey && !entry.fieldKey) {
   301→          return { type: 'section', sectionKey: entry.sectionKey }
   302→        }
   303→        if (entry.fieldKey) {
   304→          return { type: 'field', fieldKey: entry.fieldKey, sectionKey: DRE_FIELD_SECTIONS[entry.fieldKey] }
   305→        }
   306→      }
   307→    }
   308→  }
   309→  return null
   310→}

   312→/**
   313→ * When a structural section title row has a value, map it to a reasonable DRE field.
   314→ * For receita section: maps to 'prestacaoNf' (main revenue field for law firms).
   315→ * For desp sections: maps to the first field in that section.
   316→ */
   317→function sectionTitleToField(sectionKey: string): string {
   318→  if (sectionKey === 'receita') return 'prestacaoNf'
   319→  // For expense sections, find the first field in that section
   320→  const sectionFields = Object.entries(DRE_FIELD_SECTIONS)
   321→    .filter(([, section]) => section === sectionKey)
   322→    .map(([key]) => key)
   323→  return sectionFields[0] || 'naoIdentificado'
   324→}

   326→/**
   327→ * RF-04: Given a section context, try to auto-classify a description into the first matching field in that section.
   328→ */
   329→function autoClassifyInSection(description: string, sectionKey: string): string | null {
   330→  const norm = normalizeAccents(description)
   331→  const sectionFields = Object.entries(DRE_FIELD_SECTIONS)
   332→    .filter(([, section]) => section === sectionKey)
   333→    .map(([key]) => key)
   334→
   335→  for (const fieldKey of sectionFields) {
   336→    const label = DRE_FIELD_LABELS[fieldKey]
   337→    if (!label) continue
   338→    const normLabel = normalizeAccents(label)
   339→    // Check if description contains field label keywords
   340→    const labelWords = normLabel.split(/\s+/).filter(w => w.length > 2)
   341→    for (const word of labelWords) {
   342→      if (norm.includes(word)) return fieldKey
   343→    }
   344→  }
   345→  return null
   346→}

   348→/**
   349→ * Try to match a category string from column D against DRE field labels.
   350→ */
   351→function classifyByCategory(category: string): string | null {
   352→  if (!category || !category.trim()) return null
   353→  const norm = normalizeAccents(category)
   354→
   355→  for (const key of Object.keys(DRE_FIELD_LABELS)) {
   356→    if (normalizeAccents(key) === norm) return key
   357→  }
   358→  for (const [key, value] of Object.entries(DRE_FIELD_LABELS)) {
   359→    if (normalizeAccents(value) === norm) return key
   360→  }
   361→  return null
   362→}

   364→// ─── POST: Upload spreadsheet ───
   365→
   366→export async function POST(req: NextRequest) {
   367→  try {
   368→    const session = await getServerSession(authOptions)
   369→    if (!session?.user?.id) {
   370→      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
   371→    }
   372→
   373→    const userId = (session.user as any).id
   374→    const formData = await req.formData()
   375→    const file = formData.get('file') as File
   376→    const companyId = formData.get('companyId') as string
   377→    const month = parseInt(formData.get('month') as string)
   378→    const year = parseInt(formData.get('year') as string)
   379→
   380→    if (!file || !companyId || !month || !year) {
   381→      return NextResponse.json({ error: 'Parâmetros incompletos' }, { status: 400 })
   382→    }
   383→
   384→    const hasAccess = await verifyAccess(userId, companyId)
   385→    if (!hasAccess) {
   386→      return NextResponse.json({ error: 'Sem permissão' }, { status: 403 })
   387→    }
   388→
   389→    const buffer = Buffer.from(await file.arrayBuffer())
   390→    const workbook = new ExcelJS.Workbook()
   391→    await workbook.xlsx.load(buffer)
   392→
   393→    if (workbook.worksheets.length === 0) {
   394→      return NextResponse.json({ error: 'Planilha vazia' }, { status: 400 })
   395→    }
   396→
   397→    // Row 3 is completely ignored during import — treated as null/nothing.
   398→    // Import starts from row 4 onward.
   399→    const SKIP_ROW_3 = true
   400→
   401→    const worksheet = workbook.worksheets[0]
   402→    if (!worksheet) {
   403→      return NextResponse.json({ error: 'Planilha vazia' }, { status: 400 })
   404→    }
   405→
   406→    // RF-04: Track structural title context during parsing
   407→    let currentSectionContext: string | null = null
   408→    let currentFieldContext: string | null = null
   409→
   410→    const parsedRows: Array<{
   411→      description: string
   412→      date: string
   413→      totalValue: number
   414→      itemMonth: number
   415→      itemYear: number
   416→      dreField: string
   417→    }> = []
   418→
   419→    let structuralTitlesFound = 0
   420→
   421→    worksheet.eachRow((row, rowNumber) => {
   422→      // Skip header row (1) and row 3 entirely
   423→      if (rowNumber === 1 || rowNumber === 3) return
   424→
   425→      // Column A: Description
   426→      const description = row.getCell(1).value?.toString()?.trim() || ''
   427→      if (!description) return
   428→
   429→      // Column B: Date → parse to get month/year
   430→      const rawDate = row.getCell(2).value
   431→      const parsedDate = parseRowDate(rawDate)
   432→      // If date is parsed, use it; otherwise default to selected month/year
   433→      const itemMonth = parsedDate?.month ?? month
   434→      const itemYear = parsedDate?.year ?? year
   435→      const dateStr = parsedDate?.dateStr ?? ''
   436→
   437→      // Column C: Value
   438→      const valueCell = row.getCell(3)
   439→      const value = parseCellToNumber(valueCell.value, { font: valueCell.font } as ExcelCellLike)
   440→
   441→      // RF-04: Check if this row is a structural title
   442→      const titleInfo = detectStructuralTitle(description)
   443→
   444→      if (titleInfo) {
   445→        structuralTitlesFound++
   446→        if (titleInfo.type === 'section') {
   447→          currentSectionContext = titleInfo.sectionKey ?? null
   448→          currentFieldContext = null
   449→          // Section-level titles (e.g. "Receita Bruta", "Despesas Variáveis") are headers,
   450→          // never added as data rows — their totals would cause double-counting.
   451→          return
   452→        } else if (titleInfo.type === 'field') {
   453→          currentFieldContext = titleInfo.fieldKey
   454→          currentSectionContext = titleInfo.sectionKey ?? null
   455→        }
   456→        // Field-level structural titles with no value are skipped
   457→        if (value === 0) return
   458→      }
   459→
   460→      // Column D: Category (explicit auto-classification, takes priority)
   461→      const categoryCell = row.getCell(4)
   462→      let dreField = ''
   463→      if (categoryCell && categoryCell.value) {
   464→        const categoryText = String(categoryCell.value).trim()
   465→        dreField = classifyByCategory(categoryText) || ''
   466→      }
   467→
   468→      // RF-04: If no category, use structural context for auto-classification
   469→      if (!dreField) {
   470→        if (currentFieldContext) {
   471→          dreField = currentFieldContext
   472→        } else if (titleInfo?.type === 'section' && titleInfo.sectionKey) {
   473→          // Section-level title with a value: map to a catch-all field in that section
   474→          // e.g. "Receita Bruta" with value → naoIdentificado
   475→          dreField = sectionTitleToField(titleInfo.sectionKey)
   476→        } else if (currentSectionContext) {
   477→          const autoField = autoClassifyInSection(description, currentSectionContext)
   478→          if (autoField) dreField = autoField
   479→        }
   480→      }
   481→
   482→      if (value !== 0) {
   483→        parsedRows.push({ description, date: dateStr, totalValue: value, itemMonth, itemYear, dreField })
   484→      }
   485→    })
   486→
   487→    if (parsedRows.length === 0) {
   488→      return NextResponse.json(
   489→        { error: 'Nenhum dado válido encontrado. Verifique: Descrição (A), Data (B), Valor (C).' },
   490→        { status: 400 }
   491→      )
   492→    }
   493→
   494→    // Create uploaded file record (use month=0, year=0 to indicate multi-month file)
   495→    const uploadedFile = await db.uploadedFile.create({
   496→      data: { companyId, month: 0, year: 0, fileName: file.name },
   497→    })
   498→
   499→    // No C3 Receita Bruta extraction — row 3 is ignored entirely
   500→    const receitaBrutaInserted = false
   501→    const receitaBrutaValue = 0
   502→    const receitaBrutaYear = year
   503→
   504→    // Save all other qualifications
   505→    const qualifications = parsedRows.map((item) => ({
   506→      companyId,
   507→      month: item.itemMonth,
   508→      year: item.itemYear,
   509→      description: item.description,
   510→      date: item.date,
   511→      totalValue: item.totalValue,
   512→      dreField: item.dreField,
   513→      uploadedFileId: uploadedFile.id,
   514→    }))
   515→
   516→    if (qualifications.length > 0) {
   517→      await db.expenseQualification.createMany({ data: qualifications })
   518→    }
   519→
   520→    const savedItems = await db.expenseQualification.findMany({
   521→      where: { uploadedFileId: uploadedFile.id },
   522→      orderBy: { createdAt: 'asc' },
   523→    })
   524→
   525→    // Build distributed months list
   526→    const groupedByMonth = new Map<string, number[]>()
   527→    for (const row of parsedRows) {
   528→      const key = `${row.itemMonth}-${row.itemYear}`
   529→      if (!groupedByMonth.has(key)) groupedByMonth.set(key, [])
   530→      groupedByMonth.get(key)!.push(row.itemMonth)
   531→    }
   532→    const distributedMonths = Array.from(groupedByMonth.keys())
   533→      .map(key => {
   534→        const [m, y] = key.split('-')
   535→        return `${String(parseInt(m)).padStart(2, '0')}/${y}`
   536→      })
   537→      .sort((a, b) => {
   538→        const [am, ay] = a.split('/').map(Number)
   539→        const [bm, by] = b.split('/').map(Number)
   540→        return ay !== by ? ay - by : am - bm
   541→      })
   542→
   543→    // Count auto-classified items
   544→    const autoClassified = savedItems.filter(i => i.dreField).length
   545→
   546→    // Auto-update DRE entries for months that have auto-classified items (especially C3/Receita Bruta)
   547→    const autoClassifiedItems = savedItems.filter(i => i.dreField)
   548→    const autoMonthKeys = new Set(autoClassifiedItems.map(i => `${i.month}-${i.year}`))
   549→    let updatedDREForMonths: Record<string, any> = {}
   550→    for (const monthKey of autoMonthKeys) {
   551→      const [mStr, yStr] = monthKey.split('-')
   552→      const m = parseInt(mStr)
   553→      const y = parseInt(yStr)
   554→      const monthItems = autoClassifiedItems.filter(i => i.month === m && i.year === y)
   555→      
   556→      const fieldTotals: Record<string, number> = {}
   557→      for (const item of monthItems) {
   558→        if (item.dreField && !item.dreField.startsWith('custom_')) {
   559→          // January: prestacaoNf must be manually entered — never from spreadsheet
   560→          if (m === 1 && item.dreField === 'prestacaoNf') continue
   561→          fieldTotals[item.dreField] = (fieldTotals[item.dreField] || 0) + item.totalValue
   562→        }
   563→      }
   564→
   565→      if (Object.keys(fieldTotals).length > 0) {
   566→        const existing = await db.dREEntry.findUnique({
   567→          where: { companyId_month_year: { companyId, month: m, year: y } },
   568→        })
   569→        if (existing) {
   570→          // Merge: add to existing values
   571→          const mergedData: Record<string, number> = {}
   572→          for (const [field, total] of Object.entries(fieldTotals)) {
   573→            mergedData[field] = ((existing as any)[field] || 0) + total
   574→          }
   575→          await db.dREEntry.update({
   576→            where: { id: existing.id },
   577→            data: mergedData,
   578→          })
   579→          // Fetch updated entry for response
   580→          const updated = await db.dREEntry.findUnique({ where: { id: existing.id } })
   581→          if (updated) {
   582→            const { id: _id, companyId: _c, month: _m2, year: _y2, createdAt, updatedAt, company, ...fields } = updated as any
   583→            updatedDREForMonths[monthKey] = fields
   584→          }
   585→        } else {
   586→          const created = await db.dREEntry.create({
   587→            data: { companyId, month: m, year: y, ...fieldTotals },
   588→          })
   589→          const { id: _id, companyId: _c, month: _m2, year: _y2, createdAt, updatedAt, company, ...fields } = created as any
   590→          updatedDREForMonths[monthKey] = fields
   591→        }
   592→      }
   593→    }
   594→
   595→    return NextResponse.json({
   596→      items: savedItems
   597→        .map(i => ({
   598→          id: i.id, description: i.description, date: i.date,
   599→          totalValue: i.totalValue, dreField: i.dreField,
   600→          month: i.month, year: i.year,
   601→        })),
   602→      file: { id: uploadedFile.id, fileName: uploadedFile.fileName, uploadedAt: uploadedFile.uploadedAt },
   603→      distributedMonths,
   604→      structuralTitlesFound,
   605→      autoClassified,
   606→      updatedDREForMonths,
   607→    })
   608→  } catch (error) {
   609→    console.error('Expense qualification error:', error)
   610→    return NextResponse.json({ error: 'Erro ao processar planilha de despesas' }, { status: 500 })
   611→  }
   612→}

   614→// ─── PUT: Save expense classification and update DRE (RF-07: returns updated DRE data) ───
   615→
   616→export async function PUT(req: NextRequest) {
   617→  try {
   618→    const session = await getServerSession(authOptions)
   619→    if (!session?.user?.id) {
   620→      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
   621→    }
   622→
   623→    const userId = (session.user as any).id
   624→    const body = await req.json()
   625→    const { companyId, month, year, items } = body
   626→
   627→    if (!companyId || !month || !year || !items) {
   628→      return NextResponse.json({ error: 'Parâmetros incompletos' }, { status: 400 })
   629→    }
   630→
   631→    const hasAccess = await verifyAccess(userId, companyId)
   632→    if (!hasAccess) {
   633→      return NextResponse.json({ error: 'Sem permissão' }, { status: 403 })
   634→    }
   635→
   636→    // Update each item's dreField mapping in parallel
   637→    const itemsToUpdate = items.filter((item: any) => item.id && item.dreField)
   638→    await Promise.all(
   639→      itemsToUpdate.map((item: any) =>
   640→        db.expenseQualification.update({
   641→          where: { id: item.id },
   642→          data: { dreField: item.dreField },
   643→        })
   644→      )
   645→    )
   646→
   647→    // Find all distinct month/year combos that have classified items for this company
   648→    const allClassified = await db.expenseQualification.findMany({
   649→      where: { companyId, dreField: { not: '' } },
   650→      select: { month: true, year: true },
   651→      distinct: ['month', 'year'],
   652→    })
   653→
   654→    // Update DRE entries for ALL months that have classified items
   655→    let currentMonthDreEntryId: string | null = null
   656→    let currentMonthFieldTotals: Record<string, number> = {}
   657→    for (const { month: m, year: y } of allClassified) {
   658→      const monthQualifications = await db.expenseQualification.findMany({
   659→        where: { companyId, month: m, year: y, dreField: { not: '' } },
   660→      })
   661→
   662→      const fieldTotals: Record<string, number> = {}
   663→      const customFieldTotals: Record<string, number> = {}
   664→      for (const q of monthQualifications) {
   665→        if (q.dreField) {
   666→          if (q.dreField.startsWith('custom_')) {
   667→            const customFieldId = q.dreField.replace('custom_', '')
   668→            customFieldTotals[customFieldId] = (customFieldTotals[customFieldId] || 0) + q.totalValue
   669→          } else {
   670→            // January: prestacaoNf must be manually entered — never from spreadsheet
   671→            if (m === 1 && q.dreField === 'prestacaoNf') continue
   672→            fieldTotals[q.dreField] = (fieldTotals[q.dreField] || 0) + q.totalValue
   673→          }
   674→        }
   675→      }
   676→
   677→      // Save field totals for the currently selected month
   678→      if (m === month && y === year) {
   679→        currentMonthFieldTotals = fieldTotals
   680→      }
   681→
   682→      const existing = await db.dREEntry.findUnique({
   683→        where: { companyId_month_year: { companyId, month: m, year: y } },
   684→      })
   685→
   686→      const updateData: Record<string, number> = {}
   687→      for (const [field, total] of Object.entries(fieldTotals)) {
   688→        updateData[field] = total
   689→      }
   690→
   691→      let dreEntryId: string | null = null
   692→      if (Object.keys(updateData).length > 0 || Object.keys(customFieldTotals).length > 0) {
   693→        if (existing) {
   694→          dreEntryId = existing.id
   695→          if (Object.keys(updateData).length > 0) {
   696→            await db.dREEntry.update({
   697→              where: { id: existing.id },
   698→              data: updateData,
   699→            })
   700→          }
   701→        } else {
   702→          const created = await db.dREEntry.create({
   703→            data: { companyId, month: m, year: y, ...updateData },
   704→          })
   705→          dreEntryId = created.id
   706→        }
   707→      } else {
   708→        dreEntryId = existing?.id || null
   709→      }
   710→
   711→      // Handle custom field totals (with sign correction)
   712→      if (dreEntryId && Object.keys(customFieldTotals).length > 0) {
   713→        const customFieldIds = Object.keys(customFieldTotals)
   714→        const customFields = await db.customDREField.findMany({
   715→          where: { id: { in: customFieldIds } },
   716→        })
   717→        const fieldSignMap: Record<string, string> = {}
   718→        for (const cf of customFields) fieldSignMap[cf.id] = cf.sign
   719→
   720→        await Promise.all(
   721→          Object.entries(customFieldTotals).map(([customFieldId, total]) => {
   722→            const sign = fieldSignMap[customFieldId]
   723→            const adjustedTotal = sign === '-' ? -total : total
   724→            return db.customDREValue.upsert({
   725→              where: { customFieldId_dREEntryId: { customFieldId, dREEntryId } },
   726→              create: { customFieldId, dREEntryId, value: adjustedTotal },
   727→              update: { value: adjustedTotal },
   728→            })
   729→          })
   730→        )
   731→      }
   732→
   733→      // Track the entry ID for the currently selected month
   734→      if (m === month && y === year) {
   735→        currentMonthDreEntryId = dreEntryId
   736→      }
   737→    }
   738→
   739→    // RF-07: Fetch the updated DRE entry to return for reactive sync
   740→    let updatedDREData: Record<string, any> | null = null
   741→    if (currentMonthDreEntryId) {
   742→      const updatedEntry = await db.dREEntry.findUnique({ where: { id: currentMonthDreEntryId } })
   743→      if (updatedEntry) {
   744→        const { id, companyId: _cid, month: _m, year: _y, createdAt, updatedAt, company, ...fields } = updatedEntry as any
   745→        updatedDREData = fields
   746→
   747→        // Also fetch custom field values
   748→        const customValues = await db.customDREValue.findMany({
   749→          where: { dREEntryId: currentMonthDreEntryId },
   750→          include: { customField: true },
   751→        })
   752→        if (customValues.length > 0) {
   753→          updatedDREData._customFieldValues = {}
   754→          for (const cv of customValues) {
   755→            const sign = (cv.customField as any).sign || '+'
   756→            updatedDREData._customFieldValues[cv.customFieldId] = sign === '-' ? -cv.value : cv.value
   757→          }
   758→        }
   759→      }
   760→    }
   761→
   762→    return NextResponse.json({
   763→      success: true,
   764→      fieldTotals: currentMonthFieldTotals,
   765→      dreEntryId: currentMonthDreEntryId,
   766→      updatedDREData, // RF-07: reactive sync payload
   767→    })
   768→  } catch (error) {
   769→    console.error('Error updating qualification:', error)
   770→    return NextResponse.json({ error: 'Erro ao atualizar qualificação' }, { status: 500 })
   771→  }
   772→}

   774→// ─── DELETE: Remove uploaded files ───

   776→export async function DELETE(req: NextRequest) {
   777→  try {
   778→    const session = await getServerSession(authOptions)
   779→    if (!session?.user?.id) {
   780→      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
   781→    }
   782→
   783→    const userId = (session.user as any).id
   784→    const { searchParams } = new URL(req.url)
   785→    const fileIdsParam = searchParams.get('fileIds')
   786→
   787→    if (!fileIdsParam) {
   788→      return NextResponse.json({ error: 'fileIds obrigatório' }, { status: 400 })
   789→    }
   790→
   791→    const ids = fileIdsParam.split(',').filter(Boolean)
   792→    if (ids.length === 0) {
   793→      return NextResponse.json({ error: 'Nenhum ID válido' }, { status: 400 })
   794→    }
   795→
   796→    const files = await db.uploadedFile.findMany({ where: { id: { in: ids } } })
   797→    const uniqueCompanyIds = [...new Set(files.map(f => f.companyId))]
   798→    const accessChecks = await Promise.all(
   799→      uniqueCompanyIds.map(cid => verifyAccess(userId, cid))
   800→    )
   801→    const accessMap = new Map(uniqueCompanyIds.map((cid, i) => [cid, accessChecks[i]]))
   802→
   803→    for (const file of files) {
   804→      if (!accessMap.get(file.companyId)) {
   805→        return NextResponse.json({ error: 'Sem permissão para excluir arquivo' }, { status: 403 })
   806→      }
   807→    }
   808→
   809→    const deleteResult = await db.expenseQualification.deleteMany({
   810→      where: { uploadedFileId: { in: ids } },
   811→    })
   812→
   813→    const fileDeleteResult = await db.uploadedFile.deleteMany({
   814→      where: { id: { in: ids } },
   815→    })
   816→
   817→    return NextResponse.json({ success: true, deletedFiles: fileDeleteResult.count, deletedItems: deleteResult.count })
   818→  } catch (error) {
   819→    console.error('Error deleting uploaded files:', error)
   820→    return NextResponse.json({ error: 'Erro ao excluir arquivos' }, { status: 500 })
   821→  }
   822→}

   826→export async function GET(req: NextRequest) {

   829→    if (!session?.user?.id) {
   830→      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
   831→    }
   832→

   834→    const { searchParams } = new URL(req.url)
   835→    const companyId = searchParams.get('companyId')
   836→    const month = searchParams.get('month')
   837→    const year = searchParams.get('year')
   838→
   839→    if (!companyId || !month || !year) {

   884→  }
