   144→// ─── Formatters ───
   145→function brl(value: number) {
   146→  return new Intl.NumberFormat('pt-BR', { style: 'currency', currency: 'BRL' }).format(value || 0)
   147→}
   148→function pct(value: number) {
   149→  if (!isFinite(value) || isNaN(value)) return '0,00%'
   150→  return new Intl.NumberFormat('pt-BR', { style: 'percent', minimumFractionDigits: 2, maximumFractionDigits: 2 }).format(value || 0)
   151→}
   152→function safeDiv(a: number, b: number) {
   153→  return b === 0 ? 0 : a / b
   154→}

   156→// ─── DRE Calculations ───
   157→function calcDRE(d: DREData) {
   158→  const receitaBruta = d.prestacaoNf + d.honorariosSucumbenciais
   159→  const receitaLiquida = receitaBruta - d.impostosCpp
   160→  const despesasVariaveis =
   161→    d.combustivel + d.insumos + d.materialEscritorio + d.servicosContratados +
   162→    d.materiaisUsoConsumo + d.taxas + d.tarifas + d.energiaEletrica + d.agua
   163→  const despesasFixas =
   164→    d.aluguel + d.salarios + d.fgts + d.prolabore + d.bonus +
   165→    d.inss + d.iptu + d.telefoneInternet + d.valeTransporte + d.planoSaude
   166→  const resultadoLiquido = receitaLiquida - despesasVariaveis - despesasFixas
   167→  return { receitaBruta, receitaLiquida, despesasVariaveis, despesasFixas, resultadoLiquido }
   168→}

   170→// ─── Number Input Component ───
   171→function NumInput({ label, value, onChange, prefix }: {
   172→  label: string; value: number; onChange: (v: number) => void; prefix?: string
   173→}) {
   174→  const [editing, setEditing] = useState(false)
   175→  const [local, setLocal] = useState('')
   176→
   177→  const formattedValue = value === 0 ? '' : brl(value)
   178→
   179→  const commit = () => {
   180→    setEditing(false)
   181→    const parsed = parseFloat(local.replace(/\./g, '').replace(',', '.')) || 0
   182→    onChange(parsed)
   183→  }
   184→
   185→  return (
   186→    <div className="flex items-center justify-between py-1.5 group">
   187→      <label className="text-sm text-muted-foreground flex items-center gap-2">
   188→        {label}
   189→      </label>
   190→      <div className="flex items-center gap-1 w-44 justify-end">
   191→        {prefix && <span className="text-xs text-muted-foreground">{prefix}</span>}
   192→        <Input
   193→          type="text"
   194→          inputMode="decimal"
   195→          className={`h-8 text-right text-sm font-mono ${editing ? 'ring-2 ring-primary/30' : ''} transition-all`}
   196→          value={editing ? local : (value === 0 ? '' : brl(value))}
   197→          onFocus={() => { setEditing(true); setLocal(value === 0 ? '' : value.toFixed(2).replace('.', ',')) }}
   198→          onBlur={commit}
   199→          onChange={(e) => {
   200→            setLocal(e.target.value)
   201→            const parsed = parseFloat(e.target.value.replace(/\./g, '').replace(',', '.'))
   202→            if (!isNaN(parsed)) onChange(parsed)
   203→          }}
   204→          onKeyDown={(e) => e.key === 'Enter' && commit()}
   205→          placeholder="0,00"
   206→        />
   207→      </div>
   208→    </div>
   209→  )
   210→}
