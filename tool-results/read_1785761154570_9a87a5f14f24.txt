   178→// ─── DRE Calculations ───
   179→function calcDRE(d: DREData, customFieldValues?: Record<string, number>, customFields?: CustomFieldDef[]) {
   180→  const receitaBruta = d.prestacaoNf + d.honorariosSucumbenciais + d.honorarios + d.reembolsoDespesas + d.naoIdentificado
   181→  const receitaLiquida = receitaBruta - d.impostosCpp
   182→  const despesasVariaveis =
   183→    d.combustivel + d.insumos + d.materialEscritorio + d.servicosContratados +
   184→    d.materiaisUsoConsumo + d.taxas + d.tarifas + d.energiaEletrica + d.agua
   185→  const despesasFixas =
   186→    d.aluguel + d.salarios + d.fgts + d.prolabore + d.bonus +
   187→    d.inss + d.iptu + d.telefoneInternet + d.valeTransporte + d.planoSaude
   188→
   189→  // Add custom field values
   190→  let customReceita = 0
   191→  let customDespVar = 0
   192→  let customDespFixas = 0
   193→  if (customFieldValues && customFields) {
   194→    for (const cf of customFields) {
   195→      const val = customFieldValues[cf.id] || 0
   196→      if (cf.sign === '+') {
   197→        if (cf.section === 'receita') customReceita += val
   198→        else if (cf.section === 'desp_variaveis') customDespVar += val
   199→        else if (cf.section === 'desp_fixas') customDespFixas += val
   200→      } else {
   201→        if (cf.section === 'receita') customReceita -= val
   202→        else if (cf.section === 'desp_variaveis') customDespVar -= val
   203→        else if (cf.section === 'desp_fixas') customDespFixas -= val
   204→      }
   205→    }
   206→  }
   207→
   208→  const totalReceitaBruta = receitaBruta + customReceita
   209→  const totalReceitaLiquida = totalReceitaBruta - d.impostosCpp
   210→  const totalDespesasVariaveis = despesasVariaveis + customDespVar
   211→  const totalDespesasFixas = despesasFixas + customDespFixas
   212→  const resultadoLiquido = totalReceitaLiquida - totalDespesasVariaveis - totalDespesasFixas
   213→  return { receitaBruta: totalReceitaBruta, receitaLiquida: totalReceitaLiquida, despesasVariaveis: totalDespesasVariaveis, despesasFixas: totalDespesasFixas, resultadoLiquido }
   214→}

   216→// ─── ProjNumInput Component ───
   217→function ProjNumInput({ value, onChange, className }: {
   218→  value: number; onChange: (v: number) => void; className?: string
   219→}) {
   220→  const [focused, setFocused] = useState(false)
   221→  const [local, setLocal] = useState('')
   222→  const inputRef = useRef<HTMLInputElement>(null)
   223→
   224→  const displayValue = focused
   225→    ? local
   226→    : (value === 0 ? '' : value.toFixed(2).replace('.', ','))
   227→
   228→  const handleFocus = () => {
   229→    setFocused(true)
   230→    setLocal(value === 0 ? '' : value.toFixed(2).replace('.', ','))
   231→    // Move cursor to start
   232→    setTimeout(() => {
   233→      if (inputRef.current) {
   234→        inputRef.current.setSelectionRange(0, 0)
   235→      }
   236→    }, 0)
   237→  }
   238→
   239→  const handleBlur = () => {
   240→    setFocused(false)
   241→    const parsed = parseFloat(local.replace(/\./g, '').replace(',', '.')) || 0
   242→    onChange(parsed)
   243→  }
   244→
   245→  const handleChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   246→    const val = e.target.value
   247→    // Allow only numbers, commas, dots
   248→    const cleaned = val.replace(/[^\d.,-]/g, '')
   249→    setLocal(cleaned)
   250→    const parsed = parseFloat(cleaned.replace(/\./g, '').replace(',', '.')) || 0
   251→    if (!isNaN(parsed)) onChange(parsed)
   252→  }
   253→
   254→  return (
   255→    <input
   256→      ref={inputRef}
   257→      type="text"
   258→      inputMode="decimal"
   259→      className={`w-28 text-right font-mono bg-white border rounded px-1.5 py-0.5 text-xs ${className || ''}`}
   260→      value={displayValue}
   261→      onFocus={handleFocus}
   262→      onBlur={handleBlur}
   263→      onChange={handleChange}
   264→    />
   265→  )
   266→}

   268→// ─── Number Input Component ───
   269→function NumInput({ label, value, onChange, prefix, isViewer, onSuggestChange, isSuggested, dreField }: {
   270→  label: string; value: number; onChange: (v: number) => void; prefix?: string
   271→  isViewer?: boolean; onSuggestChange?: (field: string, newValue: number) => void; isSuggested?: boolean; dreField?: string
   272→}) {
   273→  const [editing, setEditing] = useState(false)
   274→  const [local, setLocal] = useState('')
   275→
   276→  const commit = () => {
   277→    setEditing(false)
   278→    const parsed = parseFloat(local.replace(/\./g, '').replace(',', '.')) || 0
   279→    if (isViewer && onSuggestChange && dreField && parsed !== value) {
   280→      onSuggestChange(dreField, parsed)
   281→      return
   282→    }
   283→    onChange(parsed)
   284→  }
   285→
   286→  return (
   287→    <div className="flex items-center justify-between py-1.5 group">
   288→      <label className="text-sm text-muted-foreground flex items-center gap-2">
   289→        {isSuggested && <span className="h-2 w-2 rounded-full bg-amber-400" />}
   290→        {label}
   291→      </label>
   292→      <div className="flex items-center gap-1 w-44 justify-end">
   293→        {prefix && <span className="text-xs text-muted-foreground">{prefix}</span>}
   294→        <Input
   295→          type="text"
   296→          inputMode="decimal"
   297→          className={`h-8 text-right text-sm font-mono ${editing ? 'ring-2 ring-primary/30' : ''} ${isSuggested ? 'bg-amber-100 border-amber-300' : ''} transition-all`}
   298→          value={editing ? local : (value === 0 ? '' : brl(value))}
   299→          onFocus={() => { setEditing(true); setLocal(value === 0 ? '' : value.toFixed(2).replace('.', ',')) }}
   300→          onBlur={commit}
   301→          onChange={(e) => {
   302→            setLocal(e.target.value)
   303→            const parsed = parseFloat(e.target.value.replace(/\./g, '').replace(',', '.'))
   304→            if (!isNaN(parsed)) onChange(parsed)
   305→          }}
   306→          onKeyDown={(e) => e.key === 'Enter' && commit()}
   307→          placeholder="0,00"
   308→        />
   309→      </div>
   310→    </div>
   311→  )
   312→}

   337→function LoginScreen() {
   338→  const [activeLoginTab, setActiveLoginTab] = useState<'login' | 'register'>('login')
   339→  const [loginEmail, setLoginEmail] = useState('')
   340→  const [loginPassword, setLoginPassword] = useState('')
   341→  const [loginLoading, setLoginLoading] = useState(false)
   342→  const [regName, setRegName] = useState('')
   343→  const [regEmail, setRegEmail] = useState('')
   344→  const [regPassword, setRegPassword] = useState('')
   345→  const [regConfirm, setRegConfirm] = useState('')
   346→  const [regLoading, setRegLoading] = useState(false)
   347→  const handleLogin = async (e: React.FormEvent) => {
   348→    e.preventDefault()
   349→    if (!loginEmail.trim() || !loginPassword) return
   350→    setLoginLoading(true)
   351→    try {
   352→      const result = await signIn('credentials', {
   353→        email: loginEmail.trim(),
   354→        password: loginPassword,
   355→        redirect: false,
   356→      })
   357→      if (result?.error) {
   358→        toast.error('Email ou senha incorretos')
   359→      } else {
   360→        toast.success('Bem-vindo!')
   361→      }
   362→    } catch {
   363→      toast.error('Erro ao fazer login')
   364→    }
   365→    setLoginLoading(false)
   366→  }
   367→
   368→  const handleRegister = async (e: React.FormEvent) => {
   369→    e.preventDefault()
   370→    if (!regName.trim() || !regEmail.trim() || !regPassword || !regConfirm) {
   371→      toast.error('Preencha todos os campos')
   372→      return
   373→    }
   374→    if (regPassword !== regConfirm) {
   375→      toast.error('As senhas não coincidem')

   379→      toast.error('A senha deve ter pelo menos 6 caracteres')

   388→      })
   389→      if (res.ok) {
   390→        toast.success('Conta criada com sucesso! Faça login para continuar.')
   391→        setActiveLoginTab('login')
   392→        setLoginEmail(regEmail)
   393→        setRegName('')
   394→        setRegEmail('')
   395→        setRegPassword('')
   396→        setRegConfirm('')
   397→      } else {
   398→        const err = await res.json()
   399→        toast.error(err.error || 'Erro ao criar conta')
   400→      }
   401→    } catch {
   402→      toast.error('Erro ao criar conta')
   403→    }

   405→  }
   406→
   407→  return (

   965→        const res = await fetch(`/api/dre/export-company?companyId=${id}`)

   974→          URL.revokeObjectURL(url)
   975→          toast.success('Dados exportados com sucesso')
   976→        }
   977→      } catch {
   978→        toast.error('Erro ao exportar dados')
   979→      }

   981→    try {
   982→      await fetch(`/api/companies?id=${id}`, { method: 'DELETE' })

   988→      }

   990→      setUploadedFiles([])
   991→      toast.success('Empresa excluída')
   992→    } catch {
   993→      toast.error('Erro ao excluir empresa')

   996→    setDeleteSaveDialogOpen(false)
   997→    setDeletingCompanyId(null)

  1000→  const handleSaveDRE = async () => {

  1004→      const res = await fetch('/api/dre', {
  1005→        method: 'POST',
  1006→        headers: { 'Content-Type': 'application/json' },
  1007→        body: JSON.stringify({ companyId: selectedCompanyId, month: selectedMonth, year: selectedYear, ...dreData }),
  1008→      })
  1009→      if (res.ok) {
  1010→        // Save custom field values if we have an entry ID
  1011→        const savedData = await res.json()
  1012→        if (savedData?.id) {
  1013→          setDreEntryId(savedData.id)
  1014→          await saveCustomFieldValues(savedData.id, customFieldValues)
  1015→        }
  1016→        toast.success('DRE salvo com sucesso')
