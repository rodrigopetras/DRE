   524→  // ─── CNPJ mask ───
   525→  const formatCnpj = (value: string) => {
   526→    const digits = value.replace(/\D/g, '').slice(0, 14)
   527→    if (digits.length <= 2) return digits
   528→    if (digits.length <= 5) return `${digits.slice(0, 2)}.${digits.slice(2)}`
   529→    if (digits.length <= 8) return `${digits.slice(0, 2)}.${digits.slice(2, 5)}.${digits.slice(5)}`
   530→    if (digits.length <= 12) return `${digits.slice(0, 2)}.${digits.slice(2, 5)}.${digits.slice(5, 8)}/${digits.slice(8)}`
   531→    return `${digits.slice(0, 2)}.${digits.slice(2, 5)}.${digits.slice(5, 8)}/${digits.slice(8, 12)}-${digits.slice(12)}`
   532→  }
