   179→// ─── DRE Calculations ───
   180→function calcDRE(d: DREData, customFieldValues?: Record<string, number>, customFields?: CustomFieldDef[]) {
   181→  const receitaBruta = d.prestacaoNf + d.honorariosSucumbenciais + d.honorarios + d.reembolsoDespesas + d.naoIdentificado
   182→  const receitaLiquida = receitaBruta - d.impostosCpp
   183→  const despesasVariaveis =
   184→    d.combustivel + d.insumos + d.materialEscritorio + d.servicosContratados +
   185→    d.materiaisUsoConsumo + d.taxas + d.tarifas + d.energiaEletrica + d.agua
   186→  const despesasFixas =
   187→    d.aluguel + d.salarios + d.fgts + d.prolabore + d.bonus +
   188→    d.inss + d.iptu + d.telefoneInternet + d.valeTransporte + d.planoSaude
   189→
   190→  // Add custom field values
   191→  let customReceita = 0
   192→  let customDespVar = 0
   193→  let customDespFixas = 0
   194→  if (customFieldValues && customFields) {
   195→    for (const cf of customFields) {
   196→      const val = customFieldValues[cf.id] || 0
   197→      if (cf.sign === '+') {
   198→        if (cf.section === 'receita') customReceita += val
   199→        else if (cf.section === 'desp_variaveis') customDespVar += val
   200→        else if (cf.section === 'desp_fixas') customDespFixas += val
   201→      } else {
   202→        if (cf.section === 'receita') customReceita -= val
   203→        else if (cf.section === 'desp_variaveis') customDespVar -= val
   204→        else if (cf.section === 'desp_fixas') customDespFixas -= val
   205→      }
   206→    }
   207→  }
   208→
   209→  const totalReceitaBruta = receitaBruta + customReceita
   210→  const totalReceitaLiquida = totalReceitaBruta - d.impostosCpp
   211→  const totalDespesasVariaveis = despesasVariaveis + customDespVar
   212→  const totalDespesasFixas = despesasFixas + customDespFixas
   213→  const resultadoLiquido = totalReceitaLiquida - totalDespesasVariaveis - totalDespesasFixas
   214→  return { receitaBruta: totalReceitaBruta, receitaLiquida: totalReceitaLiquida, despesasVariaveis: totalDespesasVariaveis, despesasFixas: totalDespesasFixas, resultadoLiquido }
   215→}

   217→// ─── ProjNumInput Component ───
   218→function ProjNumInput({ value, onChange, className }: {
   219→  value: number; onChange: (v: number) => void; className?: string
   220→}) {
   221→  const [focused, setFocused] = useState(false)
   222→  const [local, setLocal] = useState('')
   223→  const inputRef = useRef<HTMLInputElement>(null)
   224→
   225→  const displayValue = focused
   226→    ? local
   227→    : (value === 0 ? '' : value.toFixed(2).replace('.', ','))
   228→
   229→  const handleFocus = () => {
   230→    setFocused(true)
   231→    setLocal(value === 0 ? '' : value.toFixed(2).replace('.', ','))
   232→    // Move cursor to start
   233→    setTimeout(() => {
   234→      if (inputRef.current) {
   235→        inputRef.current.setSelectionRange(0, 0)
   236→      }
   237→    }, 0)
   238→  }
   239→
   240→  const handleBlur = () => {
   241→    setFocused(false)
   242→    const parsed = parseFloat(local.replace(/\./g, '').replace(',', '.')) || 0
   243→    onChange(parsed)
   244→  }
   245→
   246→  const handleChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   247→    const val = e.target.value
   248→    // Allow only numbers, commas, dots
   249→    const cleaned = val.replace(/[^\d.,-]/g, '')
   250→    setLocal(cleaned)
   251→    const parsed = parseFloat(cleaned.replace(/\./g, '').replace(',', '.')) || 0
   252→    if (!isNaN(parsed)) onChange(parsed)
   253→  }
   254→
   255→  return (
   256→    <input
   257→      ref={inputRef}
   258→      type="text"
   259→      inputMode="decimal"
   260→      className={`w-28 text-right font-mono bg-white border rounded px-1.5 py-0.5 text-xs ${className || ''}`}
   261→      value={displayValue}
   262→      onFocus={handleFocus}
   263→      onBlur={handleBlur}
   264→      onChange={handleChange}
   265→    />
   266→  )
   267→}

   269→// ─── Number Input Component ───
   270→function NumInput({ label, value, onChange, prefix, isViewer, onSuggestChange, isSuggested, dreField }: {
   271→  label: string; value: number; onChange: (v: number) => void; prefix?: string
   272→  isViewer?: boolean; onSuggestChange?: (field: string, newValue: number) => void; isSuggested?: boolean; dreField?: string
   273→}) {
   274→  const [editing, setEditing] = useState(false)
   275→  const [local, setLocal] = useState('')
   276→
   277→  const commit = () => {
   278→    setEditing(false)
   279→    const parsed = parseFloat(local.replace(/\./g, '').replace(',', '.')) || 0
   280→    if (isViewer && onSuggestChange && dreField && parsed !== value) {
   281→      onSuggestChange(dreField, parsed)
   282→      return
   283→    }
   284→    onChange(parsed)
   285→  }
   286→
   287→  return (
   288→    <div className="flex items-center justify-between py-1.5 group">
   289→      <label className="text-sm text-muted-foreground flex items-center gap-2">
   290→        {isSuggested && <span className="h-2 w-2 rounded-full bg-amber-400" />}
   291→        {label}
   292→      </label>
   293→      <div className="flex items-center gap-1 w-44 justify-end">
   294→        {prefix && <span className="text-xs text-muted-foreground">{prefix}</span>}
   295→        <Input
   296→          type="text"
   297→          inputMode="decimal"
   298→          className={`h-8 text-right text-sm font-mono ${editing ? 'ring-2 ring-primary/30' : ''} ${isSuggested ? 'bg-amber-100 border-amber-300' : ''} transition-all`}
   299→          value={editing ? local : (value === 0 ? '' : brl(value))}
   300→          onFocus={() => { setEditing(true); setLocal(value === 0 ? '' : value.toFixed(2).replace('.', ',')) }}
   301→          onBlur={commit}
   302→          onChange={(e) => {
   303→            setLocal(e.target.value)
   304→            const parsed = parseFloat(e.target.value.replace(/\./g, '').replace(',', '.'))
   305→            if (!isNaN(parsed)) onChange(parsed)
   306→          }}
   307→          onKeyDown={(e) => e.key === 'Enter' && commit()}
   308→          placeholder="0,00"
   309→        />
   310→      </div>
   311→    </div>
   312→  )
   313→}
