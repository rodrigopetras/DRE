   116→   116→// ─── Formatters ───
   117→   117→function brl(value: number) {
   118→   118→  return new Intl.NumberFormat('pt-BR', { style: 'currency', currency: 'BRL' }).format(value || 0)
   119→   119→}
   120→   120→function pct(value: number) {
   121→   121→  if (!isFinite(value) || isNaN(value)) return '0,00%'
   122→   122→  return new Intl.NumberFormat('pt-BR', { style: 'percent', minimumFractionDigits: 2, maximumFractionDigits: 2 }).format(value || 0)
   123→   123→}
   124→   124→function safeDiv(a: number, b: number) {
   125→   125→  return b === 0 ? 0 : a / b
   126→   126→}

   128→   128→// ─── DRE Calculations ───
   129→   129→function calcDRE(d: DREData) {
   130→   130→  const receitaBruta = d.prestacaoNf + d.honorariosSucumbenciais
   131→   131→  const receitaLiquida = receitaBruta - d.impostosCpp
   132→   132→  const despesasVariaveis =
   133→   133→    d.combustivel + d.insumos + d.materialEscritorio + d.servicosContratados +
   134→   134→    d.materiaisUsoConsumo + d.taxas + d.tarifas + d.energiaEletrica + d.agua
   135→   135→  const despesasFixas =
   136→   136→    d.aluguel + d.salarios + d.fgts + d.prolabore + d.bonus +
   137→   137→    d.inss + d.iptu + d.telefoneInternet + d.valeTransporte + d.planoSaude
   138→   138→  const resultadoLiquido = receitaLiquida - despesasVariaveis - despesasFixas
   139→   139→  return { receitaBruta, receitaLiquida, despesasVariaveis, despesasFixas, resultadoLiquido }
   140→   140→}

   142→   142→// ─── Number Input Component ───
   143→   143→function NumInput({ label, value, onChange, prefix }: {
   144→   144→  label: string; value: number; onChange: (v: number) => void; prefix?: string
   145→   145→}) {
   146→   146→  const [editing, setEditing] = useState(false)
   147→   147→  const [local, setLocal] = useState('')
   148→   148→
   149→   149→  const formattedValue = value === 0 ? '' : brl(value)
   150→   150→
   151→   151→  const commit = () => {
   152→   152→    setEditing(false)
   153→   153→    const parsed = parseFloat(local.replace(/\./g, '').replace(',', '.')) || 0
   154→   154→    onChange(parsed)
   155→   155→  }
   156→   156→
   157→   157→  return (
   158→   158→    <div className="flex items-center justify-between py-1.5 group">
   159→   159→      <label className="text-sm text-muted-foreground flex items-center gap-2">
   160→   160→        {label}
   161→   161→      </label>
   162→   162→      <div className="flex items-center gap-1 w-44 justify-end">
   163→   163→        {prefix && <span className="text-xs text-muted-foreground">{prefix}</span>}
   164→   164→        <Input
   165→   165→          type="text"
   166→   166→          inputMode="decimal"
   167→   167→          className={`h-8 text-right text-sm font-mono ${editing ? 'ring-2 ring-primary/30' : ''} transition-all`}
   168→   168→          value={editing ? local : (value === 0 ? '' : brl(value))}
   169→   169→          onFocus={() => { setEditing(true); setLocal(value === 0 ? '' : value.toFixed(2).replace('.', ',')) }}
   170→   170→          onBlur={commit}
   171→   171→          onChange={(e) => {
   172→   172→            setLocal(e.target.value)
   173→   173→            const parsed = parseFloat(e.target.value.replace(/\./g, '').replace(',', '.'))
   174→   174→            if (!isNaN(parsed)) onChange(parsed)
   175→   175→          }}
   176→   176→          onKeyDown={(e) => e.key === 'Enter' && commit()}
   177→   177→          placeholder="0,00"
   178→   178→        />
   179→   179→      </div>
   180→   180→    </div>
   181→   181→  )
   182→   182→}
